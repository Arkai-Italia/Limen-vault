  775: pub fn is_high_precision_fs(path: &Path) -> bool {
  776:     use std::os::windows::ffi::OsStrExt;
  777:     use windows_sys::Win32::Storage::FileSystem::GetVolumeInformationW;
  778: 
  779:     let full = match path.canonicalize() {
  780:         Ok(p) => p,
  781:         Err(_) => path.to_path_buf(),
  782:     };
  783: 
  784:     let mut root_path_str = String::new();
  785:     let s = full.to_string_lossy();
  786:     if s.starts_with(r"\\?\") {
  787:         if s.len() >= 6 && s.as_bytes()[5] == b':' {
  788:             root_path_str = format!("{}\\", &s[4..6]);
  789:         }
  790:     } else if s.len() >= 2 && s.as_bytes()[1] == b':' {
  791:         root_path_str = format!("{}\\", &s[0..2]);
  792:     }
  793:     if root_path_str.is_empty() {
  794:         if let Some(prefix) = full.components().next() {
  795:             root_path_str = format!("{}\\", prefix.as_os_str().to_string_lossy().trim_end_matches('\\'));
  796:         }
  797:     }
  798:     if root_path_str.is_empty() {
  799:         return false;
  800:     }
  801: 
  802:     let wide_root: Vec<u16> = std::ffi::OsStr::new(&root_path_str)
  803:         .encode_wide()
  804:         .chain(std::iter::once(0))
  805:         .collect();
  806: 
  807:     let mut fs_name = [0u16; 260];
  808:     let res = unsafe {
  809:         GetVolumeInformationW(
  810:             wide_root.as_ptr(),
  811:             std::ptr::null_mut(),
  812:             0,
  813:             std::ptr::null_mut(),
  814:             std::ptr::null_mut(),

  817:             fs_name.len() as u32,

  819:     };

  822:         let len = fs_name.iter().position(|&c| c == 0).unwrap_or(fs_name.len());
  823:         let fs_str = String::from_utf16_lossy(&fs_name[..len]);
  824:         matches!(fs_str.to_uppercase().as_str(), "NTFS" | "REFS")
  825:     } else {
  826:         false
  827:     }
  828: }

  835: /// Verify source integrity using filesystem metadata on NTFS / high-precision filesystems,
  836: /// or full SHA-256 recalculation on FAT32/exFAT filesystems.
  837: pub fn verify_source_integrity_with_fs_override(
  838:     path: &Path,
  839:     s: &Source,
  840:     is_high_precision: bool,
  841:     drafts: bool,
  842: ) -> Result<(), String> {
  843:     let full_path = path.join(&s.relative_path);
  844:     let meta = std::fs::metadata(&full_path).map_err(|_| "Source file missing or inaccessible".to_string())?;
  845: 
  846:     // In-memory checks on search index, catalog, automation state, and eligibility
  847:     let d = search::get_search_document(path, &s.document_id)?;
  848:     if d.sha256 != s.sha256 {
  849:         return Err("Source changed".into());
  850:     }
  851: 
  852:     // 1. Check if automation-managed source has become obsolete
  853:     if crate::automation::managed(&d.relative_path) && !crate::automation::is_current(path, &d.relative_path, &s.sha256) {
  854:         return Err("Generated source obsolete or modified".into());
  855:     }
  856: 
  857:     // 2. Check document eligibility (category, status, drafts)
  858:     if !eligible(&d.relative_path, &d.category, d.status.as_deref(), drafts)
  859:         && !crate::automation::is_current(path, &d.relative_path, &d.sha256)
  860:     {
  861:         return Err("Document access denied".into());
  862:     }
  863: 
  864:     // 3. Verifica crittografica puntuale di ciascun singolo passaggio associato alla fonte (Punto C)
  865:     for (pid, expected_sha) in &s.passage_hashes {
  866:         let passage = crate::catalog::read_passage(path, &s.document_id, pid)
  867:             .map_err(|e| format!("Integrità passaggio {} compromessa: {}", pid, e))?;
  868:         if &passage.sha256 != expected_sha {
  869:             return Err(format!(
  870:                 "Integrità passaggio {} compromessa (hash atteso {}, trovato {})",
  871:                 pid, expected_sha, passage.sha256
  872:             ));
  873:         }
  874:     }
  875: 
  876:     if is_high_precision {
  877:         let current_size = meta.len();
  878:         let current_mtime_ms = meta
  879:             .modified()
  880:             .ok()
  881:             .and_then(|t| t.duration_since(std::time::UNIX_EPOCH).ok())
  882:             .map(|dur| dur.as_millis() as u64);
  883: 
  884:         if let (Some(expected_size), Some(expected_mtime)) = (s.file_size, s.mtime_ms) {
  885:             if current_size == expected_size && current_mtime_ms == Some(expected_mtime) {
  886:                 // In NTFS with nanosecond resolution, unmodified mtime + size guarantees content integrity
  887:                 return Ok(());
  888:             }
  889:         }
  890:     }
  891: 
  892:     // On non-NTFS (FAT32/exFAT with 2-second timestamp resolution) or if metadata changed,
  893:     // recalculate SHA-256 using in-memory catalog and search index cache.
  894:     let current = read_source(path, &s.document_id, &s.sha256, drafts)?;
  895:     if current.sha256 != s.sha256 {
  896:         return Err("Source changed".into());
  897:     }
  898:     Ok(())
  899: }

  901: pub fn verify_source_integrity(path: &Path, s: &Source, drafts: bool) -> Result<(), String> {

  903:     verify_source_integrity_with_fs_override(path, s, high_prec, drafts)

  935: /// - Contiene almeno una parola rara della query (stessa regola di rarità del Punto A: df <= 2 || df/N <= 0.30)
  936: ///   utilizzando tokenizzazione esatta sui token indicizzati o confronto per parole intere (contains_whole_words);
  937: /// - OPPURE la sua similarità semantica dista dal massimo della domanda non più di PUNTO_E_SEM_DELTA_THRESHOLD (0.05).
  938: pub fn is_candidate_admitted(
  939:     item: &search::SearchResultItem,
  940:     rare_query_tokens: &[String],
  941:     search_idx: Option<&search::SearchIndexData>,
  942:     max_sem_sim: f64,
  943: ) -> bool {
  944:     // 1. Regola parola rara (con token esatti / parole intere, senza corrispondenza di sole sottostringhe)
  945:     let has_rare_word = if let Some(idx) = search_idx {
  946:         if let Some(doc) = idx.documents.get(&item.relative_path).or_else(|| idx.documents.values().find(|d| d.id == item.id)) {
  947:             rare_query_tokens.iter().any(|rt| doc.tokens.iter().any(|dt| dt == rt))
  948:         } else {
  949:             let norm_title = crate::search::normalize_text(&item.title);
  950:             let norm_snippet = crate::search::normalize_text(&item.snippet);
  951:             rare_query_tokens.iter().any(|rt| {
  952:                 let norm_rt = crate::search::normalize_text(rt);
  953:                 crate::search::contains_whole_words(&norm_title, &norm_rt)
  954:                     || crate::search::contains_whole_words(&norm_snippet, &norm_rt)
  955:             })
  956:         }
  957:     } else {
  958:         let norm_title = crate::search::normalize_text(&item.title);
  959:         let norm_snippet = crate::search::normalize_text(&item.snippet);
  960:         rare_query_tokens.iter().any(|rt| {
  961:             let norm_rt = crate::search::normalize_text(rt);
  962:             crate::search::contains_whole_words(&norm_title, &norm_rt)
  963:                 || crate::search::contains_whole_words(&norm_snippet, &norm_rt)
  964:         })
  965:     };
  966: 
  967:     // 2. Similarità semantica dista dal massimo della query non più di 0.05
  968:     let is_close_to_max = if let Some(sim) = item.semantic_similarity {
  969:         max_sem_sim > 0.0 && (max_sem_sim - sim) <= PUNTO_E_SEM_DELTA_THRESHOLD
  970:     } else {
  971:         false
  972:     };
  973: 
  974:     has_rare_word || is_close_to_max
  975: }

  977: /// Applica la selezione del Metodo c (Punto E) a un insieme di candidati:
  978: /// - Se il servizio semantico è spento o non disponibile (`has_semantic == false`), non filtra e restituisce tutti i candidati;
  979: /// - Se il servizio semantico è attivo, ammette i candidati con parola rara o delta semantico <= 0.05 dal massimo;
  980: /// - Se nessun candidato è ammesso, restituisce comunque il primo della classifica come fallback.
  981: pub fn filter_candidates_punto_e<'a>(
  982:     rows: &'a [search::SearchResultItem],
  983:     rare_query_tokens: &[String],
  984:     search_idx: Option<&search::SearchIndexData>,
  985:     has_semantic: bool,
  986: ) -> Vec<&'a search::SearchResultItem> {
  987:     if !has_semantic {
  988:         return rows.iter().collect();
  989:     }
  990:     let max_sem_sim = rows
  991:         .iter()
  992:         .filter_map(|r| r.semantic_similarity)
  993:         .fold(0.0f64, f64::max);
  994: 
  995:     let mut filtered: Vec<&search::SearchResultItem> = rows
  996:         .iter()
  997:         .filter(|r| is_candidate_admitted(r, rare_query_tokens, search_idx, max_sem_sim))
  998:         .collect();
  999: 
 1000:     // Se nessun candidato è ammesso dal filtro ma la semantica era attiva, si invia comunque il primo della classifica
 1001:     if filtered.is_empty() {
 1002:         if let Some(first) = rows.first() {
 1003:             filtered.push(first);
 1004:         }
 1005:     }
 1006:     filtered
 1007: }

 1009: /// Riconosce il tipo di localizzatore e l'intervallo numerico coperto (start..=end).
 1010: /// Supporta:
 1011: /// - "Paragrafo 4" -> ("p", 4, 4)
 1012: /// - "Paragrafi 1-4" -> ("p", 1, 4)
 1013: /// - "Pagina 1" -> ("page", 1, 1)
 1014: /// - "Pagine 1-3" -> ("page", 1, 3)
 1015: /// - "Slide 2" -> ("slide", 2, 2)
 1016: /// - "Slides 2-5" -> ("slide", 2, 5)
 1017: pub fn parse_locator_range(loc: &str) -> Option<(String, u32, u32)> {
 1018:     let lower = loc.trim().to_lowercase();
 1019:     let (prefix, rest) = if let Some(r) = lower.strip_prefix("paragrafi ") {
 1020:         ("p", r)
 1021:     } else if let Some(r) = lower.strip_prefix("paragrafo ") {
 1022:         ("p", r)
 1023:     } else if let Some(r) = lower.strip_prefix("pagine ") {
 1024:         ("page", r)
 1025:     } else if let Some(r) = lower.strip_prefix("pagina ") {
 1026:         ("page", r)
 1027:     } else if let Some(r) = lower.strip_prefix("slides ") {
 1028:         ("slide", r)
 1029:     } else if let Some(r) = lower.strip_prefix("slide ") {
 1030:         ("slide", r)
 1031:     } else {
 1032:         return None;
 1033:     };
 1034: 
 1035:     if let Some((start_s, end_s)) = rest.split_once('-') {
 1036:         let start = start_s.trim().parse::<u32>().ok()?;
 1037:         let end = end_s.trim().parse::<u32>().ok()?;
 1038:         Some((prefix.to_string(), start.min(end), start.max(end)))
 1039:     } else if let Ok(num) = rest.trim().parse::<u32>() {
 1040:         Some((prefix.to_string(), num, num))
 1041:     } else {
 1042:         None
 1043:     }
 1044: }

 1046: /// Verifica se due localizzatori condividono paragrafi (o pagine/slide) in comune.
 1047: /// Evita la duplicazione di testo nello stesso documento quando un passaggio è contenuto
 1048: /// o intersecato da un altro (es. "Paragrafo 224" vs "Paragrafi 224-234").
 1049: pub fn locators_overlap(loc1: &str, loc2: &str) -> bool {
 1050:     if loc1.trim().eq_ignore_ascii_case(loc2.trim()) {
 1051:         return true;
 1052:     }
 1053:     if let (Some((p1, s1, e1)), Some((p2, s2, e2))) = (parse_locator_range(loc1), parse_locator_range(loc2)) {
 1054:         if p1 == p2 {
 1055:             return s1.max(s2) <= e1.min(e2);
 1056:         }
 1057:     }
 1058:     false
 1059: }

 1061: pub fn contains_case_insensitive_fast(haystack: &str, needle_lower: &str) -> bool {
 1062:     if needle_lower.is_empty() {
 1063:         return true;
 1064:     }
 1065:     let needle_bytes = needle_lower.as_bytes();
 1066:     let haystack_bytes = haystack.as_bytes();
 1067:     if needle_bytes.len() > haystack_bytes.len() {
 1068:         return false;
 1069:     }
 1070:     haystack_bytes.windows(needle_bytes.len()).any(|window| {
 1071:         window.iter().zip(needle_bytes.iter()).all(|(h, n)| {
 1072:             h.to_ascii_lowercase() == *n
 1073:         })
 1074:     })
 1075: }

 1077: /// Seleziona 1–3 passaggi più pertinenti per il documento, verificando l'integrità
 1078: /// crittografica di ciascun passaggio con il rispettivo SHA-256 (Punti C & D di fase3b-piano.md).
 1079: /// Esclude categoricamente passaggi con paragrafi in comune con quelli già scelti.
 1080: /// Utilizza la similarità semantica dei passaggi da cache e normalizza i testi una sola volta.
 1081: /// Restituisce (content, locator_concatenato, passage_hashes).
 1082: pub fn extract_multi_passages_for_document(
 1083:     doc: &crate::catalog::DocumentRecord,
 1084:     primary_passage_id: Option<&str>,
 1085:     matching_passages: &[search::SearchMatchingPassage],
 1086:     query_tokens: &[String],
 1087:     rare_query_tokens: &[String],
 1088:     max_passages: usize,
 1089:     max_bytes: usize,
 1090:     query_vector: Option<&[f32]>,
 1091:     embeddings_cache: Option<&crate::embeddings::EmbeddingsCache>,
 1092: ) -> (String, Option<String>, Vec<(String, String)>) {
 1093:     if doc.passages.is_empty() {
 1094:         return (String::new(), None, Vec::new());
 1095:     }
 1096: 
 1097:     if doc.passages.len() == 1 {
 1098:         let p = &doc.passages[0];
 1099:         let actual_hash = crate::snapshots::compute_sha256(p.text.as_bytes());
 1100:         if actual_hash != p.sha256 {
 1101:             return (String::new(), None, Vec::new());
 1102:         }
 1103:         let block = format!("[{}]\n{}", p.locator, p.text);
 1104:         let content = if block.len() > max_bytes {
 1105:             block.chars().take(max_bytes).collect()
 1106:         } else {
 1107:             block
 1108:         };
 1109:         return (content, Some(p.locator.clone()), vec![(p.passage_id.clone(), p.sha256.clone())]);
 1110:     }
 1111: 
 1112:     // Identifica l'indice del passaggio primario (top ranking da matching semantico/lessicale)
 1113:     let primary_idx = primary_passage_id
 1114:         .and_then(|pid| doc.passages.iter().position(|p| p.passage_id == pid))
 1115:         .unwrap_or(0);
 1116: 
 1117:     let mut selected_indices = vec![primary_idx];
 1118:     let primary_locator = doc.passages[primary_idx].locator.clone();
 1119: 
 1120:     // Se possiamo selezionare ulteriori passaggi (fino a max_passages, tipicamente 3)
 1121:     if max_passages > 1 && doc.passages.len() > 1 {
 1122:         // Pre-normalizzazione dei token una sola volta per l'intera selezione (ottimizzazione tempo)
 1123:         let norm_rare_tokens: Vec<String> = rare_query_tokens.iter().map(|t| crate::search::normalize_text(t)).collect();
 1124:         let norm_rare_tokens_lower: Vec<String> = rare_query_tokens.iter().map(|t| t.to_ascii_lowercase()).collect();
 1125:         let norm_query_tokens: Vec<String> = query_tokens.iter().map(|t| crate::search::normalize_text(t)).collect();
 1126: 
 1127:         let mut scored_candidates: Vec<(usize, f64)> = Vec::new();
 1128:         for (i, p) in doc.passages.iter().enumerate() {
 1129:             if i == primary_idx {
 1130:                 continue;
 1131:             }
 1132: 
 1133:             // ESCLUSIONE IMMEDIATA: se condivide paragrafi con il primario o ha stesso ID, salta
 1134:             if locators_overlap(&p.locator, &primary_locator) || p.passage_id == doc.passages[primary_idx].passage_id {
 1135:                 continue;
 1136:             }
 1137: 
 1138:             let dist = (i as isize - primary_idx as isize).abs();
 1139:             let is_close = dist > 0 && dist <= 3;
 1140:             let is_first = i == 0;
 1141:             let mp_opt = matching_passages.iter().find(|mp| mp.passage_id == p.passage_id);
 1142: 
 1143:             // Filtro preliminare rapido per documenti con molti passaggi (ottimizzazione tempo):
 1144:             // In documenti molto grandi (> 50 passaggi), valuta solo passaggi vicini, primo, con match lessicale o parola rara
 1145:             let has_rare_keyword = if !is_close && !is_first && mp_opt.is_none() {
 1146:                 if doc.passages.len() > 50 {
 1147:                     norm_rare_tokens_lower.iter().any(|t| t.len() >= 3 && contains_case_insensitive_fast(&p.text, t))
 1148:                 } else {
 1149:                     true
 1150:                 }
 1151:             } else {
 1152:                 false
 1153:             };
 1154: 
 1155:             if !is_close && !is_first && mp_opt.is_none() && !has_rare_keyword {
 1156:                 continue;
 1157:             }
 1158: 
 1159:             // Calcolo similarità semantica del passaggio solo per i candidati pre-filtrati
 1160:             let mut sem_sim_opt = None;
 1161:             if let (Some(q_vec), Some(emb_cache)) = (query_vector, embeddings_cache) {
 1162:                 if let Some(entry) = emb_cache.entries.get(&p.passage_id) {
 1163:                     if entry.sha256 == p.sha256 {
 1164:                         let sim = crate::embeddings::cosine_similarity(q_vec, &entry.vector) as f64;
 1165:                         sem_sim_opt = Some(sim);
 1166:                     }
 1167:                 }
 1168:             }
 1169: 
 1170:             let mut score = 0.0f64;
 1171: 
 1172:             // 1. Similarità semantica dei passaggi già presente nella cache vettoriale (dal piano)
 1173:             if let Some(sim) = sem_sim_opt {
 1174:                 score += sim * 35.0;
 1175:             }
 1176: 
 1177:             // 2. Punteggio da matching_passages (ricerca lessicale)
 1178:             if let Some(mp) = mp_opt {
 1179:                 score += 40.0 + mp.score * 5.0;
 1180:             }
 1181: 
 1182:             // 3. Corrispondenza con token rari e comuni (normalizzazione passaggio eseguita solo sui candidati plausibili)
 1183:             let norm_p_text = crate::search::normalize_text(&p.text);
 1184:             for norm_rt in &norm_rare_tokens {
 1185:                 if crate::search::contains_whole_words(&norm_p_text, norm_rt) {
 1186:                     score += 25.0;
 1187:                 }
 1188:             }
 1189:             for norm_qt in &norm_query_tokens {
 1190:                 if crate::search::contains_whole_words(&norm_p_text, norm_qt) {
 1191:                     score += 5.0;
 1192:                 }
 1193:             }
 1194: 
 1195:             // 4. Bonus di vicinanza ridotto (leggero tie-breaker per dist <= 3, non dominante)
 1196:             if is_close {
 1197:                 score += 3.0 / (dist as f64);
 1198:             }
 1199: 
 1200:             // 5. Inquadramento iniziale del documento se non sovrapposto
 1201:             if is_first {
 1202:                 score += 8.0;
 1203:             }
 1204: 
 1205:             scored_candidates.push((i, score));
 1206:         }
 1207: 
 1208:         scored_candidates.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap_or(std::cmp::Ordering::Equal));
 1209: 
 1210:         for (cand_idx, _) in scored_candidates {
 1211:             let cand_loc = &doc.passages[cand_idx].locator;
 1212:             // Verifica che il candidato non si sovrapponga a NESSUNO dei passaggi già selezionati
 1213:             let has_overlap = selected_indices.iter().any(|&s_idx| {
 1214:                 locators_overlap(cand_loc, &doc.passages[s_idx].locator)
 1215:                     || doc.passages[cand_idx].passage_id == doc.passages[s_idx].passage_id
 1216:             });
 1217:             if !has_overlap {
 1218:                 selected_indices.push(cand_idx);
 1219:                 if selected_indices.len() >= max_passages {
 1220:                     break;
 1221:                 }
 1222:             }
 1223:         }
 1224:     }
 1225: 
 1226:     // Ordina i passaggi selezionati secondo l'ordine sequenziale nel documento originale
 1227:     selected_indices.sort_unstable();
 1228: 
 1229:     let mut blocks = Vec::new();
 1230:     let mut locators = Vec::new();
 1231:     let mut hashes = Vec::new();
 1232:     let mut current_bytes = 0;
 1233: 
 1234:     for &idx in &selected_indices {
 1235:         let p = &doc.passages[idx];
 1236:         let actual_hash = crate::snapshots::compute_sha256(p.text.as_bytes());
 1237:         if actual_hash != p.sha256 {
 1238:             continue;
 1239:         }
 1240: 
 1241:         let block_text = format!("[{}]\n{}", p.locator, p.text);
 1242:         let additional_bytes = block_text.len() + (if blocks.is_empty() { 0 } else { 2 });
 1243: 
 1244:         if current_bytes + additional_bytes <= max_bytes || blocks.is_empty() {
 1245:             let final_block = if current_bytes + additional_bytes > max_bytes {
 1246:                 let allowed = max_bytes.saturating_sub(current_bytes);
 1247:                 block_text.chars().take(allowed).collect::<String>()
 1248:             } else {
 1249:                 block_text
 1250:             };
 1251:             current_bytes += final_block.len() + 2;
 1252:             blocks.push(final_block);
 1253:             locators.push(p.locator.clone());
 1254:             hashes.push((p.passage_id.clone(), p.sha256.clone()));
 1255:         }
 1256:     }
 1257: 
 1258:     let content = blocks.join("\n\n");
 1259:     let locator = if locators.is_empty() {
 1260:         None
 1261:     } else {
 1262:         Some(locators.join(", "))
 1263:     };
 1264: 
 1265:     (content, locator, hashes)
 1266: }

 1747: /// Pulisce il testo della risposta generata dal provider:
 1748: /// - Rimuove citazioni sintetiche residue nel corpo ([S1], (S1), ecc.);

 1750: /// - Rimuove marcatori di grassetto markdown (`**testo**` -> `testo`);
 1751: /// - Rimuove marcatori di corsivo con asterischi (`*testo*` -> `testo`);
 1752: /// - Preserva gli asterischi letterali/matematici (es. `2*3`).
 1753: /// Normalizza spazi multipli e punteggiatura orfana residua.
 1754: pub fn sanitize_answer_prose(text: &str) -> String {
 1755:     use std::sync::OnceLock;
 1756:     static RE_CITATIONS: OnceLock<regex::Regex> = OnceLock::new();
 1757:     static RE_SPACE_PUNCT: OnceLock<regex::Regex> = OnceLock::new();
 1758:     static RE_MULTI_SPACE: OnceLock<regex::Regex> = OnceLock::new();
 1759:     static RE_EMPTY_PARENS: OnceLock<regex::Regex> = OnceLock::new();
 1760:     static RE_BULLET: OnceLock<regex::Regex> = OnceLock::new();
 1761:     static RE_BOLD: OnceLock<regex::Regex> = OnceLock::new();
 1762:     static RE_ITALIC: OnceLock<regex::Regex> = OnceLock::new();
 1763: 
 1764:     let re_citations = RE_CITATIONS.get_or_init(|| {
 1765:         regex::Regex::new(r"(?i)\[\s*S\d+\s*(?:[,;]\s*S\d+\s*)*\]|\(\s*S\d+\s*(?:[,;]\s*S\d+\s*)*\)").unwrap()
 1766:     });
 1767:     let re_bullet = RE_BULLET.get_or_init(|| {
 1768:         regex::Regex::new(r"(?m)^([ \t]*)\*[ \t]+").unwrap()
 1769:     });
 1770:     let re_bold = RE_BOLD.get_or_init(|| {
 1771:         regex::Regex::new(r"\*\*([^*]+)\*\*").unwrap()
 1772:     });
 1773:     let re_italic = RE_ITALIC.get_or_init(|| {
 1774:         regex::Regex::new(r"\*([^\s*](?:[^*\n]*?[^\s*])?)\*").unwrap()
 1775:     });
 1776:     let re_space_punct = RE_SPACE_PUNCT.get_or_init(|| {
 1777:         regex::Regex::new(r"[ \t]+([.,;:!?])").unwrap()
 1778:     });
 1779:     let re_multi_space = RE_MULTI_SPACE.get_or_init(|| {
 1780:         regex::Regex::new(r"[ \t]{2,}").unwrap()
 1781:     });
 1782:     let re_empty_parens = RE_EMPTY_PARENS.get_or_init(|| {
 1783:         regex::Regex::new(r"\(\s*\)|\[\s*\]").unwrap()
 1784:     });
 1785: 
 1786:     // 1. Rimuovi citazioni [S1], (S1), ecc.
 1787:     let step1 = re_citations.replace_all(text, "");
 1788: 
 1789:     // 2. Converte elenchi puntati con asterisco in trattini: "* elemento" -> "- elemento"
 1790:     let step2 = re_bullet.replace_all(&step1, "$1- ");
 1791: 
 1792:     // 3. Rimuove grassetti markdown con doppi asterischi: "**testo**" -> "testo"
 1793:     let mut step3 = step2.to_string();
 1794:     while re_bold.is_match(&step3) {
 1795:         step3 = re_bold.replace_all(&step3, "$1").to_string();
 1796:     }
 1797: 
 1798:     // 4. Rimuove corsivi markdown con singoli asterischi: "*testo*" -> "testo"
 1799:     while re_italic.is_match(&step3) {
 1800:         step3 = re_italic.replace_all(&step3, "$1").to_string();
 1801:     }
 1802: 
 1803:     // 5. Rimuove parentesi o quadre rimaste vuote (senza cancellare asterischi letterali come 2*3)
 1804:     let no_empty_parens = re_empty_parens.replace_all(&step3, "");
 1805: 
 1806:     let lines: Vec<String> = no_empty_parens
 1807:         .lines()
 1808:         .map(|line| {
 1809:             let p = re_space_punct.replace_all(line, "$1");
 1810:             let s = re_multi_space.replace_all(&p, " ");
 1811:             let trimmed_start = if s.starts_with(' ') && !s.starts_with("  ") {
 1812:                 s.trim_start()
 1813:             } else {
 1814:                 &s
 1815:             };
 1816:             trimmed_start.trim_end().to_string()
 1817:         })
 1818:         .collect();
 1819: 
 1820:     let mut res = lines.join("\n");
 1821:     if text.ends_with('\n') && !res.ends_with('\n') {
 1822:         res.push('\n');
 1823:     }
 1824:     res
 1825: }

 2168: /// Sanitizzatore incrementale per lo streaming della prosa.
 2169: /// Dotato di sliding tail buffer: trattiene la porzione terminale ambigua (fino a 25 caratteri)
 2170: /// se coincide con un prefisso di citazione (`[`, `[S`, `(`, `(S`) o di marcatore markdown (`*`, `**`),
 2171: /// garantendo che nessun frammento grezzo o asterisco appaia temporaneamente all'utente.
 2172: #[derive(Debug, Default)]
 2173: pub struct StreamProseSanitizer {
 2174:     raw_buffer: String,
 2175:     clean_accumulated: String,
 2176: }
 2177: 
 2178: impl StreamProseSanitizer {
 2179:     pub fn new() -> Self {
 2180:         Self::default()
 2181:     }

 2183:     /// Determina la lunghezza del suffisso ambiguo da trattenere nel buffer scorrevole.
 2184:     /// Garantisce sempre che qualsiasi slicing avvenga su confini di caratteri UTF-8 validi (previene panic su lettere accentate ed emoji).
 2185:     pub fn ambiguous_suffix_len(s: &str) -> usize {
 2186:         let len = s.len();
 2187:         if len == 0 {
 2188:             return 0;
 2189:         }
 2190: 
 2191:         // Helper per arretrare al confine di carattere UTF-8 valido più vicino (mai panic)
 2192:         let safe_boundary = |mut idx: usize| -> usize {
 2193:             while idx > 0 && !s.is_char_boundary(idx) {
 2194:                 idx -= 1;
 2195:             }
 2196:             idx
 2197:         };
 2198: 
 2199:         // 1. Prefisso di citazione aperto: es. `[` senza `]` o `(` senza `)` (finestra fino a 30 byte/caratteri)
 2200:         let check_window = safe_boundary(len.saturating_sub(30));
 2201:         if let Some(pos) = s[check_window..].rfind(|c| c == '[' || c == '(') {
 2202:             let abs_pos = check_window + pos;
 2203:             let candidate_slice = &s[abs_pos..];
 2204:             let has_closing = candidate_slice.contains(']') || candidate_slice.contains(')');
 2205:             if !has_closing {
 2206:                 let inside = &candidate_slice[1..];
 2207:                 let is_citation_prefix = inside.chars().all(|c| {
 2208:                     c.is_ascii_whitespace() || c == 'S' || c == 's' || c.is_ascii_digit() || c == ',' || c == ';'
 2209:                 });
 2210:                 if is_citation_prefix {
 2211:                     return len - abs_pos;
 2212:                 }
 2213:             }
 2214:         }
 2215: 
 2216:         // 2. Grassetto markdown `**` aperto (conteggio dispari nella finestra di 60 byte/caratteri)
 2217:         let star_window = safe_boundary(len.saturating_sub(60));
 2218:         let star_slice = &s[star_window..];
 2219:         let dstar_count = star_slice.matches("**").count();
 2220:         if dstar_count % 2 != 0 {
 2221:             if let Some(pos) = star_slice.rfind("**") {
 2222:                 return len - (star_window + pos);
 2223:             }
 2224:         }
 2225: 
 2226:         // 3. Corsivo markdown `*` aperto (conteggio dispari escludendo `**`)
 2227:         let without_dstar = star_slice.replace("**", "");
 2228:         let single_star_count = without_dstar.chars().filter(|&c| c == '*').count();
 2229:         if single_star_count % 2 != 0 {
 2230:             if let Some(pos) = star_slice.rfind('*') {
 2231:                 return len - (star_window + pos);
 2232:             }
 2233:         }
 2234: 
 2235:         0
 2236:     }

 2239:     pub fn feed(&mut self, chunk: &str) -> String {
 2240:         self.raw_buffer.push_str(chunk);
 2241:         let suffix_len = Self::ambiguous_suffix_len(&self.raw_buffer);
 2242:         let safe_boundary = self.raw_buffer.len().saturating_sub(suffix_len);
 2243: 
 2244:         let mut char_safe_boundary = safe_boundary;
 2245:         while char_safe_boundary > 0 && !self.raw_buffer.is_char_boundary(char_safe_boundary) {
 2246:             char_safe_boundary -= 1;
 2247:         }
 2248: 
 2249:         let safe_raw = &self.raw_buffer[..char_safe_boundary];
 2250:         let sanitized = sanitize_answer_prose(safe_raw);
 2251: 
 2252:         if sanitized.len() > self.clean_accumulated.len() && sanitized.starts_with(&self.clean_accumulated) {
 2253:             let delta = sanitized[self.clean_accumulated.len()..].to_string();
 2254:             self.clean_accumulated = sanitized;
 2255:             delta
 2256:         } else if sanitized != self.clean_accumulated {
 2257:             let common_prefix_len = sanitized
 2258:                 .chars()

 2260:                 .take_while(|(a, b)| a == b)
 2261:                 .map(|(a, _)| a.len_utf8())
 2262:                 .sum::<usize>();
 2263:             if sanitized.len() > common_prefix_len && common_prefix_len == self.clean_accumulated.len() {
 2264:                 let delta = sanitized[self.clean_accumulated.len()..].to_string();
 2265:                 self.clean_accumulated = sanitized;
 2266:                 delta
 2267:             } else {
 2268:                 String::new()
 2269:             }
 2270:         } else {
 2271:             String::new()
 2272:         }
 2273:     }

 2415:                                     emitted.push(c);

 2417:                             }
 2418:                         } else {
 2419:                             match c {

 2426:                                     if let Some(_) = self.pending_high_surrogate.take() { emitted.push('\u{FFFD}'); }

 2442:                                 }
 2443:                                 'n' => {
 2444:                                     if let Some(_) = self.pending_high_surrogate.take() { emitted.push('\u{FFFD}'); }

 2446:                                     self.in_escape = false;
 2447:                                 }

 2450:                                     emitted.push('\r');
 2451:                                     self.in_escape = false;
 2452:                                 }
 2453:                                 't' => {
 2454:                                     if let Some(_) = self.pending_high_surrogate.take() { emitted.push('\u{FFFD}'); }
 2455:                                     emitted.push('\t');
 2456:                                     self.in_escape = false;
 2457:                                 }
 2458:                                 'u' => {
 2459:                                     self.in_unicode = true;

 2461:                                 }
 2462:                                 other => {
 2463:                                     if let Some(_) = self.pending_high_surrogate.take() { emitted.push('\u{FFFD}'); }
 2464:                                     emitted.push(other);
 2465:                                     self.in_escape = false;
 2466:                                 }
 2467:                             }
 2468:                         }
 2469:                     } else if c == '\\' {
 2470:                         self.in_escape = true;
 2471:                     } else if c == '"' {
 2472:                         if let Some(_) = self.pending_high_surrogate.take() { emitted.push('\u{FFFD}'); }
 2473:                         self.state = JsonParserState::FinishedAnswer;
 2474:                     } else {
 2475:                         if let Some(_) = self.pending_high_surrogate.take() { emitted.push('\u{FFFD}'); }
 2476:                         emitted.push(c);
 2477:                     }
 2478:                 }
 2479:                 JsonParserState::FinishedAnswer => {}
 2480:             }
 2481:         }
 2482:         emitted
 2483:     }

 2498: /// Decodificatore incrementale per stream di byte di rete:
 2499: /// trattiene i byte di sequenze UTF-8 multibyte incomplete alla fine di un frammento,
 2500: /// garantendo che nessun carattere multibyte (es. "è", "à", "€", emoji) venga convertito
 2501: /// nel carattere di rimpiazzo ("") sui confini dei pacchetti di rete.
 2502: #[derive(Debug, Default)]
 2503: pub struct Utf8ChunkDecoder {
 2504:     buffer: Vec<u8>,
 2505: }
 2506: 
 2507: impl Utf8ChunkDecoder {
 2508:     pub fn new() -> Self {
 2509:         Self { buffer: Vec::new() }
 2510:     }
 2511: 
 2512:     pub fn decode(&mut self, chunk: &[u8]) -> String {
 2513:         self.buffer.extend_from_slice(chunk);
 2514:         let valid_len = match std::str::from_utf8(&self.buffer) {
 2515:             Ok(s) => s.len(),
 2516:             Err(e) => {
 2517:                 if e.error_len().is_none() {
 2518:                     // Sequenza UTF-8 multibyte incompleta alla fine del buffer
 2519:                     e.valid_up_to()
 2520:                 } else {
 2521:                     let valid_up_to = e.valid_up_to();
 2522:                     if valid_up_to > 0 {
 2523:                         valid_up_to
 2524:                     } else {
 2525:                         // Byte corrotto o non valido: consuma un byte per evitare deadlock
 2526:                         self.buffer.remove(0);
 2527:                         return "\u{FFFD}".to_string();
 2528:                     }
 2529:                 }
 2530:             }
 2531:         };
 2532: 
 2533:         if valid_len > 0 {
 2534:             let valid_str = match std::str::from_utf8(&self.buffer[..valid_len]) {
 2535:                 Ok(s) => s.to_string(),
 2536:                 Err(_) => String::from_utf8_lossy(&self.buffer[..valid_len]).to_string(),
 2537:             };
 2538:             self.buffer.drain(..valid_len);
 2539:             valid_str
 2540:         } else {
 2541:             String::new()
 2542:         }
 2543:     }
 2544: 
 2545:     pub fn flush(&mut self) -> String {
 2546:         if self.buffer.is_empty() {
 2547:             String::new()
 2548:         } else {
 2549:             let s = String::from_utf8_lossy(&self.buffer).to_string();
 2550:             self.buffer.clear();
 2551:             s
 2552:         }
 2553:     }
 2554: }
