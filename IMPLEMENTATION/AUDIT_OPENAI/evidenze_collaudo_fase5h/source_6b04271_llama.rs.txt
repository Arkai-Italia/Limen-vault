  137: pub fn rotate_llama_server_logs(logs_dir: &Path, max_files: usize) {
  138:     if let Ok(entries) = fs::read_dir(logs_dir) {
  139:         let mut files: Vec<PathBuf> = entries
  140:             .filter_map(|e| e.ok())
  141:             .map(|e| e.path())
  142:             .filter(|p| {

  144:                     && p.file_name()
  145:                         .and_then(|n| n.to_str())
  146:                         .map(|n| n.starts_with("llama-server_") && n.ends_with(".log"))
  147:                         .unwrap_or(false)
  148:             })
  149:             .collect();
  150: 
  151:         files.sort();
  152: 
  153:         if files.len() > max_files {
  154:             let to_remove = files.len() - max_files;
  155:             for p in files.iter().take(to_remove) {
  156:                 let _ = fs::remove_file(p);
  157:             }
  158:         }
  159:     }

  198: pub fn compute_file_sha256(path: &Path) -> Result<String, String> {
  199:     MODEL_SHA256_COMPUTE_COUNT.fetch_add(1, Ordering::Relaxed);
  200:     let mut file = fs::File::open(path).map_err(|e| e.to_string())?;
  201:     let mut hasher = Sha256::new();
  202:     let mut buffer = [0u8; 65536];
  203:     loop {
  204:         let count = file.read(&mut buffer).map_err(|e| e.to_string())?;
  205:         if count == 0 {
  206:             break;
  207:         }
  208:         hasher.update(&buffer[..count]);
  209:     }
  210:     Ok(format!("{:x}", hasher.finalize()))
  211: }

  213: /// Parametric verification function for production and unit testing.
  214: pub fn model_status_for_paths(
  215:     target: &Path,
  216:     cache_file: &Path,
  217:     expected_size: u64,
  218:     expected_sha256: &str,
  219:     force_recheck: bool,
  220: ) -> LocalModelReport {
  221:     // If target does not exist, check common local huggingface snapshot location
  222:     if !target.exists() {
  223:         if let Ok(home) = std::env::var("HOME") {
  224:             let hf_candidate = Path::new(&home)
  225:                 .join(".cache/huggingface/hub/models--gpustack--bge-m3-GGUF/snapshots/2d48f1737679ad900d5c26c5aad5410e9c70fdca/bge-m3-Q8_0.gguf");
  226:             if hf_candidate.exists() {
  227:                 if let Ok(meta) = fs::metadata(&hf_candidate) {
  228:                     if meta.len() == expected_size {
  229:                         #[cfg(unix)]
  230:                         let _ = std::os::unix::fs::symlink(&hf_candidate, target);
  231:                         #[cfg(windows)]
  232:                         let _ = std::os::windows::fs::symlink_file(&hf_candidate, target);
  233:                     }
  234:                 }
  235:             }
  236:         }
  237:     }
  238: 
  239:     if !target.exists() {
  240:         return LocalModelReport {
  241:             installed: false,
  242:             path: target.to_string_lossy().to_string(),
  243:             bytes: 0,
  244:             sha256_ok: false,
  245:         };
  246:     }
  247: 
  248:     let meta = match fs::metadata(target) {
  249:         Ok(m) => m,
  250:         Err(_) => {
  251:             return LocalModelReport {
  252:                 installed: false,
  253:                 path: target.to_string_lossy().to_string(),
  254:                 bytes: 0,
  255:                 sha256_ok: false,
  256:             };
  257:         }
  258:     };
  259:     let bytes = meta.len();
  260:     let mtime_ms = meta
  261:         .modified()
  262:         .ok()
  263:         .and_then(|t| t.duration_since(std::time::UNIX_EPOCH).ok())
  264:         .map(|d| d.as_millis() as u64)
  265:         .unwrap_or(0);
  266:     let target_str = target.to_string_lossy().to_string();
  267: 
  268:     // Se la dimensione su disco non coincide con quella attesa per il modello, non è il modello atteso.
  269:     if bytes != expected_size {
  270:         eprintln!("[llama] File size mismatch on {}: {} vs expected {}", target.display(), bytes, expected_size);
  271:         let _ = fs::remove_file(cache_file);
  272:         return LocalModelReport {
  273:             installed: false,
  274:             path: target_str,
  275:             bytes,
  276:             sha256_ok: false,
  277:         };
  278:     }
  279: 
  280:     // 1. Fast path: verifica tramite cache dei metadati (se non è richiesto force_recheck)
  281:     if !force_recheck {
  282:         if let Some(cache) = read_model_cache(cache_file) {
  283:             let is_ntfs = crate::ai::is_high_precision_fs(target);
  284:             let mtime_matches = if is_ntfs {
  285:                 cache.mtime_ms == mtime_ms
  286:             } else {
  287:                 cache.mtime_ms.abs_diff(mtime_ms) <= 2000
  288:             };
  289: 
  290:             let path_matches = Path::new(&cache.path) == target || cache.path == target_str;
  291:             let sha256_matches = cache.sha256.eq_ignore_ascii_case(expected_sha256);
  292:             let size_matches = cache.bytes == expected_size && bytes == expected_size;
  293: 
  294:             if size_matches && path_matches && sha256_matches && mtime_matches && cache.sha256_ok {
  295:                 return LocalModelReport {
  296:                     installed: true,
  297:                     path: target_str,
  298:                     bytes,
  299:                     sha256_ok: true,
  300:                 };
  301:             }
  302:         }
  303:     }
  304: 
  305:     // 2. Slow path: verifica completa SHA-256 su disco
  306:     let sha256_ok = match compute_file_sha256(target) {
  307:         Ok(hash) => {
  308:             let ok = hash.eq_ignore_ascii_case(expected_sha256);
  309:             if ok {
  310:                 let _ = save_model_cache(cache_file, &ModelVerificationCache {
  311:                     path: target_str.clone(),
  312:                     bytes,
  313:                     mtime_ms,
  314:                     sha256: hash,
  315:                     sha256_ok: true,
  316:                     verified_at: chrono::Utc::now().to_rfc3339(),
  317:                 });
  318:             } else {
  319:                 eprintln!("[llama] Hash mismatch on {}: {} vs expected {}", target.display(), hash, expected_sha256);
  320:                 let _ = fs::remove_file(cache_file);
  321:             }
  322:             ok
  323:         }
  324:         Err(_) => {
  325:             let _ = fs::remove_file(cache_file);
  326:             false
  327:         }
  328:     };
  329: 
  330:     LocalModelReport {
  331:         installed: sha256_ok,
  332:         path: target_str,
  333:         bytes,
  334:         sha256_ok,
  335:     }
  336: }

  338: /// Install model from source file with SINGLE I/O pass (streaming copy + simultaneous hash computation).
  339: pub fn install_model_for_paths(
  340:     source_path: &Path,
  341:     target: &Path,
  342:     cache_file: &Path,
  343:     expected_size: u64,
  344:     expected_sha256: &str,
  345: ) -> Result<LocalModelReport, String> {
  346:     let t0 = Instant::now();
  347:     if !source_path.is_file() {
  348:         return Err(format!("File non trovato: {:?}", source_path));
  349:     }
  350:     let meta = fs::metadata(source_path).map_err(|e| e.to_string())?;
  351:     let bytes = meta.len();
  352:     if bytes != expected_size {
  353:         return Err(format!(
  354:             "Dimensione file non corretta: {} byte (richiesti esattamente {} byte)",
  355:             bytes,
  356:             expected_size
  357:         ));
  358:     }
  359: 
  360:     // Copia e calcolo hash in UN'UNICA passata di I/O (singola lettura del file sorgente)
  361:     let temp_target = target.with_extension("installing");
  362:     let _ = fs::remove_file(&temp_target);
  363: 
  364:     let mut source_file = fs::File::open(source_path).map_err(|e| e.to_string())?;
  365:     let mut dest_file = fs::File::create(&temp_target).map_err(|e| e.to_string())?;
  366:     let mut hasher = Sha256::new();
  367:     let mut buffer = vec![0u8; 1024 * 1024];
  368: 
  369:     loop {
  370:         let count = source_file.read(&mut buffer).map_err(|e| e.to_string())?;
  371:         if count == 0 {
  372:             break;
  373:         }
  374:         dest_file.write_all(&buffer[..count]).map_err(|e| e.to_string())?;
  375:         hasher.update(&buffer[..count]);
  376:     }
  377:     dest_file.flush().map_err(|e| e.to_string())?;
  378:     drop(dest_file);
  379:     drop(source_file);
  380: 
  381:     MODEL_SHA256_COMPUTE_COUNT.fetch_add(1, Ordering::Relaxed);
  382: 
  383:     let hash = format!("{:x}", hasher.finalize());
  384:     if hash.to_lowercase() != expected_sha256.to_lowercase() {
  385:         let _ = fs::remove_file(&temp_target);
  386:         return Err(format!(
  387:             "Verifica SHA-256 fallita: calcolato {} vs atteso {}",
  388:             hash, expected_sha256
  389:         ));
  390:     }
  391: 
  392:     if target.exists() {
  393:         let _ = fs::remove_file(target);
  394:     }
  395:     fs::rename(&temp_target, target).or_else(|_| {
  396:         fs::copy(&temp_target, target).map(|_| ())?;
  397:         let _ = fs::remove_file(&temp_target);
  398:         Ok::<(), std::io::Error>(())
  399:     }).map_err(|e| e.to_string())?;
  400: 
  401:     let dest_meta = fs::metadata(target).map_err(|e| e.to_string())?;
  402:     let dest_mtime_ms = dest_meta
  403:         .modified()
  404:         .ok()
  405:         .and_then(|t| t.duration_since(std::time::UNIX_EPOCH).ok())
  406:         .map(|d| d.as_millis() as u64)
  407:         .unwrap_or(0);
  408:     let target_str = target.to_string_lossy().to_string();
  409: 
  410:     save_model_cache(cache_file, &ModelVerificationCache {
  411:         path: target_str.clone(),
  412:         bytes,
  413:         mtime_ms: dest_mtime_ms,
  414:         sha256: hash,
  415:         sha256_ok: true,
  416:         verified_at: chrono::Utc::now().to_rfc3339(),
  417:     })?;
  418: 
  419:     let elapsed = t0.elapsed().as_millis() as u64;
  420:     log_local_model_timing("FILE_SELECTION", elapsed, &format!("path={}, bytes={}", target_str, bytes));
  421: 
  422:     Ok(LocalModelReport {
  423:         installed: true,
  424:         path: target_str,
  425:         bytes,
  426:         sha256_ok: true,
  427:     })
  428: }

  459: pub async fn download_model_with_progress<F>(on_progress: F) -> Result<LocalModelReport, String>
  460: where
  461:     F: Fn(u64, u64, f64) + Send + 'static,
  462: {
  463:     let target = get_target_model_path()?;
  464:     let temp_target = target.with_extension("downloading");
  465:     let _ = fs::remove_file(&temp_target);
  466: 
  467:     let client = reqwest::Client::builder()
  468:         .connect_timeout(Duration::from_secs(15))
  469:         .timeout(Duration::from_secs(1800))
  470:         .build()
  471:         .map_err(|e| e.to_string())?;
  472: 
  473:     let mut resp = client
  474:         .get(MODEL_DOWNLOAD_URL)
  475:         .send()
  476:         .await
  477:         .map_err(|e| format!("Richiesta download fallita: {}", e))?;
  478: 
  479:     if !resp.status().is_success() {
  480:         return Err(format!("Download fallito con stato HTTP: {}", resp.status()));
  481:     }
  482: 
  483:     let total_bytes = resp
  484:         .content_length()
  485:         .unwrap_or(EXPECTED_MODEL_SIZE_BYTES);
  486: 
  487:     let mut out = fs::File::create(&temp_target).map_err(|e| e.to_string())?;
  488: 
  489:     let mut downloaded: u64 = 0;
  490:     let mut hasher = Sha256::new();
  491:     let mut last_emit = Instant::now();
  492: 
  493:     while let Some(chunk) = resp.chunk().await.map_err(|e| format!("Errore lettura chunk: {}", e))? {
  494:         out.write_all(&chunk).map_err(|e| e.to_string())?;
  495:         hasher.update(&chunk);
  496:         downloaded += chunk.len() as u64;
  497: 
  498:         if last_emit.elapsed() >= Duration::from_millis(200) || (total_bytes > 0 && downloaded == total_bytes) {
  499:             let pct = if total_bytes > 0 {
  500:                 (downloaded as f64 / total_bytes as f64) * 100.0
  501:             } else {
  502:                 0.0
  503:             };
  504:             on_progress(downloaded, total_bytes, pct);
  505:             last_emit = Instant::now();
  506:         }
  507:     }
  508: 
  509:     out.flush().map_err(|e| e.to_string())?;
  510:     drop(out);
  511: 
  512:     let final_hash = format!("{:x}", hasher.finalize());
  513:     if final_hash.to_lowercase() != EXPECTED_MODEL_SHA256.to_lowercase() {
  514:         let _ = fs::remove_file(&temp_target);
  515:         return Err(format!(
  516:             "Verifica SHA-256 fallita sul file scaricato: calcolato {} vs atteso {}",
  517:             final_hash, EXPECTED_MODEL_SHA256
  518:         ));
  519:     }
  520: 
  521:     fs::rename(&temp_target, &target).map_err(|e| e.to_string())?;
  522: 
  523:     let dest_meta = fs::metadata(&target).map_err(|e| e.to_string())?;
  524:     let dest_mtime_ms = dest_meta
  525:         .modified()
  526:         .ok()

  663: /// Esegue un comando esterno con limite di tempo garantito, terminando il processo in caso di scadenza.
  664: pub fn run_command_with_timeout(
  665:     mut cmd: std::process::Command,
  666:     timeout: Duration,
  667: ) -> Option<std::process::Output> {
  668:     let mut child = cmd
  669:         .stdout(std::process::Stdio::piped())
  670:         .stderr(std::process::Stdio::piped())
  671:         .spawn()
  672:         .ok()?;
  673: 
  674:     let t_start = Instant::now();
  675:     let poll_interval = Duration::from_millis(30);
  676:     while t_start.elapsed() < timeout {
  677:         match child.try_wait() {
  678:             Ok(Some(_status)) => {
  679:                 return child.wait_with_output().ok();
  680:             }
  681:             Ok(None) => {
  682:                 std::thread::sleep(poll_interval);
  683:             }
  684:             Err(_) => {
  685:                 let _ = child.kill();
  686:                 return None;
  687:             }
  688:         }
  689:     }
  690: 
  691:     // Timeout scaduto: termina forzatamente il processo
  692:     let _ = child.kill();
  693:     #[cfg(windows)]
  694:     {
  695:         let pid = child.id();
  696:         unsafe {
  697:             let handle = windows_sys::Win32::System::Threading::OpenProcess(
  698:                 windows_sys::Win32::System::Threading::PROCESS_TERMINATE,
  699:                 0,
  700:                 pid,
  701:             );
  702:             if !handle.is_null() {
  703:                 windows_sys::Win32::System::Threading::TerminateProcess(handle, 1);
  704:                 windows_sys::Win32::Foundation::CloseHandle(handle);
  705:             }
  706:         }
  707:     }
  708:     let _ = child.try_wait();
  709:     None
  710: }

  961:         if !out.status.success() {
  962:             return None;
  963:         }
  964:         let text = String::from_utf8_lossy(&out.stdout);

 1535:             None => {

 1540:                 s.auto_start_failure_reason = Some(err.clone());
 1541:                 return Err(err);
 1542:             }
 1543:         };

 1545:         // Acquisizione flag starting con guardia RAII (garantisce azzeramento anche su panic/errore)
 1546:         {
 1547:             let mut s = self.0.lock().unwrap_or_else(|e| e.into_inner());
 1548:             s.starting = true;
 1549:             s.last_error = None;
 1550:         }
 1551:         let mut starting_guard = StartingGuard(self.clone(), true);

 1554:         let free_port = match find_free_port() {
 1555:             Ok(p) => p,
 1556:             Err(e) => {
 1557:                 let mut s = self.0.lock().unwrap_or_else(|e| e.into_inner());
 1558:                 s.starting = false;
 1559:                 s.auto_start_failed = true;
 1560:                 s.auto_start_failure_reason = Some(e.clone());
 1561:                 return Err(e);
 1562:             }
 1563:         };

 1568:             "-m", &model_path.to_string_lossy(),
 1569:             "--embedding",

 1571:             "--port", &free_port.to_string(),

 1582:             use std::os::unix::process::CommandExt;

 1587:             use std::os::windows::process::CommandExt;

 1594:         cmd.stderr(Stdio::piped());

 1597:             Ok(c) => c,
 1598:             Err(e) => {
 1599:                 let mut s = self.0.lock().unwrap_or_else(|e| e.into_inner());
 1600:                 s.starting = false;
 1601:                 s.auto_start_failed = true;

 1604:                 return Err(err);

 1606:         };

 1609:         let stderr_opt = child.stderr.take();

 1616:

 1619:             let legacy_log_path = parent.join("llama-server.log");

 1621:             if let Some(mut stderr) = stderr_opt {
 1622:                 std::thread::spawn(move || {

 1625:                     let mut file_legacy = fs::File::create(&legacy_log_path).ok();

 1628:                         if n == 0 {
 1629:                             break;
 1630:                         }

 1632:                             let _ = f.write_all(&buf[..n]);
 1633:                         }
 1634:                         if let Some(ref mut f) = file_legacy {
 1635:                             let _ = f.write_all(&buf[..n]);
 1636:                         }
 1637:                     }

 1639:             }

 1641:

 1646:             s.owned_pid = Some(child_pid);

 1679:                             last.drain(..keep);

 1681:                         })
 1682:                         .unwrap_or_default();

 1684:                     let err = format!("llama-server uscito con codice {}{}", code, detail);
 1685:                     guard.auto_start_failure_reason = Some(err.clone());
 1686:                     guard.last_error = Some(err.clone());

 1691:                         Some(&binary),
 1692:                         Some(child_pid),
 1693:                         Some(port),
 1694:                         &err,
 1695:                     );
 1696:                     return Err(err);
 1697:                 }
 1698:             }
 1699:         }
 1700: 
 1701:         if !healthy {
 1702:             self.stop()?;
 1703:             let mut guard = self.0.lock().map_err(|e| e.to_string())?;
 1704:             guard.starting = false;
 1705:             guard.auto_start_failed = true;

 1707:             guard.auto_start_failure_reason = Some(err.clone());
 1708:             guard.last_error = Some(err.clone());
 1709:             let elapsed = t_start.elapsed().as_millis() as u64;
 1710:             log_local_model_timing_meta(
 1711:                 "SERVICE_START_FAILED",
 1712:                 elapsed,
 1713:                 Some(&binary),
 1714:                 Some(child_pid),
 1715:                 Some(port),
 1716:                 &err,
 1717:             );
 1718:             return Err(err);
 1719:         }
 1720: 
 1721:         // 7. Verify dimensions are exactly 1024 (guasto vero, Regola 3)
 1722:         if let Err(e) = verify_dimensions(port) {

 1728:             guard.auto_start_failure_reason = Some(err.clone());
 1729:             guard.last_error = Some(err.clone());
 1730:             let elapsed = t_start.elapsed().as_millis() as u64;
 1731:             log_local_model_timing_meta(
 1732:                 "SERVICE_START_FAILED",
 1733:                 elapsed,
 1734:                 Some(&binary),
 1735:                 Some(child_pid),
 1736:                 Some(port),
 1737:                 &err,
 1738:             );
 1739:             return Err(err);
 1740:         }
 1741: 
 1742:         starting_guard.1 = false; // Disarma StartingGuard: avvio completato con successo
 1743:         {
 1744:             let mut guard = self.0.lock().map_err(|e| e.to_string())?;
 1745:             guard.starting = false;
 1746:             guard.auto_start_failed = false;
 1747:             guard.auto_start_failure_reason = None;
 1748:         }

 1751:         log_local_model_timing_meta(
 1752:             "SERVICE_START",
 1753:             elapsed,
 1754:             Some(&binary),
 1755:             Some(child_pid),
 1756:             Some(port),
 1757:             &format!("port={}, pid={:?}", port, child_pid),
 1758:         );
 1759: 
 1760:         Ok(self.status())
 1761:     }

 1763:     pub fn ensure_running(&self) -> Result<u16, String> {
 1764:         let port = {
 1765:             let s = self.0.lock().map_err(|e| e.to_string())?;
 1766:             s.port
 1767:         };
 1768:         if port > 0 && check_health(port) {
 1769:             return Ok(port);
 1770:         }
 1771: 
 1772:         // Try restart up to 3 times (Gate A10 compliance)
 1773:         for attempt in 1..=3 {
 1774:             match self.start() {
 1775:                 Ok(rep) if rep.healthy => {
 1776:                     let mut guard = self.0.lock().unwrap_or_else(|e| e.into_inner());
 1777:                     guard.crash_count = 0;
 1778:                     return Ok(rep.port);
 1779:                 }
 1780:                 Ok(_) => {}
 1781:                 Err(e) => {
 1782:                     eprintln!("[llama] Tentativo avvio {}/3 fallito: {}", attempt, e);
 1783:                     if attempt == 3 {
 1784:                         return Err(format!("Riavvio automatico fallito dopo 3 tentativi: {}", e));
 1785:                     }
 1786:                     std::thread::sleep(Duration::from_millis(500));
 1787:                 }
 1788:             }
 1789:         }
 1790:         Err("Impossibile avviare il servizio llama-server dopo 3 tentativi".into())
 1791:     }

 1793:     pub fn stop(&self) -> Result<LocalServerReport, String> {
 1794:         let (child, owned_pid) = {
 1795:             let mut s = self.0.lock().map_err(|e| e.to_string())?;
 1796:             s.port = 0;
 1797:             s.adopted_pid = None;
 1798:             s.last_error = None;
 1799:             s.starting = false;
 1800:             let owned = s.child.as_ref().map(|c| c.id()).or(s.owned_pid);
 1801:             s.owned_pid = None;
 1802:             (s.child.take(), owned)
 1803:         }; // Lock rilasciato IMMEDIATAMENTE: nessuna attesa su processi o I/O sotto lock!
 1804: 
 1805:         if let Some(mut c) = child {
 1806:             #[cfg(unix)]
 1807:             unsafe {
 1808:                 let pid = c.id() as i32;
 1809:                 libc::kill(-pid, libc::SIGTERM);
 1810:                 libc::kill(pid, libc::SIGTERM);
 1811:             }
 1812:             #[cfg(windows)]
 1813:             {
 1814:                 let _ = c.kill(); // Termina esplicitamente SOLO il figlio posseduto dall'app
 1815:             }
 1816:             for _ in 0..20 {
 1817:                 if c.try_wait().ok().flatten().is_some() {
 1818:                     break;
 1819:                 }
 1820:                 std::thread::sleep(Duration::from_millis(50));
 1821:             }
 1822:             #[cfg(unix)]
 1823:             unsafe {
 1824:                 let pid = c.id() as i32;
 1825:                 libc::kill(-pid, libc::SIGKILL);
 1826:                 libc::kill(pid, libc::SIGKILL);
 1827:             }
 1828:             #[cfg(windows)]
 1829:             {
 1830:                 let _ = c.kill();
 1831:             }
 1832:             for _ in 0..10 {
 1833:                 if c.try_wait().ok().flatten().is_some() {
 1834:                     break;
 1835:                 }
 1836:                 std::thread::sleep(Duration::from_millis(50));
 1837:             }
 1838:         }
 1839: 
 1840:         if let Some(pid) = owned_pid {
 1841:             remove_service_registration_if_owned(pid);
 1842:         }
 1843: 
 1844:         Ok(self.status())
 1845:     }
 1846: }
