  227: fn obsidian_path() -> Option<PathBuf> {
  228:     let mut paths = vec![PathBuf::from("/Applications/Obsidian.app")];
  229:     if let Ok(home) = std::env::var("HOME") {
  230:         paths.push(Path::new(&home).join("Applications/Obsidian.app"));
  231:     }
  232:     if let Some(p) = paths.into_iter().find(|p| p.is_dir()) {
  233:         return Some(p);
  234:     }
  235:     let output = Command::new("/usr/bin/mdfind")
  236:         .arg("kMDItemCFBundleIdentifier == 'md.obsidian'")
  237:         .output()
  238:         .ok()?;
  239:     String::from_utf8_lossy(&output.stdout)
  240:         .lines()
  241:         .map(PathBuf::from)
  242:         .find(|p| p.extension().is_some_and(|e| e == "app") && p.is_dir())
  243: }

  250: #[tauri::command]
  251: fn open_obsidian(vault_path: String) -> Result<(), String> {
  252:     let app = obsidian_path().ok_or("OBSIDIAN_NOT_INSTALLED")?;
  253:     let path = Path::new(&vault_path)
  254:         .canonicalize()
  255:         .map_err(|e| e.to_string())?;
  256:     if !path.is_dir() {
  257:         return Err("Vault path is not a directory".into());
  258:     }
  259:     let registered = std::env::var("HOME")
  260:         .ok()
  261:         .and_then(|home| {
  262:             std::fs::read_to_string(
  263:                 Path::new(&home).join("Library/Application Support/obsidian/obsidian.json"),
  264:             )
  265:             .ok()
  266:         })
  267:         .and_then(|raw| serde_json::from_str::<serde_json::Value>(&raw).ok())
  268:         .is_some_and(|registry| obsidian_registered(&registry, &path));
  269:     let uri = if registered {
  270:         format!(
  271:             "obsidian://open?path={}",
  272:             urlencoding::encode(&path.to_string_lossy())
  273:         )
  274:     } else {
  275:         "obsidian://choose-vault".to_string()
  276:     };
  277:     let output = Command::new("/usr/bin/open")
  278:         .arg("-a")
  279:         .arg(app)
  280:         .arg(uri)
  281:         .output()
  282:         .map_err(|e| e.to_string())?;
  283:     if output.status.success() && !registered {
  284:         Err(format!("OBSIDIAN_REGISTRATION_REQUIRED: In Obsidian, use Open folder as vault → Open and select {}. This is required once; then press Open in Obsidian again.", path.display()))
  285:     } else if output.status.success() {
  286:         Ok(())
  287:     } else {
  288:         Err(String::from_utf8_lossy(&output.stderr).into())
  289:     }
  290: }

  292: fn obsidian_registered(registry: &serde_json::Value, target: &Path) -> bool {
  293:     registry
  294:         .get("vaults")
  295:         .and_then(serde_json::Value::as_object)
  296:         .is_some_and(|vaults| {
  297:             vaults
  298:                 .values()
  299:                 .filter_map(|v| v.get("path").and_then(serde_json::Value::as_str))
  300:                 .any(|p| Path::new(p).canonicalize().is_ok_and(|p| p == target))
  301:         })
  302: }

  975: #[tauri::command]
  976: async fn embeddings_sync_vault(
  977:     app: tauri::AppHandle,
  978:     vault_path: String,
  979:     api_key: Option<String>,
  980:     model: Option<String>,
  981:     llama_state: tauri::State<'_, limen_vault::llama::LlamaServerState>,
  982: ) -> Result<limen_vault::embeddings::EmbeddingsStatusReport, String> {
  983:     let rep = limen_vault::embeddings::get_embeddings_provider(Path::new(&vault_path), 0);
  984:     let port = if rep.provider == "local" {
  985:         Some(llama_state.ensure_running()?)
  986:     } else {
  987:         None
  988:     };
  989: 
  990:     // Avanzamento reale per la barra "Ricalcolo cache in corso": un evento all'avvio e uno per lotto.
  991:     let app_handle = app.clone();
  992:     let on_progress = move |processed: usize, total: usize| {
  993:         use tauri::Emitter;
  994:         let percent = if total == 0 { 100.0 } else { (processed as f64 / total as f64) * 100.0 };
  995:         let _ = app_handle.emit(
  996:             "embeddings_sync_progress",
  997:             serde_json::json!({ "processed": processed, "total": total, "percent": percent }),
  998:         );
  999:     };
 1000: 
 1001:     limen_vault::embeddings::sync_embeddings_with_progress(
 1002:         Path::new(&vault_path),
 1003:         api_key.as_deref().unwrap_or(""),
 1004:         model.as_deref(),
 1005:         port,
 1006:         Some(&on_progress),
 1007:     )
 1008:     .await
 1009: }
