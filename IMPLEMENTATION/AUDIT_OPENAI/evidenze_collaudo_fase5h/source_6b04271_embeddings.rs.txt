  102: /// Read embeddings provider from 00_SYSTEM/SYNC_PROFILE.json.
  103: /// If absent, defaults to "openai".
  104: pub fn get_embeddings_provider(vault_path: &Path, active_port: u16) -> EmbeddingsProviderReport {
  105:     let profile_path = vault_path.join("00_SYSTEM").join("SYNC_PROFILE.json");
  106:     let mut configured_provider = None;
  107: 
  108:     if let Ok(content) = std::fs::read_to_string(&profile_path) {
  109:         if let Ok(val) = serde_json::from_str::<Value>(&content) {
  110:             if let Some(p) = val.get("embeddingsProvider").and_then(Value::as_str) {
  111:                 let trimmed = p.trim().to_lowercase();
  112:                 if trimmed == "local" || trimmed == "openai" {
  113:                     configured_provider = Some(trimmed);
  114:                 }
  115:             }
  116:         }
  117:     }
  118: 
  119:     let provider = configured_provider.unwrap_or_else(|| "openai".to_string());
  120:     let is_local = provider == "local";
  121:     let model = if is_local {
  122:         "bge-m3".to_string()
  123:     } else {
  124:         DEFAULT_EMBEDDINGS_MODEL.to_string()
  125:     };
  126:     let dimensions = if is_local { 1024 } else { 1536 };
  127:     let endpoint = if is_local {
  128:         if active_port > 0 {
  129:             format!("http://127.0.0.1:{}/v1/embeddings", active_port)
  130:         } else {
  131:             String::new()
  132:         }
  133:     } else {
  134:         DEFAULT_EMBEDDINGS_ENDPOINT.to_string()
  135:     };
  136: 
  137:     let cache = load_embeddings_cache(vault_path).unwrap_or_default();
  138:     let cache_entries = cache.entries.len();
  139:     // Cache assente o vuota: nessuna dimensione memorizzata da confrontare. Senza questo azzeramento il
  140:     // default della struct (1536) veniva mostrato in UI come "cache a 1536 dim" con 0 passaggi
  141:     // (collaudo del 20/09/2026 su vault senza EMBEDDINGS_CACHE.json).
  142:     let cache_dimensions = if cache_entries == 0 { 0 } else { cache.dimensions };
  143: 
  144:     let mut total_passages = 0;
  145:     let mut matched_passages = 0;
  146:     if let Ok(catalog) = load_catalog(vault_path) {
  147:         for doc in catalog.documents.values() {
  148:             for p in &doc.passages {
  149:                 total_passages += 1;
  150:                 if let Some(entry) = cache.entries.get(&p.passage_id) {
  151:                     if entry.sha256 == p.sha256 && (cache_dimensions == 0 || cache_dimensions == dimensions) {
  152:                         matched_passages += 1;
  153:                     }
  154:                 }
  155:             }
  156:         }
  157:     }
  158: 
  159:     let needs_reindex = if total_passages > 0 {
  160:         matched_passages < total_passages || (cache_entries > 0 && cache_dimensions != dimensions)
  161:     } else {
  162:         cache_entries > 0 && cache_dimensions != dimensions
  163:     };
  164: 
  165:     EmbeddingsProviderReport {
  166:         provider,
  167:         model,
  168:         dimensions,
  169:         endpoint,
  170:         cache_dimensions,
  171:         cache_entries,
  172:         total_passages,
  173:         matched_passages,
  174:         needs_reindex,
  175:     }
  176: }

  178: /// Atomically write embeddingsProvider to 00_SYSTEM/SYNC_PROFILE.json.
  179: /// Endpoint and model are NOT persisted: endpoint is built dynamically at runtime from active_port.
  180: pub fn set_embeddings_provider(
  181:     vault_path: &Path,
  182:     provider: &str,
  183:     active_port: u16,
  184: ) -> Result<EmbeddingsProviderReport, String> {
  185:     let prov = provider.trim().to_lowercase();
  186:     if prov != "local" && prov != "openai" {
  187:         return Err(format!(
  188:             "Fornitore non valido '{}': ammessi esclusivamente 'openai' o 'local'",
  189:             provider
  190:         ));
  191:     }
  192: 
  193:     let profile_path = vault_path.join("00_SYSTEM").join("SYNC_PROFILE.json");
  194:     let mut val = if profile_path.exists() {
  195:         let content = std::fs::read_to_string(&profile_path).map_err(|e| e.to_string())?;
  196:         serde_json::from_str::<Value>(&content).unwrap_or_else(|_| json!({}))
  197:     } else {
  198:         json!({})
  199:     };
  200: 
  201:     if !val.is_object() {
  202:         val = json!({});
  203:     }
  204: 
  205:     let obj = val.as_object_mut().unwrap();
  206:     obj.insert("embeddingsProvider".into(), Value::String(prov.clone()));
  207: 
  208:     // Never persist endpoint or model: endpoint is runtime-constructed from live server port
  209:     obj.remove("embeddingsEndpoint");
  210:     obj.remove("embeddingsModel");
  211: 
  212:     let sys_dir = vault_path.join("00_SYSTEM");
  213:     if !sys_dir.exists() {
  214:         std::fs::create_dir_all(&sys_dir).map_err(|e| e.to_string())?;
  215:     }
  216: 
  217:     let serialized = serde_json::to_string_pretty(&val).map_err(|e| e.to_string())?;
  218:     std::fs::write(&profile_path, serialized.as_bytes()).map_err(|e| e.to_string())?;
  219:     let _ = crate::ai::save_embeddings_provider_setting(&prov);
  220: 
  221:     Ok(get_embeddings_provider(vault_path, active_port))
  222: }

  224: /// Security validation: if embeddingsProvider is "local", ensure the endpoint is strictly a loopback address.
  225: pub fn validate_provider_endpoint(vault_path: &Path, endpoint: &str) -> Result<(), String> {
  226:     if endpoint.trim().is_empty() {
  227:         return Ok(());
  228:     }
  229:     let profile_path = vault_path.join("00_SYSTEM").join("SYNC_PROFILE.json");
  230:     if let Ok(content) = std::fs::read_to_string(&profile_path) {
  231:         if let Ok(val) = serde_json::from_str::<Value>(&content) {
  232:             if let Some(p) = val.get("embeddingsProvider").and_then(Value::as_str) {
  233:                 if p.trim().to_lowercase() == "local" {

  240:                     }
  241:                 }
  242:             }
  243:         }
  244:     }
  245:     Ok(())
  246: }

  260: /// Validates an embeddings endpoint URL for security.
  261: /// Returns Ok(true) if the endpoint is loopback (allowing http://),
  262: /// Ok(false) if the endpoint is a valid external HTTPS endpoint,
  263: /// or Err(message) if an external endpoint attempts to use unencrypted HTTP or if the URL is invalid.
  264: pub fn is_loopback_endpoint(endpoint: &str) -> Result<bool, String> {
  265:     let url = reqwest::Url::parse(endpoint)
  266:         .map_err(|e| format!("URL endpoint non valido '{}': {}", endpoint, e))?;
  267: 
  268:     let host = url.host_str().unwrap_or("").trim();
  269:     let is_loopback = host == "127.0.0.1" || host == "localhost" || host == "::1" || host == "[::1]";
  270: 
  271:     match url.scheme() {
  272:         "http" => {
  273:             if is_loopback {
  274:                 Ok(true)
  275:             } else {
  276:                 Err(format!(
  277:                     "Rifiutato endpoint HTTP non sicuro '{}': http:// è consentito esclusivamente su loopback (127.0.0.1, localhost, ::1)",
  278:                     endpoint
  279:                 ))
  280:             }
  281:         }
  282:         "https" => Ok(is_loopback),
  283:         other => Err(format!("Schema URL non supportato '{}': supportati solo http e https", other)),
  284:     }
  285: }

  346: /// Compute cosine similarity between two float vectors.
  347: pub fn cosine_similarity(a: &[f32], b: &[f32]) -> f32 {
  348:     if a.len() != b.len() || a.is_empty() {
  349:         return 0.0;
  350:     }
  351:     let mut dot = 0.0f32;
  352:     let mut norm_a = 0.0f32;
  353:     let mut norm_b = 0.0f32;
  354:     for (x, y) in a.iter().zip(b.iter()) {
  355:         dot += x * y;
  356:         norm_a += x * x;
  357:         norm_b += y * y;
  358:     }
  359:     let denom = norm_a.sqrt() * norm_b.sqrt();
  360:     if denom <= 0.0000001 {
  361:         0.0
  362:     } else {
  363:         dot / denom
  364:     }
  365: }

  367: /// Load embeddings cache from 00_SYSTEM/EMBEDDINGS_CACHE.json or default if absent.
  368: pub fn load_embeddings_cache(vault_path: &Path) -> Result<Arc<EmbeddingsCache>, String> {
  369:     let r = root(vault_path)?;
  370:     let sys = match child(&r, "00_SYSTEM") {
  371:         Ok(d) => d,
  372:         Err(_) => return Ok(Arc::new(EmbeddingsCache::default())),
  373:     };
  374:     if !names(&sys)?.iter().any(|s| s == EMBEDDINGS_CACHE_FILE) {
  375:         return Ok(Arc::new(EmbeddingsCache::default()));
  376:     }
  377: 
  378:     let emb_file = vault_path.join("00_SYSTEM").join(EMBEDDINGS_CACHE_FILE);
  379:     let current_rev = get_embeddings_cache_revision(vault_path);
  380: 
  381:     if let Ok(meta) = std::fs::metadata(&emb_file) {
  382:         let size = meta.len();
  383:         let mtime = meta.modified().unwrap_or(SystemTime::UNIX_EPOCH);
  384:         if let Ok(guard) = EMBEDDINGS_IN_MEMORY_CACHE.lock() {
  385:             if let Some(entry) = guard.get(vault_path) {
  386:                 let is_high_prec = crate::ai::is_high_precision_fs(vault_path);
  387:                 let is_mtime_match = if is_high_prec {
  388:                     entry.mtime == mtime
  389:                 } else {
  390:                     entry.mtime == mtime
  391:                         || entry.mtime.duration_since(mtime).map(|d| d.as_millis() < 2000).unwrap_or(false)
  392:                         || mtime.duration_since(entry.mtime).map(|d| d.as_millis() < 2000).unwrap_or(false)
  393:                 };
  394:                 if entry.size == size && is_mtime_match && entry.revision == current_rev {
  395:                     return Ok(entry.cache.clone());
  396:                 }
  397:             }
  398:         }
  399:     }
  400: 
  401:     increment_embeddings_disk_read_count(vault_path);
  402:     let bytes = read(&sys, EMBEDDINGS_CACHE_FILE)?;
  403:     if bytes.len() > 512 * 1024 * 1024 {
  404:         return Err("Embeddings cache file exceeds 512 MB".into());
  405:     }
  406:     let cache: EmbeddingsCache = serde_json::from_slice(&bytes).map_err(err)?;
  407:     let cache_arc = Arc::new(cache);
  408: 
  409:     if let Ok(meta) = std::fs::metadata(&emb_file) {
  410:         let size = meta.len();
  411:         let mtime = meta.modified().unwrap_or(SystemTime::UNIX_EPOCH);
  412:         let current_rev = get_embeddings_cache_revision(vault_path);
  413:         if let Ok(mut guard) = EMBEDDINGS_IN_MEMORY_CACHE.lock() {
  414:             guard.insert(vault_path.to_path_buf(), EmbeddingsCacheEntry {
  415:                 vault_path: vault_path.to_path_buf(),
  416:                 size,
  417:                 mtime,
  418:                 revision: current_rev,
  419:                 cache: cache_arc.clone(),
  420:             });
  421:         }
  422:     }
  423: 
  424:     Ok(cache_arc)
  425: }

  427: /// Atomically save embeddings cache to 00_SYSTEM/EMBEDDINGS_CACHE.json
  428: pub fn save_embeddings_cache(vault_path: &Path, cache: &mut EmbeddingsCache) -> Result<(), String> {
  429:     cache.last_updated_at = now_iso();
  430:     let r = root(vault_path)?;
  431:     let sys = child(&r, "00_SYSTEM")?;
  432:     let tmp = format!(".embeddings-{}.tmp", crate::ai::random_token()?);
  433:     let bytes = serde_json::to_vec(cache).map_err(err)?;
  434:     write_new(&sys, &tmp, &bytes)?;
  435:     let res = sys.rename(&tmp, &sys, EMBEDDINGS_CACHE_FILE).map_err(err);
  436:     if res.is_err() {
  437:         let _ = sys.remove_file(&tmp);
  438:     } else {
  439:         bump_embeddings_cache_revision(vault_path);
  440:     }
  441:     res
  442: }

  444: /// Get current embeddings status and coverage
  445: pub fn embeddings_status(vault_path: &Path) -> Result<EmbeddingsStatusReport, String> {
  446:     let catalog = load_catalog(vault_path)?;
  447:     let cache = load_embeddings_cache(vault_path)?;
  448: 
  449:     let mut total_passages: usize = 0;
  450:     let mut cached_passages: usize = 0;
  451: 
  452:     for doc in catalog.documents.values() {
  453:         for p in &doc.passages {
  454:             total_passages += 1;
  455:             if let Some(entry) = cache.entries.get(&p.passage_id) {
  456:                 if entry.sha256 == p.sha256 {
  457:                     cached_passages += 1;
  458:                 }
  459:             }
  460:         }
  461:     }
  462: 
  463:     let missing = total_passages.saturating_sub(cached_passages);
  464:     let coverage = if total_passages == 0 {
  465:         1.0
  466:     } else {
  467:         (cached_passages as f64) / (total_passages as f64)
  468:     };
  469: 
  470:     Ok(EmbeddingsStatusReport {
  471:         total_passages,
  472:         cached_passages,
  473:         missing_passages: missing,
  474:         coverage,
  475:         model: cache.model.clone(),
  476:         dimensions: cache.dimensions,
  477:         is_available: cached_passages > 0,
  478:         last_updated_at: cache.last_updated_at.clone(),
  479:     })
  480: }

  506: pub async fn fetch_openai_embeddings_with_endpoint_and_timeout(
  507:     endpoint: &str,
  508:     api_key: &str,
  509:     model: &str,
  510:     texts: &[String],
  511:     timeout_override: Option<Duration>,
  512: ) -> Result<Vec<Vec<f32>>, String> {
  513:     if texts.is_empty() {
  514:         return Ok(Vec::new());
  515:     }
  516: 
  517:     let is_loopback = is_loopback_endpoint(endpoint)?;
  518: 
  519:     // Non-loopback endpoints strictly require an API key
  520:     if !is_loopback && api_key.trim().is_empty() {
  521:         return Err("OpenAI API key non configurata".into());
  522:     }
  523: 
  524:     let request_timeout = timeout_override.unwrap_or_else(|| {
  525:         if is_loopback {
  526:             BATCH_EMBEDDINGS_TIMEOUT
  527:         } else {
  528:             Duration::from_secs(60)
  529:         }
  530:     });
  531:     let mut client_builder = reqwest::Client::builder()
  532:         .connect_timeout(Duration::from_secs(10))
  533:         .timeout(request_timeout);
  534: 
  535:     if !is_loopback {
  536:         client_builder = client_builder.https_only(true);
  537:     }
  538: 
  539:     let client = client_builder
  540:         .build()
  541:         .map_err(|e| format!("HTTP client error: {}", e))?;
  542: 
  543:     let body = json!({
  544:         "model": model,
  545:         "input": texts
  546:     });
  547: 
  548:     let mut req = client.post(endpoint).json(&body);
  549:     if !api_key.trim().is_empty() {
  550:         req = req.bearer_auth(api_key);
  551:     }
  552: 
  553:     let resp = req
  554:         .send()
  555:         .await
  556:         .map_err(|e| {
  557:             if e.is_timeout() {
  558:                 format!("Richiesta embeddings verso '{}' scaduta per timeout: {}", endpoint, e)
  559:             } else {
  560:                 format!("Richiesta embeddings fallita verso '{}': {}", endpoint, e)
  561:             }
  562:         })?;
  563: 
  564:     if !resp.status().is_success() {
  565:         let status = resp.status().as_u16();
  566:         let err_text = resp.text().await.unwrap_or_default();
  567:         return Err(match status {
  568:             401 => "Chiave API non valida o non autorizzata".into(),
  569:             429 => "Quota esaurita o rate limit raggiunto".into(),
  570:             code => format!("Embeddings error HTTP {}: {}", code, err_text),
  571:         });
  572:     }
  573: 
  574:     let val: Value = resp
  575:         .json()
  576:         .await
  577:         .map_err(|e| format!("Risposta JSON embeddings non valida: {}", e))?;
  578: 
  579:     let data = val
  580:         .get("data")
  581:         .and_then(Value::as_array)
  582:         .ok_or("Campo data mancante nella risposta embeddings")?;
  583: 
  584:     let mut embeddings = Vec::with_capacity(texts.len());
  585:     for item in data {
  586:         let arr = item
  587:             .get("embedding")
  588:             .and_then(Value::as_array)
  589:             .ok_or("Vettore embedding non valido")?;
  590:         let vec: Vec<f32> = arr
  591:             .iter()
  592:             .filter_map(|v| v.as_f64().map(|f| f as f32))
  593:             .collect();
  594:         embeddings.push(vec);
  595:     }
  596: 
  597:     if embeddings.len() != texts.len() {
  598:         return Err(format!(
  599:             "Numero vettori ricevuti ({}) diverso dai testi inviati ({})",
  600:             embeddings.len(),
  601:             texts.len()
  602:         ));
  603:     }
  604: 
  605:     Ok(embeddings)
  606: }

  608: /// Build contextualized passage text for embedding:
  609: /// - Note title (frontmatter title if available, else file_name without extension)
  610: /// - Document category (folder name e.g. "01_CLIENTS" ... "10_APPROVED_OUTPUTS" or category)
  611: /// - Section locator (p.locator e.g. "Pagina N", "Slide N", "Paragrafi N-M")
  612: /// - Raw passage text
  613: /// Prefix context is capped at 20% of the character length of the passage.
  614: pub fn build_passage_embedding_text(
  615:     title: &str,
  616:     category: &str,
  617:     locator: &str,
  618:     passage_text: &str,
  619: ) -> String {
  620:     let mut ctx_lines = Vec::new();
  621:     let trimmed_title = title.trim();
  622:     if !trimmed_title.is_empty() {
  623:         ctx_lines.push(trimmed_title);
  624:     }
  625:     let trimmed_cat = category.trim();
  626:     if !trimmed_cat.is_empty() {
  627:         ctx_lines.push(trimmed_cat);
  628:     }
  629:     let trimmed_loc = locator.trim();
  630:     if !trimmed_loc.is_empty() {
  631:         ctx_lines.push(trimmed_loc);
  632:     }
  633: 
  634:     if ctx_lines.is_empty() || passage_text.trim().is_empty() {
  635:         return passage_text.to_string();
  636:     }
  637: 
  638:     let raw_ctx = ctx_lines.join("\n");
  639:     let max_len = (passage_text.chars().count() as f64 * 0.20).floor() as usize;
  640:     let final_ctx: String = if raw_ctx.chars().count() > max_len {
  641:         raw_ctx.chars().take(max_len).collect()
  642:     } else {
  643:         raw_ctx
  644:     };
  645: 
  646:     if final_ctx.is_empty() {
  647:         passage_text.to_string()
  648:     } else {
  649:         format!("{}\n{}", final_ctx, passage_text)
  650:     }
  651: }

  742:                             last_updated_at: now_iso(),

  744:                     }
  745:                 } else {
  746:                     EmbeddingsCache {
  747:                         version: 1,
  748:                         model: target_model.to_string(),
  749:                         dimensions: target_dimensions,
  750:                         entries: BTreeMap::new(),
  751:                         last_updated_at: now_iso(),

  753:                 }
  754:             } else {
  755:                 EmbeddingsCache {
  756:                     version: 1,

  758:                     dimensions: target_dimensions,
  759:                     entries: BTreeMap::new(),
  760:                     last_updated_at: now_iso(),
  761:                 }
  762:             }
  763:         } else {

  765:                 version: 1,

  932:                 total_missing,
  933:                 (processed_count as f64 / total_missing as f64) * 100.0
  934:             );
  935:         }
  936:     }
  937: 
  938:     if timing {
  939:         eprintln!(
  940:             "[sync-timing] TOTALE passaggi={} lotto={} totale_s={:.1} richieste_s={:.1} scritture_staging={} scritture_s={:.1}",
  941:             total_missing, batch_size, sync_started.elapsed().as_secs_f64(), t_req_total as f64 / 1000.0, writes, t_write_total as f64 / 1000.0
  942:         );
  943:     }
  944:     if let Some(first_entry) = working_cache.entries.values().next() {

  946:     } else {
  947:         working_cache.dimensions = target_dimensions;
  948:     }
  949:     working_cache.model = target_model.to_string();
  950: 
  951:     if is_model_migration {
  952:         // Migration complete: atomically replace EMBEDDINGS_CACHE.json
  953:         save_embeddings_cache(vault_path, &mut working_cache)?;
  954:         let _ = std::fs::remove_file(&staging_path);
  955:     } else {
  956:         save_embeddings_cache(vault_path, &mut working_cache)?;
  957:     }
  958: 
  959:     embeddings_status(vault_path)
  960: }

  962: /// Semantic score for a document given a query embedding vector.
  963: /// Returns (best_score, matching_passage_id, matching_locator, passage_snippet).
  964: pub fn rank_document_semantic(
  965:     _doc_id: &str,
  966:     passages: &[DocumentPassage],
  967:     query_vector: &[f32],
  968:     cache: &EmbeddingsCache,
  969: ) -> (f32, Option<String>, Option<String>, Option<String>) {
  970:     let mut best_score = 0.0f32;
  971:     let mut best_passage_id = None;
  972:     let mut best_locator = None;
  973:     let mut best_snippet = None;
  974: 
  975:     for p in passages {
  976:         if let Some(entry) = cache.entries.get(&p.passage_id) {
  977:             if entry.sha256 == p.sha256 {
  978:                 let sim = cosine_similarity(query_vector, &entry.vector);
  979:                 if sim > best_score {
  980:                     best_score = sim;
  981:                     best_passage_id = Some(p.passage_id.clone());
  982:                     best_locator = Some(p.locator.clone());
  983:                     best_snippet = Some(p.text.chars().take(240).collect::<String>());
  984:                 }
  985:             }
  986:         }
  987:     }
  988: 
  989:     (best_score, best_passage_id, best_locator, best_snippet)
  990: }

 1528:                 semantic_similarity: None,
 1529:             }
 1530:         } else {
 1531:             continue;
 1532:         };
 1533: 
 1534:         // If semantic found a better matching passage or if locator is missing, enrich item
 1535:         if let Some((_, pid, loc, snip)) = sem_info {
 1536:             if item.matching_locator.is_none() || (lex_item.is_none() && snip.is_some()) {
 1537:                 if let Some(l) = loc { item.matching_locator = Some(l.clone()); }
 1538:                 if let Some(p) = pid { item.matching_passage_id = Some(p.clone()); }
 1539:                 if let Some(s) = snip { item.snippet = s.clone(); }
 1540:             }
 1541:         }
 1542: 
 1543:         let matches_term = item.title.to_lowercase().contains(&term_lower) || item.snippet.to_lowercase().contains(&term_lower);
 1544: 
 1545:         intermediate.push(IntermediateCandidate {
 1546:             item,
 1547:             base_score,
 1548:             sem_sim,
 1549:             matches_term,
 1550:         });
 1551:     }
 1552: 
 1553:     // Min-Max normalization bounds for this query across retrieved candidates
 1554:     let min_lex = intermediate.iter().map(|c| c.base_score).fold(f64::INFINITY, f64::min);
 1555:     let max_lex = intermediate.iter().map(|c| c.base_score).fold(f64::NEG_INFINITY, f64::max);
 1556:     let min_sem = intermediate.iter().map(|c| c.sem_sim).fold(f64::INFINITY, f64::min);
 1557:     let max_sem = intermediate.iter().map(|c| c.sem_sim).fold(f64::NEG_INFINITY, f64::max);
 1558: 
 1559:     struct FusedCandidate {
 1560:         item: SearchResultItem,
 1561:         fused_score: f64,
 1562:     }
 1563: 
 1564:     let mut fused_candidates: Vec<FusedCandidate> = Vec::new();
 1565: 
 1566:     for mut c in intermediate {
 1567:         let lex_norm = if max_lex > min_lex {
 1568:             (c.base_score - min_lex) / (max_lex - min_lex)
 1569:         } else if c.base_score > 0.0 {
 1570:             1.0
 1571:         } else {
 1572:             0.0
 1573:         };
 1574: 
 1575:         let sem_norm = if max_sem > min_sem {
 1576:             (c.sem_sim - min_sem) / (max_sem - min_sem)
 1577:         } else if c.sem_sim > 0.0 {
 1578:             1.0
 1579:         } else {
 1580:             0.0
 1581:         };
 1582: 
 1583:         let exact_bonus = if c.matches_term {
 1584:             if is_exact_code { 0.20 } else { 0.05 }
 1585:         } else {
 1586:             0.0
 1587:         };
 1588: 
 1589:         // C10 formula (F2, k = 0.20): max(lex_norm, sem_norm) + 0.20 * min(lex_norm, sem_norm) + exact_bonus
 1590:         // Ensures documents with a single strong signal (e.g. semantic-only with lex_norm = 0)
 1591:         // preserve their score and rank above mediocre dual-signal documents, while continuing
 1592:         // to reward multi-modal concurrence. Selected via dev tuning on 30 queries (Recall@10 = 0.933).
 1593:         let fused_score = lex_norm.max(sem_norm) + 0.20 * lex_norm.min(sem_norm) + exact_bonus;
 1594:         c.item.score = fused_score;
 1595:         c.item.semantic_similarity = if c.sem_sim > 0.0 { Some(c.sem_sim) } else { None };
 1596:         fused_candidates.push(FusedCandidate { item: c.item, fused_score });
 1597:     }
 1598: 
 1599:     // Sort descending by fused score
 1600:     fused_candidates.sort_by(|a, b| b.fused_score.partial_cmp(&a.fused_score).unwrap_or(std::cmp::Ordering::Equal));
 1601: 
 1602:     let limit = q.limit.unwrap_or(50);
 1603:     let offset = q.offset.unwrap_or(0);
 1604:     let results: Vec<SearchResultItem> = fused_candidates
 1605:         .into_iter()
 1606:         .skip(offset)
 1607:         .take(limit)
 1608:         .map(|c| c.item)
 1609:         .collect();
 1610:     let t_fuse_ms = t_fuse_start.elapsed().as_millis() as u64;
 1611:     let t_search_ms = t_search_start.elapsed().as_millis() as u64;
 1612: 
 1613:     let timings = SearchPhaseTimings {
 1614:         t_index_cache_ms,
 1615:         t_search_ms,
 1616:         t_embed_ms: 0,
 1617:         t_words_ms,
 1618:         t_sem_ms,
 1619:         t_fuse_ms,
 1620:         embed_error: None,
 1621:     };
 1622: 
 1623:     Ok((results, timings))
 1624: }
