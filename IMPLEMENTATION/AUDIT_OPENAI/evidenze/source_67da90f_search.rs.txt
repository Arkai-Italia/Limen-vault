147: fn validate(data: &SearchIndexData) -> Result<(), String> {
148:     if data.version != VERSION {
149:         return Err("Unsupported search index version".into());
150:     }
151:     for (p, d) in &data.documents {
152:         relative(p)?;
153:         if (d.id != id(p) && !d.id.starts_with("doc_"))
154:             || d.relative_path != *p
155:             || d.sha256.len() != 64
156:             || !d
157:                 .sha256
158:                 .bytes()
159:                 .all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b))
160:             || d.mtime_ms > 9_007_199_254_740_991
161:             || !valid_category(&d.category)
162:             || d.status.as_ref().is_some_and(|s| !valid_status(s))
163:         {
164:             return Err("Invalid search document".into());

225: pub fn load_with_vault_path(system: &Dir, vault_path: Option<&Path>) -> Result<Option<std::sync::Arc<SearchIndexData>>, String> {
226:     if !names(system)?.iter().any(|n| n == "SEARCH_INDEX.json") {
227:         return Ok(None);
228:     }
229: 
230:     if let Some(v_path) = vault_path {
231:         let idx_file = v_path.join("00_SYSTEM").join("SEARCH_INDEX.json");
232:         if let Ok(meta) = std::fs::metadata(&idx_file) {
233:             let size = meta.len();
234:             let mtime = meta.modified().unwrap_or(std::time::SystemTime::UNIX_EPOCH);
235:             let current_rev = get_search_index_revision(v_path);
236:             if let Ok(guard) = SEARCH_INDEX_CACHE.lock() {
237:                 if let Some(entry) = guard.get(v_path) {
238:                     let is_high_prec = crate::ai::is_high_precision_fs(v_path);
239:                     let is_mtime_match = if is_high_prec {
240:                         entry.mtime == mtime
241:                     } else {
242:                         entry.mtime == mtime
243:                             || entry.mtime.duration_since(mtime).map(|d| d.as_millis() < 2000).unwrap_or(false)
244:                             || mtime.duration_since(entry.mtime).map(|d| d.as_millis() < 2000).unwrap_or(false)
245:                     };
246:                     if entry.size == size && is_mtime_match && entry.revision == current_rev {
247:                         return Ok(Some(entry.data.clone()));
248:                     }
249:                 }
250:             }
251:         }
252:     }

354: fn patterns() -> &'static Vec<(Regex, &'static str)> {

370: pub fn strip_markdown(raw: &str) -> String {
371:     let mut s = raw.to_string();
372:     for (r, rep) in patterns() {
373:         s = r.replace_all(&s, *rep).into_owned();
374:     }
375:     s.split_whitespace().collect::<Vec<_>>().join(" ")

377: pub fn tokenize_text(raw: &str) -> Vec<String> {
378:     static SEP: OnceLock<Regex> = OnceLock::new();
379:     let lower = normalize_text(&strip_markdown(raw));
380:     SEP.get_or_init(|| Regex::new(r"[^\p{L}\p{N}]+").unwrap())
381:         .split(&lower)
382:         .filter(|t| t.chars().count() >= 2 && !spec().stop_words.contains(*t))
383:         .map(str::to_string)
384:         .collect()
385: }
386: /// Vero se `needle` compare in `hay` delimitato da caratteri non alfanumerici (o dai bordi), cioe' come
387: /// parola o sequenza di parole intere: "personal" non e' contenuto in "personale", "lead" non in "leader".
388: /// Entrambe le stringhe devono essere gia' normalizzate con `normalize_text`.
389: pub fn contains_whole_words(hay: &str, needle: &str) -> bool {
390:     let needle = needle.trim();
391:     if needle.is_empty() {
392:         return false;
393:     }
394:     let h: Vec<char> = hay.chars().collect();
395:     let n: Vec<char> = needle.chars().collect();
396:     if n.len() > h.len() {
397:         return false;
398:     }
399:     let is_word = |c: char| c.is_alphanumeric();
400:     let mut i = 0;
401:     while i + n.len() <= h.len() {
402:         if h[i..i + n.len()] == n[..] {
403:             let before_ok = i == 0 || !is_word(h[i - 1]);
404:             let after_ok = i + n.len() == h.len() || !is_word(h[i + n.len()]);
405:             if before_ok && after_ok {
406:                 return true;
407:             }
408:         }
409:         i += 1;
410:     }
411:     false
412: }

414: pub fn extract_snippet(content: &str, terms: &[String], max: usize) -> String {
415:     if max == 0 {
416:         return String::new();
417:     }
418:     let chars: Vec<_> = strip_markdown(content).chars().collect();
419:     let mut folded = Vec::new();
420:     let mut map = Vec::new();
421:     for (i, c) in chars.iter().enumerate() {
422:         for n in normalize_text(&c.to_string()).chars() {
423:             folded.push(n);
424:             map.push(i);
425:         }
426:     }
427:     let mut found = 0;
428:     for t in terms {
429:         let q: Vec<_> = normalize_text(t).chars().collect();
430:         if !q.is_empty() {
431:             if let Some(i) = folded.windows(q.len()).position(|w| w == q.as_slice()) {
432:                 found = map[i];
433:                 break;
434:             }
435:         }
436:     }
437:     let mut from = found.saturating_sub(max / 2);
438:     if chars.len() - from < max {
439:         from = chars.len().saturating_sub(max);
440:     }
441:     let to = chars.len().min(from.saturating_add(max));
442:     format!(
443:         "{}{}{}",
444:         if from > 0 { "..." } else { "" },
445:         chars[from..to].iter().collect::<String>(),
446:         if to < chars.len() { "..." } else { "" }
447:     )
448: }
449: fn read_file(d: &Dir, name: &str) -> Result<(Vec<u8>, u64), String> {
450:     use cap_std::fs::MetadataExt;
451:     let mut opts = OpenOptions::new();
452:     opts.read(true);
453:     #[cfg(unix)]
454:     {
455:         use std::os::unix::fs::OpenOptionsExt;
456:         opts.custom_flags(libc::O_NOFOLLOW | libc::O_NONBLOCK);
457:     }
458:     let mut f = d.open_with(name, &opts).map_err(err)?;
459:     let a = f.metadata().map_err(err)?;
460:     if !a.is_file() || a.len() > 4 * 1024 * 1024 {
461:         return Err("Not a regular file or exceeds 4 MiB".into());
462:     }
463:     let mut bytes = Vec::new();
464:     (&mut f).take(4*1024*1024+1).read_to_end(&mut bytes).map_err(err)?;
465:     if bytes.len()>4*1024*1024{return Err("Source exceeds 4 MiB".into())}
466:     let b = f.metadata().map_err(err)?;
467:     if a.len() != b.len()
468:         || a.modified().map_err(err)? != b.modified().map_err(err)?
469:     {
470:         return Err("Source changed while reading".into());
471:     }
472:     #[cfg(unix)]
473:     {
474:         use cap_std::fs::MetadataExt;
475:         if a.ctime() != b.ctime() || a.ctime_nsec() != b.ctime_nsec() {
476:             return Err("Source changed while reading".into());
477:         }
478:     }
479:     drop(f);
480:     let time = a
481:         .modified()
482:         .map_err(err)?
483:         .into_std()
484:         .duration_since(std::time::UNIX_EPOCH)
485:         .map(|d| d.as_millis() as u64)
486:         .unwrap_or(0);
487:     Ok((bytes, time))
488: }
489: fn parse(
490:     content: &str,
491:     p: &str,
492:     category: &str,
493:     mtime: u64,
494:     sha: String,
495: ) -> Result<SearchDocumentRecord, String> {
496:     let content = content.strip_prefix('\u{feff}').unwrap_or(content);
497:     let normalized = content.replace("\r\n", "\n");
498:     let fm = if normalized.starts_with("---\n") || normalized == "---" {
499:         let rest = normalized
500:             .strip_prefix("---\n")
501:             .ok_or("Unclosed frontmatter")?;
502:         let end = rest
503:             .match_indices("\n---")
504:             .find(|(i, _)| {
505:                 rest.get(i + 4..)
506:                     .is_some_and(|s| s.is_empty() || s.starts_with('\n'))
507:             })
508:             .map(|(i, _)| i)
509:             .ok_or("Unclosed frontmatter")?;
510:         let v: Value = serde_yaml::from_str(&rest[..end]).map_err(err)?;
511:         if !v.is_object() {
512:             return Err("Frontmatter must be a mapping".into());
513:         }
514:         v
515:     } else {
516:         json!({})
517:     };
518:     let body = patterns()[0].0.replace(content, "").trim().to_string();
519:     for k in [
520:         "id",
521:         "title",
522:         "type",
523:         "client",
524:         "project",
525:         "brand",
526:         "status",
527:         "created_at",
528:         "updated_at",
529:     ] {
530:         if fm.get(k).is_some_and(|v| !v.is_string()) {
531:             return Err(format!("{p}: Invalid {k}"));
532:         }
533:     }
534:     let tags: Vec<String> = match fm.get("tags") {
535:         Some(v) => serde_json::from_value(v.clone()).map_err(err)?,
536:         None => Vec::new(),
537:     };
538:     let val = |k: &str| fm.get(k).and_then(Value::as_str).map(str::to_string);
539:     let cat = val("type").unwrap_or_else(|| category.into());
540:     if !valid_category(&cat) || val("status").is_some_and(|s| !valid_status(&s)) {
541:         return Err(format!("{p}: Invalid category/status"));
542:     }
543:     let fallback = Path::new(p)
544:         .file_stem()
545:         .unwrap_or_default()
546:         .to_string_lossy()
547:         .replace(['-', '_'], " ");
548:     let title = val("title").unwrap_or_else(|| {
549:         body.lines()
550:             .find_map(|l| l.strip_prefix("# "))
551:             .map(str::trim)
552:             .filter(|s| !s.is_empty())
553:             .unwrap_or(&fallback)
554:             .into()
555:     });
556:     let date = chrono::DateTime::from_timestamp_millis(mtime as i64)
557:         .ok_or("Invalid mtime")?
558:         .to_rfc3339_opts(chrono::SecondsFormat::Millis, true);
559:     let doc_id = id(p);
560:     let chunked = crate::catalog::chunk_text_to_passages(&doc_id, &body);
561:     let passages: Vec<SearchPassageRecord> = chunked
562:         .into_iter()
563:         .map(|cp| {
564:             let tokens = tokenize_text(&cp.text);
565:             SearchPassageRecord {
566:                 passage_id: cp.passage_id,
567:                 locator: cp.locator,
568:                 sha256: cp.sha256,
569:                 text: cp.text,
570:                 tokens,
571:             }
572:         })
573:         .collect();
574: 
575:     Ok(SearchDocumentRecord {
576:         id: doc_id,
577:         note_id: val("id"),
578:         relative_path: p.into(),
579:         sha256: sha,
580:         mtime_ms: mtime,
581:         title: title.clone(),
582:         category: cat,
583:         client: val("client"),
584:         project: val("project"),
585:         brand: val("brand"),
586:         tags: tags.clone(),
587:         status: val("status"),
588:         created_at: val("created_at").unwrap_or_else(|| date.clone()),
589:         updated_at: val("updated_at").unwrap_or(date),
590:         tokens: tokenize_text(&format!("{title} {body} {}", tags.join(" "))),
591:         content_preview: body,
592:         passages,
593:     })
594: }
595: fn walk(
596:     d: &Dir,
597:     rel: &str,
598:     cat: &str,
599:     depth: usize,
600:     old: Option<&SearchIndexData>,
601:     docs: &mut BTreeMap<String, SearchDocumentRecord>,
602: ) -> Result<(), String> {
603:     if depth > 64 {
604:         return Err("Directory depth exceeds 64".into());
605:     }
606:     for n in names(d)? {
607:         if n.starts_with('.') {
608:             continue;
609:         }
610:         let p = format!("{rel}/{n}");
611:         let m = d.symlink_metadata(&n).map_err(err)?;
612:         if m.is_symlink() {
613:             return Err(format!("{p}: Symlinks are not allowed"));
614:         }
615:         if m.is_dir() {
616:             walk(&child(d, &n)?, &p, cat, depth + 1, old, docs)?;
617:         } else if !m.is_file() {
618:             return Err(format!("{p}: Not a regular file"));
619:         } else if ["md", "markdown"].contains(
620:             &Path::new(&n)
621:                 .extension()
622:                 .unwrap_or_default()
623:                 .to_string_lossy()
624:                 .to_lowercase()
625:                 .as_str(),
626:         ) {
627:             let (bytes, mtime) = read_file(d, &n)?;
628:             let sha = compute_sha256(&bytes);
629:             let cached = old
630:                 .filter(|i| i.version == VERSION)
631:                 .and_then(|i| i.documents.get(&p));
632:             let doc = if let Some(c) = cached.filter(|c| c.sha256 == sha && c.mtime_ms == mtime) {
633:                 c.clone()
634:             } else {
635:                 parse(&String::from_utf8(bytes).map_err(err)?, &p, cat, mtime, sha)?
636:             };
637:             docs.insert(p, doc);
638:         }
639:     }
640:     Ok(())
641: }
642: pub fn index_vault_search(path: &Path) -> Result<IndexStatusReport, String> {
643:     let r = root(path)?;
644:     let s = child(&r, "00_SYSTEM")?;
645:     if s.create_dir(".search-lock").is_err() {
646:         let previous=child(&s,".search-lock")?;
647:         let owner:Value=serde_json::from_slice(&read(&previous,"owner.json")?).map_err(err)?;
648:         let pid = owner["pid"].as_i64().unwrap_or(0) as i32;
649:         #[cfg(unix)]
650:         let active = unsafe { libc::kill(pid, 0) } == 0 || std::io::Error::last_os_error().raw_os_error() != Some(libc::ESRCH);
651:         #[cfg(not(unix))]
652:         let active = pid == std::process::id() as i32;
653:         if active { return Err("Search lock held by an active process".into()); }
654:         previous.remove_file("owner.json").map_err(err)?;
655:         drop(previous);
656:         s.remove_dir(".search-lock").map_err(err)?;
657:         s.create_dir(".search-lock").map_err(err)?;
658:     }
659:     let lock_dir = child(&s, ".search-lock")?;
660:     write_new(&lock_dir, "owner.json", json!({"pid": std::process::id()}).to_string().as_bytes())?;
661:     let _lock = Lock { system: &s, opened: Some(lock_dir) };
662:     let old = load_with_vault_path(&s, Some(path))?;
663:     let mut docs = BTreeMap::new();
664:     let entries = names(&r)?;
665:     for (cat, kind) in &spec().categories {
666:         if entries.contains(cat) {
667:             walk(&child(&r, cat)?, cat, kind, 0, old.as_ref().map(|v| &**v), &mut docs)?;
668:         }
669:     }
670: 
671:     // Sync catalog so new, modified, or removed raw sources are updated
672:     let _ = crate::catalog::sync_catalog(path);
673: 
674:     // Index catalog raw sources (20_RAW_SOURCES) if VAULT_CATALOG.json exists
675:     if let Ok(catalog) = crate::catalog::load_catalog_arc(path) {
676:         for doc in catalog.documents.values() {
677:             if doc.original_path.starts_with("20_RAW_SOURCES/") && doc.extraction_status == crate::catalog::ExtractionStatus::Ready && !doc.passages.is_empty() {
678:                 let p = doc.original_path.clone();
679:                 let cached = old
680:                     .as_ref()
681:                     .map(|v| &**v)
682:                     .filter(|i| i.version == VERSION)
683:                     .and_then(|i| i.documents.get(&p));
684:                 let search_doc = if let Some(c) = cached.filter(|c| c.sha256 == doc.content_hash) {
685:                     c.clone()
686:                 } else {
687:                     let passages: Vec<SearchPassageRecord> = doc.passages.iter().map(|pass| {
688:                         SearchPassageRecord {
689:                             passage_id: pass.passage_id.clone(),
690:                             locator: pass.locator.clone(),
691:                             sha256: pass.sha256.clone(),
692:                             text: pass.text.clone(),
693:                             tokens: tokenize_text(&pass.text),
694:                         }
695:                     }).collect();
696: 
697:                     let mut all_tokens = tokenize_text(&format!("{} {}", doc.file_name, doc.tags.join(" ")));
698:                     for pass in &passages {
699:                         all_tokens.extend(pass.tokens.clone());
700:                     }
701:                     // C11 fix: do not deduplicate tokens so term frequency (tf) accurately reflects occurrences.
702:                     // tokens is never used via binary_search, so full bag-of-words ordering is preserved identically to knowledge documents (line 455).
703: 
704:                     let preview = passages.first().map(|pass| pass.text.chars().take(500).collect::<String>()).unwrap_or_default();
705: 
706:                     SearchDocumentRecord {
707:                         id: doc.document_id.clone(),
708:                         note_id: None,
709:                         relative_path: p.clone(),
710:                         sha256: doc.content_hash.clone(),
711:                         mtime_ms: 0,
712:                         title: doc.file_name.clone(),
713:                         category: doc.category.clone().unwrap_or_else(|| "raw_source".to_string()),
714:                         client: doc.client.clone(),
715:                         project: doc.project.clone(),
716:                         brand: None,
717:                         tags: doc.tags.clone(),
718:                         status: Some(doc.editorial_status.clone()),
719:                         created_at: doc.imported_at.clone(),
720:                         updated_at: doc.updated_at.clone(),
721:                         tokens: all_tokens,
722:                         content_preview: preview,
723:                         passages,
724:                     }
725:                 };
726:                 docs.insert(p, search_doc);
727:             }
728:         }
729:     }
730: 
731:     let data = SearchIndexData {
732:         version: VERSION,
733:         last_indexed_at: chrono::Utc::now().to_rfc3339_opts(chrono::SecondsFormat::Millis, true),
734:         documents: docs,
735:     };
736:     save(&s, &data)?;
737:     bump_search_index_revision(path);
738:     Ok(status(Some(&data)))
739: }

744: pub fn search_vault_filtered<F>(
745:     path: &Path,
746:     q: SearchQuery,
747:     filter_fn: Option<F>,
748: ) -> Result<Vec<SearchResultItem>, String>
749: where
750:     F: Fn(&SearchDocumentRecord) -> bool,
751: {
752:     let limit = q.limit.unwrap_or(50);
753:     let offset = q.offset.unwrap_or(0);
754:     if limit > 200
755:         || offset > 1_000_000
756:         || q.term
757:             .as_ref()
758:             .is_some_and(|t| t.encode_utf16().count() > 2000)
759:     {
760:         return Err("Invalid search query/pagination".into());
761:     }
762:     let r = root(path)?;
763:     let _sys = child(&r, "00_SYSTEM")?;
764:     let data = load_with_vault_path(&child(&r, "00_SYSTEM")?, Some(path))?
765:         .filter(|d| d.version == VERSION)
766:         .ok_or("Search index missing or outdated; re-index required")?;
767:     let terms: Vec<_> = tokenize_text(q.term.as_deref().unwrap_or(""))
768:         .into_iter()
769:         .collect::<BTreeSet<_>>()
770:         .into_iter()
771:         .collect();
772:     if !q.term.as_deref().unwrap_or("").trim().is_empty() && terms.is_empty() {
773:         return Ok(Vec::new());
774:     }
775:     let df: BTreeMap<_, _> = terms
776:         .iter()
777:         .map(|t| {
778:             (
779:                 t,
780:                 data.documents
781:                     .values()
782:                     .filter(|d| d.tokens.contains(t))
783:                     .count(),
784:             )
785:         })
786:         .collect();
787:     // BM25 (C11, seconda meta'): normalizzazione sulla lunghezza del documento.
788:     // Senza questo fattore, con tf reale i documenti lunghi dominano per costruzione
789:     // (misurato: il file piu' grande del corpus al rango 1 in 47 query su 82).
790:     const BM25_K1: f64 = 1.2;
791:     const BM25_B: f64 = 0.75;
792:     let avgdl = {
793:         let n = data.documents.len().max(1) as f64;
794:         (data.documents.values().map(|d| d.tokens.len() as f64).sum::<f64>() / n).max(1.0)
795:     };
796:     let mut results = Vec::new();
797:     for d in data.documents.values() {
798:         if crate::automation::managed(&d.relative_path) && !crate::automation::is_current(path,&d.relative_path,&d.sha256){continue;}
799:         if let Some(ref f) = filter_fn {
800:             if !f(d) {
801:                 continue;
802:             }
803:         }
804:         if q.category
805:             .as_ref()
806:             .is_some_and(|c| !c.is_empty() && *c != d.category)
807:         {
808:             continue;
809:         }
810:         if [
811:             (&q.client, &d.client),
812:             (&q.project, &d.project),
813:             (&q.status, &d.status),
814:         ]
815:         .iter()
816:         .any(|(a, b)| {
817:             a.as_ref().is_some_and(|s| {
818:                 !s.is_empty() && normalize_text(s) != normalize_text(b.as_deref().unwrap_or(""))
819:             })
820:         }) {
821:             continue;
822:         }
823:         if q.tags.as_ref().is_some_and(|ts| {
824:             ts.iter().any(|t| {
825:                 !d.tags
826:                     .iter()
827:                     .any(|x| normalize_text(x) == normalize_text(t))
828:             })
829:         }) {
830:             continue;
831:         }
832:         let mut score = if terms.is_empty() { 1.0 } else { 0.0 };
833:         let title_tokens = tokenize_text(&d.title);
834:         let tags_tokens = tokenize_text(&d.tags.join(" "));
835:         let dl = d.tokens.len() as f64;
836:         let length_norm = 1.0 - BM25_B + BM25_B * dl / avgdl;
837:         let doc_count = data.documents.len().max(1);
838:         for t in &terms {
839:             let tf = d.tokens.iter().filter(|x| *x == t).count() as f64;
840:             let idf =
841:                 ((data.documents.len() + 1) as f64 / (*df.get(t).unwrap_or(&0) + 1) as f64).ln() + 1.0;
842:             if tf > 0.0 {
843:                 score += idf * (tf * (BM25_K1 + 1.0)) / (tf + BM25_K1 * length_norm);
844:             }
845:             // FASE 3a (Punto A): bonus del titolo assegnato solo a parole rare / poco frequenti nel vault
846:             // (df <= 2 o df/N <= 0.30, tarato sulle 30 query di sviluppo A05_DEV_QUERIES).
847:             // Evita che parole generiche come "progetto" (df ~ 35.4%) ottengano il bonus scavalcando documenti specifici.
848:             let term_df = *df.get(t).unwrap_or(&0);
849:             let is_infrequent = term_df <= 2 || (term_df as f64 / doc_count as f64) <= 0.30;
850:             if title_tokens.contains(t) && is_infrequent {
851:                 score += 10.0;
852:             }
853:             if tags_tokens.contains(t) {
854:                 score += 5.0;
855:             }
856:         }
857: 
858:         // Passages matching and locator detection
859:         let mut matching_passages = Vec::new();
860:         let mut best_passage_locator = None;
861:         let mut best_passage_id = None;
862:         let mut best_passage_snippet = None;
863:         let mut best_passage_score = 0.0;
864: 
865:         let dynamic_passages: Vec<SearchPassageRecord>;
866:         let passages_ref = if !d.passages.is_empty() {
867:             &d.passages
868:         } else {
869:             dynamic_passages = crate::catalog::chunk_text_to_passages(&d.id, &d.content_preview)
870:                 .into_iter()
871:                 .map(|p| SearchPassageRecord {
872:                     passage_id: p.passage_id,
873:                     locator: p.locator,
874:                     sha256: p.sha256,
875:                     text: p.text.clone(),
876:                     tokens: tokenize_text(&p.text),
877:                 })
878:                 .collect();
879:             &dynamic_passages
880:         };
881: 
882:         for p in passages_ref {
883:             let mut p_score = 0.0;
884:             for t in &terms {
885:                 let p_tf = p.tokens.iter().filter(|x| *x == t).count() as f64;
886:                 if p_tf > 0.0 {
887:                     let idf = ((data.documents.len() + 1) as f64 / (*df.get(t).unwrap_or(&0) + 1) as f64).ln() + 1.0;
888:                     p_score += p_tf * idf;
889:                 }
890:             }
891:             if let Some(ref term) = q.term {
892:                 let norm_term = normalize_text(term);
893:                 let norm_p = normalize_text(&p.text);
894:                 // C12 (21/09/2026): confronto a parola intera. Con `contains` un termine breve prendeva il bonus
895:                 // anche come sottostringa di altre parole ("personal" in "personale", "lead" in "leader") e,
896:                 // poiche' il bonus vale solo per i documenti con BM25 zero ed e' piu' alto di qualsiasi BM25,
897:                 // i documenti con la sola sottostringa superavano quelli con il termine vero: misurato sul
898:                 // corpus reale, 50 documenti a pari merito con punteggio 15,0 per "personal", "cita", "lead".
899:                 if contains_whole_words(&norm_p, &norm_term) {
900:                     p_score += 15.0;
901:                 }
902:             }
903:             if p_score > 0.0 {
904:                 let snippet = extract_snippet(&p.text, &terms, 180);
905:                 matching_passages.push(SearchMatchingPassage {
906:                     passage_id: p.passage_id.clone(),
907:                     locator: p.locator.clone(),
908:                     snippet: snippet.clone(),
909:                     score: (p_score * 100.0).round() / 100.0,
910:                 });
911:                 if p_score > best_passage_score {
912:                     best_passage_score = p_score;
913:                     best_passage_locator = Some(p.locator.clone());
914:                     best_passage_id = Some(p.passage_id.clone());
915:                     best_passage_snippet = Some(snippet);
916:                 }
917:             }
918:         }
919: 
920:         if score <= 0.0 {
921:             if best_passage_score > 0.0 {
922:                 score = best_passage_score;
923:             } else {
924:                 continue;
925:             }
926:         }
927: 
928:         if compute_sha256(&read_path(&r, &d.relative_path)?) != d.sha256 {
929:             return Err(format!(
930:                 "Search index stale: {}; re-index required",
931:                 d.relative_path
932:             ));
933:         }
934: 
935:         let snippet = best_passage_snippet.unwrap_or_else(|| {
936:             extract_snippet(&d.content_preview, &terms, 180)
937:         });
938: 
939:         matching_passages.sort_by(|a, b| b.score.total_cmp(&a.score));
940: 
941:         results.push(SearchResultItem {
942:             id: d.id.clone(),
943:             title: d.title.clone(),
944:             relative_path: d.relative_path.clone(),
945:             category: d.category.clone(),
946:             client: d.client.clone(),
947:             project: d.project.clone(),
948:             tags: d.tags.clone(),
949:             status: d.status.clone(),
950:             snippet,
951:             score: (score * 100.0).round() / 100.0,
952:             updated_at: Some(d.updated_at.clone()),
953:             sha256: d.sha256.clone(),
954:             matching_locator: best_passage_locator,
955:             matching_passage_id: best_passage_id,
956:             passages: matching_passages,
957:             semantic_similarity: None,
958:         });
959:     }
960:     results.sort_by(|a, b| {
961:         b.score
962:             .total_cmp(&a.score)
963:             .then_with(|| b.updated_at.cmp(&a.updated_at))
964:             .then_with(|| a.relative_path.cmp(&b.relative_path))
965:     });
966:     Ok(results.into_iter().skip(offset).take(limit).collect())
967: }

1318: pub fn get_search_document(path: &Path, document_id: &str) -> Result<SearchDocumentRecord, String> {

1326: /// Read the exact indexed bytes or extracted text through a pinned directory, for AI/MCP citations.
1327: pub fn read_indexed_document(path: &Path, document_id: &str, expected: &str) -> Result<(SearchDocumentRecord, String), String> {
1328:     let r=root(path)?;
1329:     let data=load_with_vault_path(&child(&r,"00_SYSTEM")?, Some(path))?.filter(|d|d.version==VERSION).ok_or("Search index missing or outdated")?;
1330:     let d=data.documents.values().find(|d|d.id==document_id).ok_or("Unknown document")?;
1331:     if d.sha256!=expected {return Err("Source changed; select again".into())}
1332:     relative(&d.relative_path)?;
1333:     let parts:Vec<_>=d.relative_path.split('/').collect();
1334: 
1335:     // SPECIAL HANDLING FOR 20_RAW_SOURCES (PDF, DOCX, binary sources)
1336:     if parts[0] == "20_RAW_SOURCES" {
1337:         let mut dir=r;
1338:         for part in &parts[..parts.len()-1]{dir=child(&dir,part)?;}
1339:         let (bytes,_)=read_file(&dir,parts[parts.len()-1])?;
1340:         if compute_sha256(&bytes)!=expected{return Err("Source changed; re-index required".into())}
1341: 
1342:         // Return extracted plain text rather than binary bytes
1343:         let content = if !d.passages.is_empty() {
1344:             d.passages.iter().map(|p| p.text.as_str()).collect::<Vec<_>>().join("\n\n")
1345:         } else {
1346:             crate::catalog::read_document_text(path, &d.id).unwrap_or_else(|_| d.content_preview.clone())
1347:         };
1348:         if content.trim().is_empty() {
1349:             return Err("Extracted text empty for raw source".into());
1350:         }
1351:         return Ok((d.clone(), content));
1352:     }
1353: 
1354:     let mut dir=r;
1355:     for part in &parts[..parts.len()-1]{dir=child(&dir,part)?;}
1356:     let (bytes,_)=read_file(&dir,parts[parts.len()-1])?;
1357:     if bytes.len()>256*1024{return Err("AI document exceeds 256 KiB".into())}
1358:     if compute_sha256(&bytes)!=expected{return Err("Source changed; re-index required".into())}
1359:     let content=String::from_utf8(bytes).map_err(|_|"Invalid UTF-8")?;
1360:     let category=spec().categories.get(parts[0]).ok_or("Invalid category directory")?;
1361:     let actual=parse(&content,&d.relative_path,category,d.mtime_ms,d.sha256.clone())?;
1362:     if actual.status!=d.status||actual.category!=d.category||actual.client!=d.client||actual.project!=d.project||actual.tags!=d.tags||actual.title!=d.title{return Err("Indexed metadata differs from source; re-index required".into())}
1363:     Ok((actual,content))
1364: }
