227: pub fn log_ask_timing_detailed(entry: &AskTimingLogEntry) {
228:     let log_path = get_ask_timing_log_path();
229:     if let Some(parent) = log_path.parent() {
230:         let _ = std::fs::create_dir_all(parent);
231:     }
232:     let json_line = serde_json::to_string(entry).unwrap_or_default();
233:     let ui_str = entry
234:         .t_ui_total_ms
235:         .map(|ms| format!("{} ms", ms))
236:         .unwrap_or_else(|| "non misurato".into());
237:     let status_str = if entry.status == "incomplete" {
238:         format!("INCOMPLETA ({})", entry.incomplete_reason.as_deref().unwrap_or("max_output_tokens"))
239:     } else if entry.status == "error" || entry.status.starts_with("errore") {
240:         format!("ERRORE ({})", entry.incomplete_reason.as_deref().unwrap_or("errore"))
241:     } else if entry.status == "annullata" || entry.status.starts_with("annullata") {
242:         format!("ANNULLATA ({})", entry.incomplete_reason.as_deref().unwrap_or("annullata"))
243:     } else {
244:         entry.status.clone()
245:     };
246: 
247:     let mut breakdown_parts = Vec::new();
248:     if let Some(inp) = entry.tokens_prompt {
249:         breakdown_parts.push(format!("input: {}", inp));
250:     }
251:     if let Some(out) = entry.tokens_completion {
252:         breakdown_parts.push(format!("output: {}", out));
253:     }
254:     if let Some(reas) = entry.tokens_reasoning {
255:         breakdown_parts.push(format!("ragionamento: {}", reas));
256:     }
257:     let tokens_str = match (entry.tokens_used, breakdown_parts.is_empty()) {
258:         (Some(t), false) => format!("{} ({})", t, breakdown_parts.join(", ")),
259:         (Some(t), true) => t.to_string(),
260:         (None, false) => format!("({})", breakdown_parts.join(", ")),
261:         (None, true) => "null".into(),
262:     };
263: 
264:     let mut sources_section = String::new();
265:     if !entry.sources.is_empty() {
266:         let total_bytes: usize = entry.sources.iter().map(|s| s.bytes).sum();
267:         let cited_sources: Vec<&str> = entry
268:             .sources
269:             .iter()
270:             .filter(|s| s.cited)
271:             .map(|s| s.id.as_str())
272:             .collect();
273:         let cited_count = cited_sources.len();
274:         let cited_summary = if cited_count > 0 {
275:             cited_sources.join(", ")
276:         } else {
277:             "nessuna".to_string()
278:         };
279: 
280:         sources_section.push_str(&format!(
281:             "Fonti inviate a OpenAI ({} fonti, {} byte totali):\n",
282:             entry.sources.len(),
283:             total_bytes
284:         ));
285: 
286:         let num_sources = entry.sources.len();
287:         for (idx, s) in entry.sources.iter().enumerate() {
288:             let branch = if idx == num_sources - 1 {
289:                 "  └─"
290:             } else {
291:                 "  ├─"
292:             };
293:             let loc_str = match &s.locator {
294:                 Some(loc) => format!(", loc: \"{}\"", loc),
295:                 None => String::new(),
296:             };
297:             let status_str = if s.cited { "CITATA" } else { "NON CITATA" };
298:             sources_section.push_str(&format!(
299:                 "{} [{}] {} ({} B{}) -> {}\n",
300:                 branch, s.id, s.relative_path, s.bytes, loc_str, status_str
301:             ));
302:         }
303:         sources_section.push_str(&format!(
304:             "Riepilogo citazioni: {} citata/e su {} consultate ({}).\n",
305:             cited_count, num_sources, cited_summary
306:         ));
307:     }
308: 
309:     let first_chunk_str = entry
310:         .t_first_chunk_ms
311:         .or(entry.ask.t_first_chunk_ms)
312:         .map(|ms| format!(" (primo frammento / t_first_chunk_ms: {} ms)", ms))
313:         .unwrap_or_default();
314: 
315:     let search_sum = entry.preview.search_words_ms
316:         + entry.preview.search_sem_ms
317:         + entry.preview.search_fuse_ms
318:         + entry.preview.search_admit_ms;
319: 
320:     let formatted_block = format!(
321:         "================================================================================\n\
322:          REGISTRO TEMPI RISPOSTA [{}]\n\
323:          Modello: {} | Stato: {} | Token usati: {}\n\
324:          Tempo visto da UI (click -> risposta): {}\n\
325:          Tempo totale Backend: {} ms\n\
326:            ├─ 1. ANTEPRIMA (Preview): {} ms\n\
327:            │    ├─ Indice & cache: {} ms\n\
328:            │    ├─ Vettore semantico (embed): {} ms\n\
329:            │    ├─ Ricerca ibrida (search): {} ms\n\
330:            │    │    ├─ Parole (search_words_ms): {} ms\n\
331:            │    │    ├─ Semantica (search_sem_ms): {} ms\n\
332:            │    │    ├─ Fusione (search_fuse_ms): {} ms\n\
333:            │    │    └─ Ammissibilità (search_admit_ms): {} ms\n\
334:            │    ├─ Lettura documenti (doc_read): {} ms\n\
335:            │    └─ Estrazione passaggi (passage_extract): {} ms\n\
336:            ├─ 2. PASSAGGIO ANTEPRIMA-INVIO (Handoff IPC/UI): {} ms\n\
337:            └─ 3. INTERROGAZIONE (Ask): {} ms\n\
338:                 ├─ Verifica impronte pre-chiamata (verify_pre): {} ms\n\
339:                 ├─ Costruzione richiesta (payload): {} ms\n\
340:                 ├─ Chiamata OpenAI (openai): {} ms{}\n\
341:                 ├─ Verifica impronte post-chiamata (verify_post): {} ms\n\
342:                 └─ Lettura risposta (parse): {} ms\n\
343:          Verifica quadratura somme:\n\
344:            Backend = Preview ({} ms) + Handoff ({} ms) + Ask ({} ms) = {} ms\n\
345:            Search  = Words ({} ms) + Sem ({} ms) + Fuse ({} ms) + Admit ({} ms) = {} ms (vs search_ms: {} ms)\n\
346:            Ask     = Verify_pre ({} ms) + Payload ({} ms) + OpenAI ({} ms) + Verify_post ({} ms) + Parse ({} ms) = {} ms\n\
347:          {}\
348:          JSON: {}\n\
349:          ================================================================================\n\n",
350:         entry.timestamp,
351:         entry.model,
352:         status_str,
353:         tokens_str,
354:         ui_str,
355:         entry.t_backend_total_ms,
356:         entry.preview.total_ms,
357:         entry.preview.index_cache_ms,
358:         entry.preview.embed_ms,
359:         entry.preview.search_ms,
360:         entry.preview.search_words_ms,
361:         entry.preview.search_sem_ms,
362:         entry.preview.search_fuse_ms,
363:         entry.preview.search_admit_ms,
364:         entry.preview.doc_read_ms,
365:         entry.preview.passage_extract_ms,
366:         entry.t_handoff_ms,
367:         entry.ask.total_ms,
368:         entry.ask.verify_pre_ms,
369:         entry.ask.payload_ms,
370:         entry.ask.openai_ms,
371:         first_chunk_str,
372:         entry.ask.verify_post_ms,
373:         entry.ask.parse_ms,
374:         entry.preview.total_ms,
375:         entry.t_handoff_ms,
376:         entry.ask.total_ms,
377:         entry.preview.total_ms + entry.t_handoff_ms + entry.ask.total_ms,
378:         entry.preview.search_words_ms,
379:         entry.preview.search_sem_ms,
380:         entry.preview.search_fuse_ms,
381:         entry.preview.search_admit_ms,
382:         search_sum,
383:         entry.preview.search_ms,
384:         entry.ask.verify_pre_ms,
385:         entry.ask.payload_ms,
386:         entry.ask.openai_ms,
387:         entry.ask.verify_post_ms,
388:         entry.ask.parse_ms,
389:         entry.ask.verify_pre_ms + entry.ask.payload_ms + entry.ask.openai_ms + entry.ask.verify_post_ms + entry.ask.parse_ms,
390:         sources_section,
391:         json_line
392:     );
393: 
394:     if let Ok(mut f) = std::fs::OpenOptions::new().create(true).append(true).open(&log_path) {
395:         use std::io::Write;
396:         let _ = f.write_all(formatted_block.as_bytes());
397:     }
398: }

629: /// Verify source integrity using filesystem metadata on NTFS / high-precision filesystems,
630: /// or full SHA-256 recalculation on FAT32/exFAT filesystems.
631: pub fn verify_source_integrity_with_fs_override(
632:     path: &Path,
633:     s: &Source,
634:     is_high_precision: bool,
635:     drafts: bool,
636: ) -> Result<(), String> {
637:     let full_path = path.join(&s.relative_path);
638:     let meta = std::fs::metadata(&full_path).map_err(|_| "Source file missing or inaccessible".to_string())?;
639: 
640:     // In-memory checks on search index, catalog, automation state, and eligibility
641:     let d = search::get_search_document(path, &s.document_id)?;
642:     if d.sha256 != s.sha256 {
643:         return Err("Source changed".into());
644:     }
645: 
646:     // 1. Check if automation-managed source has become obsolete
647:     if crate::automation::managed(&d.relative_path) && !crate::automation::is_current(path, &d.relative_path, &s.sha256) {
648:         return Err("Generated source obsolete or modified".into());
649:     }
650: 
651:     // 2. Check document eligibility (category, status, drafts)
652:     if !eligible(&d.relative_path, &d.category, d.status.as_deref(), drafts)
653:         && !crate::automation::is_current(path, &d.relative_path, &d.sha256)
654:     {
655:         return Err("Document access denied".into());
656:     }
657: 
658:     // 3. Verifica crittografica puntuale di ciascun singolo passaggio associato alla fonte (Punto C)
659:     for (pid, expected_sha) in &s.passage_hashes {
660:         let passage = crate::catalog::read_passage(path, &s.document_id, pid)
661:             .map_err(|e| format!("Integrità passaggio {} compromessa: {}", pid, e))?;
662:         if &passage.sha256 != expected_sha {
663:             return Err(format!(
664:                 "Integrità passaggio {} compromessa (hash atteso {}, trovato {})",
665:                 pid, expected_sha, passage.sha256
666:             ));
667:         }
668:     }
669: 
670:     if is_high_precision {
671:         let current_size = meta.len();
672:         let current_mtime_ms = meta
673:             .modified()
674:             .ok()
675:             .and_then(|t| t.duration_since(std::time::UNIX_EPOCH).ok())
676:             .map(|dur| dur.as_millis() as u64);
677: 
678:         if let (Some(expected_size), Some(expected_mtime)) = (s.file_size, s.mtime_ms) {
679:             if current_size == expected_size && current_mtime_ms == Some(expected_mtime) {
680:                 // In NTFS with nanosecond resolution, unmodified mtime + size guarantees content integrity
681:                 return Ok(());
682:             }
683:         }
684:     }
685: 
686:     // On non-NTFS (FAT32/exFAT with 2-second timestamp resolution) or if metadata changed,
687:     // recalculate SHA-256 using in-memory catalog and search index cache.
688:     let current = read_source(path, &s.document_id, &s.sha256, drafts)?;
689:     if current.sha256 != s.sha256 {
690:         return Err("Source changed".into());
691:     }
692:     Ok(())
693: }

696:     let high_prec = is_high_precision_fs(path);
697:     verify_source_integrity_with_fs_override(path, s, high_prec, drafts)

1443:         .lines()
1444:         .map(|line| {
1445:             let p = re_space_punct.replace_all(line, "$1");
1446:             let s = re_multi_space.replace_all(&p, " ");

1448:                 s.trim_start()
1449:             } else {
1450:                 &s
1451:             };
1452:             trimmed_start.trim_end().to_string()

1457:     if text.ends_with('\n') && !res.ends_with('\n') {
1458:         res.push('\n');
1459:     }

1794: /// Sanitizzatore incrementale per lo streaming della prosa.
1795: /// Dotato di sliding tail buffer: trattiene la porzione terminale ambigua (fino a 25 caratteri)
1796: /// se coincide con un prefisso di citazione (`[`, `[S`, `(`, `(S`) o di marcatore markdown (`*`, `**`),
1797: /// garantendo che nessun frammento grezzo o asterisco appaia temporaneamente all'utente.
1798: #[derive(Debug, Default)]
1799: pub struct StreamProseSanitizer {
1800:     raw_buffer: String,
1801:     clean_accumulated: String,
1802: }
1803: 
1804: impl StreamProseSanitizer {
1805:     pub fn new() -> Self {
1806:         Self::default()
1807:     }
1808: 
1809:     /// Determina la lunghezza del suffisso ambiguo da trattenere nel buffer scorrevole.
1810:     /// Garantisce sempre che qualsiasi slicing avvenga su confini di caratteri UTF-8 validi (previene panic su lettere accentate ed emoji).
1811:     pub fn ambiguous_suffix_len(s: &str) -> usize {
1812:         let len = s.len();
1813:         if len == 0 {
1814:             return 0;
1815:         }
1816: 
1817:         // Helper per arretrare al confine di carattere UTF-8 valido più vicino (mai panic)
1818:         let safe_boundary = |mut idx: usize| -> usize {
1819:             while idx > 0 && !s.is_char_boundary(idx) {
1820:                 idx -= 1;
1821:             }
1822:             idx
1823:         };
1824: 
1825:         // 1. Prefisso di citazione aperto: es. `[` senza `]` o `(` senza `)` (finestra fino a 30 byte/caratteri)
1826:         let check_window = safe_boundary(len.saturating_sub(30));
1827:         if let Some(pos) = s[check_window..].rfind(|c| c == '[' || c == '(') {
1828:             let abs_pos = check_window + pos;
1829:             let candidate_slice = &s[abs_pos..];
1830:             let has_closing = candidate_slice.contains(']') || candidate_slice.contains(')');
1831:             if !has_closing {
1832:                 let inside = &candidate_slice[1..];
1833:                 let is_citation_prefix = inside.chars().all(|c| {
1834:                     c.is_ascii_whitespace() || c == 'S' || c == 's' || c.is_ascii_digit() || c == ',' || c == ';'
1835:                 });
1836:                 if is_citation_prefix {
1837:                     return len - abs_pos;
1838:                 }
1839:             }
1840:         }
1841: 
1842:         // 2. Grassetto markdown `**` aperto (conteggio dispari nella finestra di 60 byte/caratteri)
1843:         let star_window = safe_boundary(len.saturating_sub(60));
1844:         let star_slice = &s[star_window..];
1845:         let dstar_count = star_slice.matches("**").count();
1846:         if dstar_count % 2 != 0 {
1847:             if let Some(pos) = star_slice.rfind("**") {
1848:                 return len - (star_window + pos);
1849:             }
1850:         }
1851: 
1852:         // 3. Corsivo markdown `*` aperto (conteggio dispari escludendo `**`)
1853:         let without_dstar = star_slice.replace("**", "");
1854:         let single_star_count = without_dstar.chars().filter(|&c| c == '*').count();
1855:         if single_star_count % 2 != 0 {
1856:             if let Some(pos) = star_slice.rfind('*') {
1857:                 return len - (star_window + pos);
1858:             }
1859:         }
1860: 
1861:         0
1862:     }
1863: 
1864:     /// Ingestiona un nuovo frammento grezzo di prosa, sanifica la porzione sicura ed emette solo il delta pulito.
1865:     pub fn feed(&mut self, chunk: &str) -> String {
1866:         self.raw_buffer.push_str(chunk);
1867:         let suffix_len = Self::ambiguous_suffix_len(&self.raw_buffer);
1868:         let safe_boundary = self.raw_buffer.len().saturating_sub(suffix_len);
1869: 
1870:         let mut char_safe_boundary = safe_boundary;
1871:         while char_safe_boundary > 0 && !self.raw_buffer.is_char_boundary(char_safe_boundary) {
1872:             char_safe_boundary -= 1;
1873:         }
1874: 
1875:         let safe_raw = &self.raw_buffer[..char_safe_boundary];
1876:         let sanitized = sanitize_answer_prose(safe_raw);
1877: 
1878:         if sanitized.len() > self.clean_accumulated.len() && sanitized.starts_with(&self.clean_accumulated) {
1879:             let delta = sanitized[self.clean_accumulated.len()..].to_string();
1880:             self.clean_accumulated = sanitized;
1881:             delta
1882:         } else if sanitized != self.clean_accumulated {
1883:             let common_prefix_len = sanitized
1884:                 .chars()
1885:                 .zip(self.clean_accumulated.chars())
1886:                 .take_while(|(a, b)| a == b)
1887:                 .map(|(a, _)| a.len_utf8())
1888:                 .sum::<usize>();
1889:             if sanitized.len() > common_prefix_len && common_prefix_len == self.clean_accumulated.len() {
1890:                 let delta = sanitized[self.clean_accumulated.len()..].to_string();
1891:                 self.clean_accumulated = sanitized;
1892:                 delta
1893:             } else {
1894:                 String::new()
1895:             }
1896:         } else {
1897:             String::new()
1898:         }
1899:     }
1900: 
1901:     /// Svuota l'eventuale residuo trattenuto nel buffer a fine stream.
1902:     pub fn flush(&mut self) -> String {
1903:         let sanitized = sanitize_answer_prose(&self.raw_buffer);
1904:         if sanitized.len() > self.clean_accumulated.len() && sanitized.starts_with(&self.clean_accumulated) {
1905:             let delta = sanitized[self.clean_accumulated.len()..].to_string();
1906:             self.clean_accumulated = sanitized;
1907:             delta
1908:         } else if sanitized != self.clean_accumulated {
1909:             let common_prefix_len = sanitized
1910:                 .chars()
1911:                 .zip(self.clean_accumulated.chars())
1912:                 .take_while(|(a, b)| a == b)
1913:                 .map(|(a, _)| a.len_utf8())
1914:                 .sum::<usize>();
1915:             let delta = if sanitized.len() > common_prefix_len {
1916:                 sanitized[common_prefix_len..].to_string()
1917:             } else {
1918:                 String::new()
1919:             };
1920:             self.clean_accumulated = sanitized;
1921:             delta
1922:         } else {
1923:             String::new()
1924:         }
1925:     }
1926: 
1927:     pub fn get_accumulated(&self) -> &str {
1928:         &self.clean_accumulated
1929:     }

1962: impl JsonStreamAnswerParser {
1963:     pub fn new() -> Self {
1964:         Self::default()
1965:     }
1966: 
1967:     pub fn feed(&mut self, chunk: &str) -> String {
1968:         let mut emitted = String::new();
1969:         let mut chars = chunk.chars().peekable();
1970: 
1971:         while let Some(c) = chars.next() {
1972:             match self.state {
1973:                 JsonParserState::SeekingKey => {
1974:                     self.raw_buffer.push(c);
1975:                     if let Some(pos) = self.raw_buffer.find("\"answer\"") {
1976:                         let after_key = &self.raw_buffer[pos + 8..];
1977:                         if let Some(quote_pos) = after_key.find('"') {
1978:                             let between = &after_key[..quote_pos];
1979:                             if between.contains(':') {
1980:                                 self.state = JsonParserState::InAnswerString;
1981:                                 self.raw_buffer.clear();
1982:                             }
1983:                         }
1984:                     }
1985:                 }
1986:                 JsonParserState::InAnswerString => {
1987:                     if self.in_escape {
1988:                         if self.in_unicode {
1989:                             self.unicode_buffer.push(c);
1990:                             if self.unicode_buffer.len() == 4 {
1991:                                 if let Ok(val) = u16::from_str_radix(&self.unicode_buffer, 16) {
1992:                                     if let Some(ch) = char::from_u32(val as u32) {
1993:                                         emitted.push(ch);
1994:                                     }
1995:                                 }
1996:                                 self.in_escape = false;
1997:                                 self.in_unicode = false;
1998:                                 self.unicode_buffer.clear();
1999:                             }
2000:                         } else {
2001:                             match c {
2002:                                 '"' => { emitted.push('"'); self.in_escape = false; }
2003:                                 '\\' => { emitted.push('\\'); self.in_escape = false; }
2004:                                 '/' => { emitted.push('/'); self.in_escape = false; }
2005:                                 'b' => { self.in_escape = false; }
2006:                                 'f' => { self.in_escape = false; }
2007:                                 'n' => { emitted.push('\n'); self.in_escape = false; }
2008:                                 'r' => { emitted.push('\r'); self.in_escape = false; }
2009:                                 't' => { emitted.push('\t'); self.in_escape = false; }
2010:                                 'u' => {
2011:                                     self.in_unicode = true;
2012:                                     self.unicode_buffer.clear();
2013:                                 }
2014:                                 other => {
2015:                                     emitted.push(other);
2016:                                     self.in_escape = false;
2017:                                 }
2018:                             }
2019:                         }
2020:                     } else if c == '\\' {
2021:                         self.in_escape = true;
2022:                     } else if c == '"' {
2023:                         self.state = JsonParserState::FinishedAnswer;
2024:                     } else {
2025:                         emitted.push(c);
2026:                     }
2027:                 }
2028:                 JsonParserState::FinishedAnswer => {}
2029:             }
2030:         }
2031:         emitted
2032:     }
2033: 
2034:     pub fn is_finished(&self) -> bool {
2035:         self.state == JsonParserState::FinishedAnswer
2036:     }
2037: }

2039: /// Decodificatore incrementale per stream di byte di rete:
2040: /// trattiene i byte di sequenze UTF-8 multibyte incomplete alla fine di un frammento,
2041: /// garantendo che nessun carattere multibyte (es. "è", "à", "€", emoji) venga convertito
2042: /// nel carattere di rimpiazzo ("") sui confini dei pacchetti di rete.
2043: #[derive(Debug, Default)]
2044: pub struct Utf8ChunkDecoder {
2045:     buffer: Vec<u8>,
2046: }
2047: 
2048: impl Utf8ChunkDecoder {
2049:     pub fn new() -> Self {
2050:         Self { buffer: Vec::new() }
2051:     }
2052: 
2053:     pub fn decode(&mut self, chunk: &[u8]) -> String {
2054:         self.buffer.extend_from_slice(chunk);
2055:         let valid_len = match std::str::from_utf8(&self.buffer) {
2056:             Ok(s) => s.len(),
2057:             Err(e) => {
2058:                 if e.error_len().is_none() {
2059:                     // Sequenza UTF-8 multibyte incompleta alla fine del buffer
2060:                     e.valid_up_to()
2061:                 } else {
2062:                     let valid_up_to = e.valid_up_to();
2063:                     if valid_up_to > 0 {
2064:                         valid_up_to
2065:                     } else {
2066:                         // Byte corrotto o non valido: consuma un byte per evitare deadlock
2067:                         self.buffer.remove(0);
2068:                         return "\u{FFFD}".to_string();
2069:                     }
2070:                 }
2071:             }
2072:         };
2073: 
2074:         if valid_len > 0 {
2075:             let valid_str = match std::str::from_utf8(&self.buffer[..valid_len]) {
2076:                 Ok(s) => s.to_string(),
2077:                 Err(_) => String::from_utf8_lossy(&self.buffer[..valid_len]).to_string(),
2078:             };
2079:             self.buffer.drain(..valid_len);
2080:             valid_str
2081:         } else {
2082:             String::new()
2083:         }
2084:     }
2085: 
2086:     pub fn flush(&mut self) -> String {
2087:         if self.buffer.is_empty() {
2088:             String::new()
2089:         } else {
2090:             let s = String::from_utf8_lossy(&self.buffer).to_string();
2091:             self.buffer.clear();
2092:             s
2093:         }
2094:     }
2095: }
