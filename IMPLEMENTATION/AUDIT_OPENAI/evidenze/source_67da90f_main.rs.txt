219: fn obsidian_path() -> Option<PathBuf> {
220:     let mut paths = vec![PathBuf::from("/Applications/Obsidian.app")];
221:     if let Ok(home) = std::env::var("HOME") {
222:         paths.push(Path::new(&home).join("Applications/Obsidian.app"));
223:     }
224:     if let Some(p) = paths.into_iter().find(|p| p.is_dir()) {
225:         return Some(p);
226:     }
227:     let output = Command::new("/usr/bin/mdfind")
228:         .arg("kMDItemCFBundleIdentifier == 'md.obsidian'")
229:         .output()
230:         .ok()?;
231:     String::from_utf8_lossy(&output.stdout)
232:         .lines()
233:         .map(PathBuf::from)
234:         .find(|p| p.extension().is_some_and(|e| e == "app") && p.is_dir())
235: }

242: #[tauri::command]
243: fn open_obsidian(vault_path: String) -> Result<(), String> {
244:     let app = obsidian_path().ok_or("OBSIDIAN_NOT_INSTALLED")?;
245:     let path = Path::new(&vault_path)
246:         .canonicalize()
247:         .map_err(|e| e.to_string())?;
248:     if !path.is_dir() {
249:         return Err("Vault path is not a directory".into());
250:     }
251:     let registered = std::env::var("HOME")
252:         .ok()
253:         .and_then(|home| {
254:             std::fs::read_to_string(
255:                 Path::new(&home).join("Library/Application Support/obsidian/obsidian.json"),
256:             )
257:             .ok()
258:         })
259:         .and_then(|raw| serde_json::from_str::<serde_json::Value>(&raw).ok())
260:         .is_some_and(|registry| obsidian_registered(&registry, &path));
261:     let uri = if registered {
262:         format!(
263:             "obsidian://open?path={}",
264:             urlencoding::encode(&path.to_string_lossy())
265:         )
266:     } else {
267:         "obsidian://choose-vault".to_string()
268:     };
269:     let output = Command::new("/usr/bin/open")
270:         .arg("-a")
271:         .arg(app)
272:         .arg(uri)
273:         .output()
274:         .map_err(|e| e.to_string())?;
275:     if output.status.success() && !registered {
276:         Err(format!("OBSIDIAN_REGISTRATION_REQUIRED: In Obsidian, use Open folder as vault → Open and select {}. This is required once; then press Open in Obsidian again.", path.display()))
277:     } else if output.status.success() {
278:         Ok(())
279:     } else {
280:         Err(String::from_utf8_lossy(&output.stderr).into())
281:     }
282: }

284: fn obsidian_registered(registry: &serde_json::Value, target: &Path) -> bool {
285:     registry
286:         .get("vaults")
287:         .and_then(serde_json::Value::as_object)
288:         .is_some_and(|vaults| {
289:             vaults
290:                 .values()
291:                 .filter_map(|v| v.get("path").and_then(serde_json::Value::as_str))
292:                 .any(|p| Path::new(p).canonicalize().is_ok_and(|p| p == target))
293:         })
294: }
