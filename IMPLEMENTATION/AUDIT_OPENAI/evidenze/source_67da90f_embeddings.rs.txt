90: /// Read embeddings provider from 00_SYSTEM/SYNC_PROFILE.json.
91: /// If absent, defaults to "openai".
92: pub fn get_embeddings_provider(vault_path: &Path, active_port: u16) -> EmbeddingsProviderReport {
93:     let profile_path = vault_path.join("00_SYSTEM").join("SYNC_PROFILE.json");
94:     let mut configured_provider = None;
95: 
96:     if let Ok(content) = std::fs::read_to_string(&profile_path) {
97:         if let Ok(val) = serde_json::from_str::<Value>(&content) {
98:             if let Some(p) = val.get("embeddingsProvider").and_then(Value::as_str) {
99:                 let trimmed = p.trim().to_lowercase();
100:                 if trimmed == "local" || trimmed == "openai" {
101:                     configured_provider = Some(trimmed);
102:                 }
103:             }
104:         }
105:     }
106: 
107:     let provider = configured_provider.unwrap_or_else(|| "openai".to_string());
108:     let is_local = provider == "local";
109:     let model = if is_local {
110:         "bge-m3".to_string()
111:     } else {
112:         DEFAULT_EMBEDDINGS_MODEL.to_string()
113:     };
114:     let dimensions = if is_local { 1024 } else { 1536 };
115:     let endpoint = if is_local {
116:         if active_port > 0 {
117:             format!("http://127.0.0.1:{}/v1/embeddings", active_port)
118:         } else {
119:             String::new()
120:         }
121:     } else {
122:         DEFAULT_EMBEDDINGS_ENDPOINT.to_string()
123:     };
124: 
125:     let cache = load_embeddings_cache(vault_path).unwrap_or_default();
126:     let cache_entries = cache.entries.len();
127:     // Cache assente o vuota: nessuna dimensione memorizzata da confrontare. Senza questo azzeramento il
128:     // default della struct (1536) veniva mostrato in UI come "cache a 1536 dim" con 0 passaggi
129:     // (collaudo del 20/09/2026 su vault senza EMBEDDINGS_CACHE.json).
130:     let cache_dimensions = if cache_entries == 0 { 0 } else { cache.dimensions };
131: 
132:     let mut total_passages = 0;
133:     let mut matched_passages = 0;
134:     if let Ok(catalog) = load_catalog(vault_path) {
135:         for doc in catalog.documents.values() {
136:             for p in &doc.passages {
137:                 total_passages += 1;
138:                 if let Some(entry) = cache.entries.get(&p.passage_id) {
139:                     if entry.sha256 == p.sha256 && (cache_dimensions == 0 || cache_dimensions == dimensions) {
140:                         matched_passages += 1;
141:                     }
142:                 }
143:             }
144:         }
145:     }
146: 
147:     let needs_reindex = if total_passages > 0 {
148:         matched_passages < total_passages || (cache_entries > 0 && cache_dimensions != dimensions)
149:     } else {
150:         cache_entries > 0 && cache_dimensions != dimensions
151:     };
152: 
153:     EmbeddingsProviderReport {
154:         provider,
155:         model,
156:         dimensions,
157:         endpoint,
158:         cache_dimensions,
159:         cache_entries,
160:         total_passages,
161:         matched_passages,
162:         needs_reindex,
163:     }
164: }

166: /// Atomically write embeddingsProvider to 00_SYSTEM/SYNC_PROFILE.json.
167: /// Endpoint and model are NOT persisted: endpoint is built dynamically at runtime from active_port.
168: pub fn set_embeddings_provider(
169:     vault_path: &Path,
170:     provider: &str,
171:     active_port: u16,
172: ) -> Result<EmbeddingsProviderReport, String> {
173:     let prov = provider.trim().to_lowercase();
174:     if prov != "local" && prov != "openai" {
175:         return Err(format!(
176:             "Fornitore non valido '{}': ammessi esclusivamente 'openai' o 'local'",
177:             provider
178:         ));
179:     }
180: 
181:     let profile_path = vault_path.join("00_SYSTEM").join("SYNC_PROFILE.json");
182:     let mut val = if profile_path.exists() {
183:         let content = std::fs::read_to_string(&profile_path).map_err(|e| e.to_string())?;
184:         serde_json::from_str::<Value>(&content).unwrap_or_else(|_| json!({}))
185:     } else {
186:         json!({})
187:     };
188: 
189:     if !val.is_object() {
190:         val = json!({});
191:     }
192: 
193:     let obj = val.as_object_mut().unwrap();
194:     obj.insert("embeddingsProvider".into(), Value::String(prov.clone()));
195: 
196:     // Never persist endpoint or model: endpoint is runtime-constructed from live server port
197:     obj.remove("embeddingsEndpoint");
198:     obj.remove("embeddingsModel");
199: 
200:     let sys_dir = vault_path.join("00_SYSTEM");
201:     if !sys_dir.exists() {
202:         std::fs::create_dir_all(&sys_dir).map_err(|e| e.to_string())?;
203:     }
204: 
205:     let serialized = serde_json::to_string_pretty(&val).map_err(|e| e.to_string())?;
206:     std::fs::write(&profile_path, serialized.as_bytes()).map_err(|e| e.to_string())?;
207: 
208:     Ok(get_embeddings_provider(vault_path, active_port))
209: }

211: /// Security validation: if embeddingsProvider is "local", ensure the endpoint is strictly a loopback address.
212: pub fn validate_provider_endpoint(vault_path: &Path, endpoint: &str) -> Result<(), String> {
213:     if endpoint.trim().is_empty() {
214:         return Ok(());
215:     }
216:     let profile_path = vault_path.join("00_SYSTEM").join("SYNC_PROFILE.json");
217:     if let Ok(content) = std::fs::read_to_string(&profile_path) {
218:         if let Ok(val) = serde_json::from_str::<Value>(&content) {
219:             if let Some(p) = val.get("embeddingsProvider").and_then(Value::as_str) {
220:                 if p.trim().to_lowercase() == "local" {
221:                     let is_loopback = is_loopback_endpoint(endpoint)?;
222:                     if !is_loopback {
223:                         return Err(format!(
224:                             "Rifiutato endpoint esterno '{}' con provider 'local': il provider locale autorizza esclusivamente indirizzi loopback (127.0.0.1, localhost, ::1)",
225:                             endpoint
226:                         ));
227:                     }
228:                 }
229:             }
230:         }
231:     }
232:     Ok(())
233: }

247: /// Validates an embeddings endpoint URL for security.
248: /// Returns Ok(true) if the endpoint is loopback (allowing http://),
249: /// Ok(false) if the endpoint is a valid external HTTPS endpoint,
250: /// or Err(message) if an external endpoint attempts to use unencrypted HTTP or if the URL is invalid.
251: pub fn is_loopback_endpoint(endpoint: &str) -> Result<bool, String> {
252:     let url = reqwest::Url::parse(endpoint)
253:         .map_err(|e| format!("URL endpoint non valido '{}': {}", endpoint, e))?;
254: 
255:     let host = url.host_str().unwrap_or("").trim();
256:     let is_loopback = host == "127.0.0.1" || host == "localhost" || host == "::1" || host == "[::1]";
257: 
258:     match url.scheme() {
259:         "http" => {
260:             if is_loopback {
261:                 Ok(true)
262:             } else {
263:                 Err(format!(
264:                     "Rifiutato endpoint HTTP non sicuro '{}': http:// è consentito esclusivamente su loopback (127.0.0.1, localhost, ::1)",
265:                     endpoint
266:                 ))
267:             }
268:         }
269:         "https" => Ok(is_loopback),
270:         other => Err(format!("Schema URL non supportato '{}': supportati solo http e https", other)),
271:     }
272: }

333: /// Compute cosine similarity between two float vectors.
334: pub fn cosine_similarity(a: &[f32], b: &[f32]) -> f32 {
335:     if a.len() != b.len() || a.is_empty() {
336:         return 0.0;
337:     }
338:     let mut dot = 0.0f32;
339:     let mut norm_a = 0.0f32;
340:     let mut norm_b = 0.0f32;
341:     for (x, y) in a.iter().zip(b.iter()) {
342:         dot += x * y;
343:         norm_a += x * x;
344:         norm_b += y * y;
345:     }
346:     let denom = norm_a.sqrt() * norm_b.sqrt();
347:     if denom <= 0.0000001 {
348:         0.0
349:     } else {
350:         dot / denom
351:     }
352: }

354: /// Load embeddings cache from 00_SYSTEM/EMBEDDINGS_CACHE.json or default if absent.
355: pub fn load_embeddings_cache(vault_path: &Path) -> Result<Arc<EmbeddingsCache>, String> {
356:     let r = root(vault_path)?;
357:     let sys = match child(&r, "00_SYSTEM") {
358:         Ok(d) => d,
359:         Err(_) => return Ok(Arc::new(EmbeddingsCache::default())),
360:     };
361:     if !names(&sys)?.iter().any(|s| s == EMBEDDINGS_CACHE_FILE) {
362:         return Ok(Arc::new(EmbeddingsCache::default()));
363:     }
364: 
365:     let emb_file = vault_path.join("00_SYSTEM").join(EMBEDDINGS_CACHE_FILE);
366:     let current_rev = get_embeddings_cache_revision(vault_path);
367: 
368:     if let Ok(meta) = std::fs::metadata(&emb_file) {
369:         let size = meta.len();
370:         let mtime = meta.modified().unwrap_or(SystemTime::UNIX_EPOCH);
371:         if let Ok(guard) = EMBEDDINGS_IN_MEMORY_CACHE.lock() {
372:             if let Some(entry) = guard.get(vault_path) {
373:                 let is_high_prec = crate::ai::is_high_precision_fs(vault_path);
374:                 let is_mtime_match = if is_high_prec {
375:                     entry.mtime == mtime
376:                 } else {
377:                     entry.mtime == mtime
378:                         || entry.mtime.duration_since(mtime).map(|d| d.as_millis() < 2000).unwrap_or(false)
379:                         || mtime.duration_since(entry.mtime).map(|d| d.as_millis() < 2000).unwrap_or(false)
380:                 };
381:                 if entry.size == size && is_mtime_match && entry.revision == current_rev {
382:                     return Ok(entry.cache.clone());
383:                 }
384:             }
385:         }
386:     }
387: 
388:     increment_embeddings_disk_read_count(vault_path);
389:     let bytes = read(&sys, EMBEDDINGS_CACHE_FILE)?;
390:     if bytes.len() > 512 * 1024 * 1024 {
391:         return Err("Embeddings cache file exceeds 512 MB".into());
392:     }
393:     let cache: EmbeddingsCache = serde_json::from_slice(&bytes).map_err(err)?;
394:     let cache_arc = Arc::new(cache);
395: 
396:     if let Ok(meta) = std::fs::metadata(&emb_file) {
397:         let size = meta.len();
398:         let mtime = meta.modified().unwrap_or(SystemTime::UNIX_EPOCH);
399:         let current_rev = get_embeddings_cache_revision(vault_path);
400:         if let Ok(mut guard) = EMBEDDINGS_IN_MEMORY_CACHE.lock() {
401:             guard.insert(vault_path.to_path_buf(), EmbeddingsCacheEntry {
402:                 vault_path: vault_path.to_path_buf(),
403:                 size,
404:                 mtime,
405:                 revision: current_rev,
406:                 cache: cache_arc.clone(),
407:             });
408:         }
409:     }
410: 
411:     Ok(cache_arc)
412: }

414: /// Atomically save embeddings cache to 00_SYSTEM/EMBEDDINGS_CACHE.json
415: pub fn save_embeddings_cache(vault_path: &Path, cache: &mut EmbeddingsCache) -> Result<(), String> {
416:     cache.last_updated_at = now_iso();
417:     let r = root(vault_path)?;
418:     let sys = child(&r, "00_SYSTEM")?;
419:     let tmp = format!(".embeddings-{}.tmp", crate::ai::random_token()?);
420:     let bytes = serde_json::to_vec(cache).map_err(err)?;
421:     write_new(&sys, &tmp, &bytes)?;
422:     let res = sys.rename(&tmp, &sys, EMBEDDINGS_CACHE_FILE).map_err(err);
423:     if res.is_err() {
424:         let _ = sys.remove_file(&tmp);
425:     } else {
426:         bump_embeddings_cache_revision(vault_path);
427:     }
428:     res
429: }

431: /// Get current embeddings status and coverage
432: pub fn embeddings_status(vault_path: &Path) -> Result<EmbeddingsStatusReport, String> {
433:     let catalog = load_catalog(vault_path)?;
434:     let cache = load_embeddings_cache(vault_path)?;
435: 
436:     let mut total_passages: usize = 0;
437:     let mut cached_passages: usize = 0;
438: 
439:     for doc in catalog.documents.values() {
440:         for p in &doc.passages {
441:             total_passages += 1;
442:             if let Some(entry) = cache.entries.get(&p.passage_id) {
443:                 if entry.sha256 == p.sha256 {
444:                     cached_passages += 1;
445:                 }
446:             }
447:         }
448:     }
449: 
450:     let missing = total_passages.saturating_sub(cached_passages);
451:     let coverage = if total_passages == 0 {
452:         1.0
453:     } else {
454:         (cached_passages as f64) / (total_passages as f64)
455:     };
456: 
457:     Ok(EmbeddingsStatusReport {
458:         total_passages,
459:         cached_passages,
460:         missing_passages: missing,
461:         coverage,
462:         model: cache.model.clone(),
463:         dimensions: cache.dimensions,
464:         is_available: cached_passages > 0,
465:         last_updated_at: cache.last_updated_at.clone(),
466:     })
467: }

478: /// Fetch embeddings for a batch of strings from a configurable endpoint.
479: /// Allows http:// ONLY on loopback (127.0.0.1, localhost, ::1); rejects unencrypted HTTP for external hosts.
480: /// Does not require or send an API key if the endpoint is loopback and key is empty.
481: pub async fn fetch_openai_embeddings_with_endpoint(
482:     endpoint: &str,
483:     api_key: &str,
484:     model: &str,
485:     texts: &[String],
486: ) -> Result<Vec<Vec<f32>>, String> {
487:     if texts.is_empty() {
488:         return Ok(Vec::new());
489:     }
490: 
491:     let is_loopback = is_loopback_endpoint(endpoint)?;
492: 
493:     // Non-loopback endpoints strictly require an API key
494:     if !is_loopback && api_key.trim().is_empty() {
495:         return Err("OpenAI API key non configurata".into());
496:     }
497: 
498:     // Loopback: il server locale calcola i vettori sul Mac (misurato il 20/09/2026 su M2: 0,65 s/passaggio
499:     // senza GPU, 0,09 s con Metal). Con 60 s un lotto veniva annullato a meta' calcolo: nel log di
500:     // llama-server "cancel task" arrivava esattamente 60 s dopo la richiesta e l'utente vedeva
501:     // "error sending request for url". 300 s coprono un lotto di 16 passaggi con ampio margine anche su
502:     // un Mac lento; l'API remota resta a 60 s.
503:     let request_timeout = if is_loopback { Duration::from_secs(300) } else { Duration::from_secs(60) };
504:     let mut client_builder = reqwest::Client::builder()
505:         .connect_timeout(Duration::from_secs(10))
506:         .timeout(request_timeout);
507: 
508:     if !is_loopback {
509:         client_builder = client_builder.https_only(true);
510:     }
511: 
512:     let client = client_builder
513:         .build()
514:         .map_err(|e| format!("HTTP client error: {}", e))?;
515: 
516:     let body = json!({
517:         "model": model,
518:         "input": texts
519:     });
520: 
521:     let mut req = client.post(endpoint).json(&body);
522:     if !api_key.trim().is_empty() {
523:         req = req.bearer_auth(api_key);
524:     }
525: 
526:     let resp = req
527:         .send()
528:         .await
529:         .map_err(|e| format!("Richiesta embeddings fallita verso '{}': {}", endpoint, e))?;
530: 
531:     if !resp.status().is_success() {
532:         let status = resp.status().as_u16();
533:         let err_text = resp.text().await.unwrap_or_default();
534:         return Err(match status {
535:             401 => "Chiave API non valida o non autorizzata".into(),
536:             429 => "Quota esaurita o rate limit raggiunto".into(),
537:             code => format!("Embeddings error HTTP {}: {}", code, err_text),
538:         });
539:     }
540: 
541:     let val: Value = resp
542:         .json()
543:         .await
544:         .map_err(|e| format!("Risposta JSON embeddings non valida: {}", e))?;
545: 
546:     let data = val
547:         .get("data")
548:         .and_then(Value::as_array)
549:         .ok_or("Campo data mancante nella risposta embeddings")?;
550: 
551:     let mut embeddings = Vec::with_capacity(texts.len());
552:     for item in data {
553:         let arr = item
554:             .get("embedding")
555:             .and_then(Value::as_array)
556:             .ok_or("Vettore embedding non valido")?;
557:         let vec: Vec<f32> = arr
558:             .iter()
559:             .filter_map(|v| v.as_f64().map(|f| f as f32))
560:             .collect();
561:         embeddings.push(vec);
562:     }
563: 
564:     if embeddings.len() != texts.len() {
565:         return Err(format!(
566:             "Numero vettori ricevuti ({}) diverso dai testi inviati ({})",
567:             embeddings.len(),
568:             texts.len()
569:         ));
570:     }
571: 
572:     Ok(embeddings)
573: }

575: /// Build contextualized passage text for embedding:
576: /// - Note title (frontmatter title if available, else file_name without extension)
577: /// - Document category (folder name e.g. "01_CLIENTS" ... "10_APPROVED_OUTPUTS" or category)
578: /// - Section locator (p.locator e.g. "Pagina N", "Slide N", "Paragrafi N-M")
579: /// - Raw passage text
580: /// Prefix context is capped at 20% of the character length of the passage.
581: pub fn build_passage_embedding_text(
582:     title: &str,
583:     category: &str,
584:     locator: &str,
585:     passage_text: &str,
586: ) -> String {
587:     let mut ctx_lines = Vec::new();
588:     let trimmed_title = title.trim();
589:     if !trimmed_title.is_empty() {
590:         ctx_lines.push(trimmed_title);
591:     }
592:     let trimmed_cat = category.trim();
593:     if !trimmed_cat.is_empty() {
594:         ctx_lines.push(trimmed_cat);
595:     }
596:     let trimmed_loc = locator.trim();
597:     if !trimmed_loc.is_empty() {
598:         ctx_lines.push(trimmed_loc);
599:     }
600: 
601:     if ctx_lines.is_empty() || passage_text.trim().is_empty() {
602:         return passage_text.to_string();
603:     }
604: 
605:     let raw_ctx = ctx_lines.join("\n");
606:     let max_len = (passage_text.chars().count() as f64 * 0.20).floor() as usize;
607:     let final_ctx: String = if raw_ctx.chars().count() > max_len {
608:         raw_ctx.chars().take(max_len).collect()
609:     } else {
610:         raw_ctx
611:     };
612: 
613:     if final_ctx.is_empty() {
614:         passage_text.to_string()
615:     } else {
616:         format!("{}\n{}", final_ctx, passage_text)
617:     }
618: }

639: /// Come `sync_embeddings_with_port`, con callback `(elaborati, totale)` invocata all'avvio e dopo ogni
640: /// lotto: e' l'unica sorgente dell'evento `embeddings_sync_progress` che alimenta la barra in UI
641: /// (prima di questa callback l'evento non veniva mai emesso e la barra restava a 0,0%).
642: pub async fn sync_embeddings_with_progress(
643:     vault_path: &Path,
644:     api_key: &str,
645:     model: Option<&str>,
646:     active_port: Option<u16>,
647:     on_progress: Option<&(dyn Fn(usize, usize) + Sync)>,
648: ) -> Result<EmbeddingsStatusReport, String> {
649:     SYNC_CANCELLED.store(false, Ordering::SeqCst);
650: 
651:     let catalog = load_catalog(vault_path)?;
652:     let current_cache = load_embeddings_cache(vault_path)?;
653:     let prov_rep = get_embeddings_provider(vault_path, active_port.unwrap_or(0));
654:     let endpoint = prov_rep.endpoint.clone();
655:     let is_local = prov_rep.provider == "local";
656: 
657:     if is_local {
658:         if endpoint.is_empty() || active_port.unwrap_or(0) == 0 {
659:             return Err("Server di embedding locale non attivo. Avvia il server locale da Impostazioni prima di sincronizzare.".into());
660:         }
661:         let port = active_port.unwrap();
662:         if !crate::llama::check_health(port) {
663:             return Err(format!(
664:                 "Server di embedding locale non raggiungibile su {}. Avvia il server locale da Impostazioni prima di sincronizzare.",
665:                 endpoint
666:             ));
667:         }
668:     }
669: 
670:     validate_provider_endpoint(vault_path, &endpoint)?;
671:     let is_loopback = is_loopback_endpoint(&endpoint)?;
672:     let target_model = model.unwrap_or(&prov_rep.model);
673:     let target_dimensions = prov_rep.dimensions;
674: 
675:     let key_from_keychain = if !is_loopback && api_key.trim().is_empty() {
676:         crate::keychain::load().ok().flatten().filter(|k| !k.trim().is_empty())
677:     } else {
678:         None
679:     };
680:     let effective_key = if is_loopback {
681:         ""
682:     } else if !api_key.trim().is_empty() {
683:         api_key
684:     } else if let Some(ref k) = key_from_keychain {
685:         k.as_str()
686:     } else {
687:         return Err("Chiave API mancante. Configurala in Impostazioni per sincronizzare gli embeddings.".into());
688:     };
689: 
690:     // Staging cache for safe migration:
691:     // If target_model != current_cache.model or target_dimensions != current_cache.dimensions,
692:     // we use a staging cache file (00_SYSTEM/EMBEDDINGS_CACHE.staging.json) so the existing cache
693:     // is preserved until 100% complete, and if cancelled or interrupted, can resume!
694:     let is_model_migration = current_cache.model != target_model || (current_cache.dimensions > 0 && current_cache.dimensions != target_dimensions);
695:     let staging_path = vault_path.join("00_SYSTEM").join("EMBEDDINGS_CACHE.staging.json");
696: 
697:     let mut working_cache = if is_model_migration {
698:         if staging_path.exists() {
699:             if let Ok(st_content) = std::fs::read_to_string(&staging_path) {
700:                 if let Ok(st_cache) = serde_json::from_str::<EmbeddingsCache>(&st_content) {
701:                     if st_cache.model == target_model && st_cache.dimensions == target_dimensions {
702:                         st_cache
703:                     } else {
704:                         EmbeddingsCache {
705:                             version: 1,
706:                             model: target_model.to_string(),
707:                             dimensions: target_dimensions,
708:                             entries: BTreeMap::new(),
709:                             last_updated_at: now_iso(),
710:                         }
711:                     }
712:                 } else {
713:                     EmbeddingsCache {
714:                         version: 1,
715:                         model: target_model.to_string(),
716:                         dimensions: target_dimensions,
717:                         entries: BTreeMap::new(),
718:                         last_updated_at: now_iso(),
719:                     }
720:                 }
721:             } else {
722:                 EmbeddingsCache {
723:                     version: 1,
724:                     model: target_model.to_string(),
725:                     dimensions: target_dimensions,
726:                     entries: BTreeMap::new(),
727:                     last_updated_at: now_iso(),
728:                 }
729:             }
730:         } else {
731:             EmbeddingsCache {
732:                 version: 1,
733:                 model: target_model.to_string(),
734:                 dimensions: target_dimensions,
735:                 entries: BTreeMap::new(),
736:                 last_updated_at: now_iso(),
737:             }
738:         }
739:     } else {
740:         (*current_cache).clone()
741:     };
742: 
743:     // Collect missing or updated passages
744:     struct MissingPassage {
745:         passage_id: String,
746:         document_id: String,
747:         relative_path: String,
748:         locator: String,
749:         sha256: String,
750:         embedded_text_sha256: String,
751:         text_to_embed: String,
752:     }
753: 
754:     let mut missing: Vec<MissingPassage> = Vec::new();
755:     let mut valid_passage_ids = BTreeSet::new();
756: 
757:     for doc in catalog.documents.values() {
758:         let doc_title = if let Ok(content) = std::fs::read_to_string(vault_path.join(&doc.original_path)) {
759:             crate::vault::frontmatter(&content)
760:                 .ok()
761:                 .flatten()
762:                 .and_then(|fm| fm["title"].as_str().map(|s| s.trim().to_string()))
763:                 .filter(|s| !s.is_empty())
764:                 .unwrap_or_else(|| {
765:                     Path::new(&doc.file_name)
766:                         .file_stem()
767:                         .and_then(|s| s.to_str())
768:                         .unwrap_or(&doc.file_name)
769:                         .to_string()
770:                 })
771:         } else {
772:             Path::new(&doc.file_name)
773:                 .file_stem()
774:                 .and_then(|s| s.to_str())
775:                 .unwrap_or(&doc.file_name)
776:                 .to_string()
777:         };
778: 
779:         let doc_category = if let Some(first_segment) = doc.original_path.split('/').next() {
780:             if first_segment.starts_with("0") || first_segment.starts_with("1") {
781:                 first_segment.to_string()
782:             } else {
783:                 doc.category.clone().unwrap_or_else(|| first_segment.to_string())
784:             }
785:         } else {
786:             doc.category.clone().unwrap_or_default()
787:         };
788: 
789:         for p in &doc.passages {
790:             valid_passage_ids.insert(p.passage_id.clone());
791: 
792:             let text_to_embed = build_passage_embedding_text(&doc_title, &doc_category, &p.locator, &p.text);
793:             let embedded_sha = compute_sha256(text_to_embed.as_bytes());
794: 
795:             let needs_update = match working_cache.entries.get(&p.passage_id) {
796:                 Some(entry) => {
797:                     entry.sha256 != p.sha256
798:                         || entry.model != target_model
799:                         || entry.dimensions != target_dimensions
800:                         || entry.embedded_text_sha256.as_deref() != Some(&embedded_sha)
801:                 }
802:                 None => true,
803:             };
804: 
805:             if needs_update && !p.text.trim().is_empty() {
806:                 missing.push(MissingPassage {
807:                     passage_id: p.passage_id.clone(),
808:                     document_id: doc.document_id.clone(),
809:                     relative_path: doc.original_path.clone(),
810:                     locator: p.locator.clone(),
811:                     sha256: p.sha256.clone(),
812:                     embedded_text_sha256: embedded_sha,
813:                     text_to_embed,
814:                 });
815:             }
816:         }
817:     }
818: 
819:     // Prune stale passages from working cache
820:     working_cache.entries.retain(|id, _| valid_passage_ids.contains(id));
821: 
822:     // Lotti: 16 in locale (avanzamento a passi piccoli, annullamento entro un lotto), 32 verso l'API remota.
823:     // LIMEN_SYNC_BATCH e LIMEN_SYNC_TIMING servono solo alle misure sul banco di prova (21/09/2026):
824:     // senza variabili il comportamento e' quello predefinito.
825:     let default_batch = if is_loopback { 16 } else { 32 };
826:     let batch_size = std::env::var("LIMEN_SYNC_BATCH")
827:         .ok()
828:         .and_then(|v| v.parse::<usize>().ok())
829:         .filter(|n| *n > 0)
830:         .unwrap_or(default_batch);
831:     let timing = std::env::var("LIMEN_SYNC_TIMING").is_ok();
832:     let sync_started = std::time::Instant::now();
833:     let (mut t_req_total, mut t_write_total, mut writes) = (0u128, 0u128, 0usize);
834:     let total_missing = missing.len();
835:     let mut processed_count = 0;
836:     if total_missing > 0 {
837:         println!("Starting embedding sync: {} missing passages to process...", total_missing);
838:     }
839:     if let Some(cb) = on_progress {
840:         cb(0, total_missing);
841:     }
842: 
843:     for chunk in missing.chunks(batch_size) {
844:         if SYNC_CANCELLED.load(Ordering::SeqCst) {
845:             if is_model_migration {
846:                 let _ = std::fs::write(&staging_path, serde_json::to_string(&working_cache).unwrap_or_default());
847:             }
848:             return Err("Sincronizzazione semantica annullata dall'utente".to_string());
849:         }
850: 
851:         let texts: Vec<String> = chunk.iter().map(|m| m.text_to_embed.clone()).collect();
852:         let chars: usize = texts.iter().map(|t| t.chars().count()).sum();
853:         let t0 = std::time::Instant::now();
854:         let vectors = fetch_openai_embeddings_with_endpoint(&endpoint, effective_key, target_model, &texts).await?;
855:         let req_ms = t0.elapsed().as_millis();
856:         t_req_total += req_ms;
857: 
858:         for (m, vec) in chunk.iter().zip(vectors.into_iter()) {
859:             working_cache.entries.insert(
860:                 m.passage_id.clone(),
861:                 PassageEmbeddingEntry {
862:                     passage_id: m.passage_id.clone(),
863:                     document_id: m.document_id.clone(),
864:                     relative_path: m.relative_path.clone(),
865:                     locator: m.locator.clone(),
866:                     sha256: m.sha256.clone(),
867:                     embedded_text_sha256: Some(m.embedded_text_sha256.clone()),
868:                     model: target_model.to_string(),
869:                     dimensions: vec.len(),
870:                     vector: vec,
871:                     updated_at: now_iso(),
872:                 },
873:             );
874:         }
875: 
876:         processed_count += chunk.len();
877:         if let Some(cb) = on_progress {
878:             cb(processed_count, total_missing);
879:         }
880:         let mut write_ms = 0u128;
881:         if is_model_migration && (processed_count % 320 == 0 || processed_count == total_missing) {
882:             let t1 = std::time::Instant::now();
883:             let _ = std::fs::write(&staging_path, serde_json::to_string(&working_cache).unwrap_or_default());
884:             write_ms = t1.elapsed().as_millis();
885:             t_write_total += write_ms;
886:             writes += 1;
887:         }
888:         if timing {
889:             eprintln!(
890:                 "[sync-timing] lotto n={} caratteri={} richiesta_ms={} scrittura_ms={} elaborati={}/{} trascorso_s={:.1}",
891:                 texts.len(), chars, req_ms, write_ms, processed_count, total_missing, sync_started.elapsed().as_secs_f64()
892:             );
893:         }
894: 
895:         if processed_count % 320 == 0 || processed_count == total_missing {
896:             println!(
897:                 "Embedding progress: {}/{} passages ({:.1}%)",
898:                 processed_count,
899:                 total_missing,
900:                 (processed_count as f64 / total_missing as f64) * 100.0
901:             );
902:         }
903:     }
904: 
905:     if timing {
906:         eprintln!(
907:             "[sync-timing] TOTALE passaggi={} lotto={} totale_s={:.1} richieste_s={:.1} scritture_staging={} scritture_s={:.1}",
908:             total_missing, batch_size, sync_started.elapsed().as_secs_f64(), t_req_total as f64 / 1000.0, writes, t_write_total as f64 / 1000.0
909:         );
910:     }
911:     if let Some(first_entry) = working_cache.entries.values().next() {
912:         working_cache.dimensions = first_entry.dimensions;
913:     } else {
914:         working_cache.dimensions = target_dimensions;
915:     }
916:     working_cache.model = target_model.to_string();
917: 
918:     if is_model_migration {
919:         // Migration complete: atomically replace EMBEDDINGS_CACHE.json
920:         save_embeddings_cache(vault_path, &mut working_cache)?;
921:         let _ = std::fs::remove_file(&staging_path);
922:     } else {
923:         save_embeddings_cache(vault_path, &mut working_cache)?;
924:     }
925: 
926:     embeddings_status(vault_path)
927: }

929: /// Semantic score for a document given a query embedding vector.
930: /// Returns (best_score, matching_passage_id, matching_locator, passage_snippet).
931: pub fn rank_document_semantic(
932:     _doc_id: &str,
933:     passages: &[DocumentPassage],
934:     query_vector: &[f32],
935:     cache: &EmbeddingsCache,
936: ) -> (f32, Option<String>, Option<String>, Option<String>) {
937:     let mut best_score = 0.0f32;
938:     let mut best_passage_id = None;
939:     let mut best_locator = None;
940:     let mut best_snippet = None;
941: 
942:     for p in passages {
943:         if let Some(entry) = cache.entries.get(&p.passage_id) {
944:             if entry.sha256 == p.sha256 {
945:                 let sim = cosine_similarity(query_vector, &entry.vector);
946:                 if sim > best_score {
947:                     best_score = sim;
948:                     best_passage_id = Some(p.passage_id.clone());
949:                     best_locator = Some(p.locator.clone());
950:                     best_snippet = Some(p.text.chars().take(240).collect::<String>());
951:                 }
952:             }
953:         }
954:     }
955: 
956:     (best_score, best_passage_id, best_locator, best_snippet)
957: }

1043:         Some(t) if !t.trim().is_empty() => t.trim(),
1044:         _ => {

1046:             timings.t_index_cache_ms = sub_timings.t_index_cache_ms;

1050:             timings.t_fuse_ms = sub_timings.t_fuse_ms;
1051:             return Ok((items, false, timings, None));
1052:         }
1053:     };

1056:     if is_local && (endpoint.is_empty() || active_port.unwrap_or(0) == 0) {
1057:         let (items, sub_timings) = hybrid_search_vault_with_vector_filtered_timed(vault_path, q, None, false, filter_fn.as_ref()).await?;

1062:         timings.t_fuse_ms = sub_timings.t_fuse_ms;
1063:         return Ok((items, true, timings, None));
1064:     }
1065: 
1066:     // Mandatory security validation: local provider strictly forbids non-loopback endpoints
1067:     validate_provider_endpoint(vault_path, &endpoint)?;
1068: 
1069:     let cache = match load_embeddings_cache(vault_path) {
1070:         Ok(c) if !c.entries.is_empty() => c,
1071:         _ => {
1072:             let (items, sub_timings) = hybrid_search_vault_with_vector_filtered_timed(vault_path, q, None, false, filter_fn.as_ref()).await?;

1383:         && term.chars().all(|c| c.is_ascii_alphanumeric() || c == '-' || c == '_')
1384:         && term.len() >= 3;

1390:             id[..20].to_string()

1429:             it

1463:         }

1473:     }

1476:     let min_lex = intermediate.iter().map(|c| c.base_score).fold(f64::INFINITY, f64::min);
1477:     let max_lex = intermediate.iter().map(|c| c.base_score).fold(f64::NEG_INFINITY, f64::max);
1478:     let min_sem = intermediate.iter().map(|c| c.sem_sim).fold(f64::INFINITY, f64::min);
1479:     let max_sem = intermediate.iter().map(|c| c.sem_sim).fold(f64::NEG_INFINITY, f64::max);
1480: 
1481:     struct FusedCandidate {
1482:         item: SearchResultItem,
1483:         fused_score: f64,
1484:     }
1485: 
1486:     let mut fused_candidates: Vec<FusedCandidate> = Vec::new();

1490:             (c.base_score - min_lex) / (max_lex - min_lex)
1491:         } else if c.base_score > 0.0 {
1492:             1.0
1493:         } else {
1494:             0.0
1495:         };
1496: 
1497:         let sem_norm = if max_sem > min_sem {
1498:             (c.sem_sim - min_sem) / (max_sem - min_sem)
1499:         } else if c.sem_sim > 0.0 {
1500:             1.0
1501:         } else {
1502:             0.0
1503:         };
1504: 
1505:         let exact_bonus = if c.matches_term {
1506:             if is_exact_code { 0.20 } else { 0.05 }
1507:         } else {
1508:             0.0
1509:         };
1510: 
1511:         // C10 formula (F2, k = 0.20): max(lex_norm, sem_norm) + 0.20 * min(lex_norm, sem_norm) + exact_bonus
1512:         // Ensures documents with a single strong signal (e.g. semantic-only with lex_norm = 0)
1513:         // preserve their score and rank above mediocre dual-signal documents, while continuing
1514:         // to reward multi-modal concurrence. Selected via dev tuning on 30 queries (Recall@10 = 0.933).
1515:         let fused_score = lex_norm.max(sem_norm) + 0.20 * lex_norm.min(sem_norm) + exact_bonus;
1516:         c.item.score = fused_score;
1517:         c.item.semantic_similarity = if c.sem_sim > 0.0 { Some(c.sem_sim) } else { None };
1518:         fused_candidates.push(FusedCandidate { item: c.item, fused_score });
1519:     }
1520: 
1521:     // Sort descending by fused score
1522:     fused_candidates.sort_by(|a, b| b.fused_score.partial_cmp(&a.fused_score).unwrap_or(std::cmp::Ordering::Equal));
1523: 
1524:     let limit = q.limit.unwrap_or(50);
1525:     let offset = q.offset.unwrap_or(0);
1526:     let results: Vec<SearchResultItem> = fused_candidates
1527:         .into_iter()
1528:         .skip(offset)
1529:         .take(limit)
1530:         .map(|c| c.item)
1531:         .collect();
1532:     let t_fuse_ms = t_fuse_start.elapsed().as_millis() as u64;
1533:     let t_search_ms = t_search_start.elapsed().as_millis() as u64;
1534: 
1535:     let timings = SearchPhaseTimings {
1536:         t_index_cache_ms,
1537:         t_search_ms,
1538:         t_embed_ms: 0,
1539:         t_words_ms,
1540:         t_sem_ms,
1541:         t_fuse_ms,
1542:     };
1543: 
1544:     Ok((results, timings))
1545: }
