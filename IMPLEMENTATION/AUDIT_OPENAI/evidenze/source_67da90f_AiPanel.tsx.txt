40: // Clean titles: remove technical prefixes (e.g. doc_..._) and .md extension
41: function cleanTitle(title: string, relativePath: string): string {
42:   if (title && !title.startsWith('doc_') && !title.endsWith('.md')) {
43:     return title;
44:   }
45:   const base = relativePath.split('/').pop() || relativePath;
46:   return base.replace(/\.md$/i, '').replace(/^[a-f0-9_-]{8,16}_/i, '');
47: }
