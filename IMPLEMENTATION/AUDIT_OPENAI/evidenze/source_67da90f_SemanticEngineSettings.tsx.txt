34: function formatBytes(bytes: number): string {
35:   if (bytes === 0) return '0 B';
36:   const k = 1024;
37:   const sizes = ['B', 'KB', 'MB', 'GB'];
38:   const i = Math.floor(Math.log(bytes) / Math.log(k));
39:   return `${(bytes / Math.pow(k, i)).toFixed(1)} ${sizes[i]}`;
40: }

704:                   <div style={{ display: 'flex', justifyContent: 'space-between', fontSize: 12, marginBottom: 4 }}>

716:                       overflow: 'hidden',
717:                       marginBottom: 10,
718:                     }}

752:         </div>

758:           border: '1px solid #cbd5e1',
759:           borderRadius: 8,
760:           padding: 18,
761:           backgroundColor: '#ffffff',
762:           marginBottom: 16,
763:         }}
764:       >
765:         <div style={{ fontSize: 14, fontWeight: 700, color: '#0f172a', marginBottom: 6 }}>

795:             value={selectedOpenaiModel}
