109: pub fn log_local_model_timing_meta(
110:     event: &str,
111:     elapsed_ms: u64,
112:     service_binary: Option<&Path>,
113:     service_pid: Option<u32>,
114:     service_port: Option<u16>,
115:     details: &str,
116: ) {

118:     if let Some(parent) = log_path.parent() {
119:         let _ = fs::create_dir_all(parent);
120:     }

122: 
123:     let app_exe = std::env::current_exe()
124:         .map(|p| p.to_string_lossy().to_string())
125:         .unwrap_or_else(|_| "sconosciuto".to_string());
126:     let app_pid = std::process::id();
127: 
128:     let s_bin = service_binary
129:         .map(|p| p.to_string_lossy().to_string())
130:         .or_else(|| service_pid.and_then(get_process_exe_path).map(|p| p.to_string_lossy().to_string()))

132:             if service_pid.is_none() && service_port.is_none() {

134:             } else {
135:                 None
136:             }
137:         })
138:         .unwrap_or_else(|| "non_rilevato".to_string());
139:     let s_pid = service_pid.map(|p| p.to_string()).unwrap_or_else(|| "none".to_string());
140:     let s_port = service_port.map(|p| p.to_string()).unwrap_or_else(|| "none".to_string());
141: 
142:     let line = format!(
143:         "{} | event={} | elapsed_ms={} | app_exe={} | app_pid={} | service_exe={} | service_pid={} | service_port={} | details={}\n",

146:     if let Ok(mut f) = fs::OpenOptions::new().create(true).append(true).open(&log_path) {

178:     if let Ok(entries) = fs::read_dir(logs_dir) {
179:         let mut files: Vec<PathBuf> = entries
180:             .filter_map(|e| e.ok())
181:             .map(|e| e.path())
182:             .filter(|p| {
183:                 p.is_file()
184:                     && p.file_name()
185:                         .and_then(|n| n.to_str())
186:                         .map(|n| n.starts_with("llama-server_") && n.ends_with(".log"))
187:                         .unwrap_or(false)
188:             })
189:             .collect();
190: 
191:         files.sort();
192: 
193:         if files.len() > max_files {
194:             let to_remove = files.len() - max_files;
195:             for p in files.iter().take(to_remove) {
196:                 let _ = fs::remove_file(p);
197:             }
198:         }
199:     }

254: pub fn compute_file_sha256(path: &Path) -> Result<String, String> {
255:     MODEL_SHA256_COMPUTE_COUNT.fetch_add(1, Ordering::Relaxed);
256:     let mut file = fs::File::open(path).map_err(|e| e.to_string())?;
257:     let mut hasher = Sha256::new();
258:     let mut buffer = [0u8; 65536];
259:     loop {
260:         let count = file.read(&mut buffer).map_err(|e| e.to_string())?;
261:         if count == 0 {
262:             break;
263:         }
264:         hasher.update(&buffer[..count]);
265:     }
266:     Ok(format!("{:x}", hasher.finalize()))
267: }

269: /// Parametric verification function for production and unit testing.
270: pub fn model_status_for_paths(
271:     target: &Path,
272:     cache_file: &Path,
273:     expected_size: u64,
274:     expected_sha256: &str,
275:     force_recheck: bool,
276: ) -> LocalModelReport {
277:     // If target does not exist, check common local huggingface snapshot location
278:     if !target.exists() {
279:         if let Ok(home) = std::env::var("HOME") {
280:             let hf_candidate = Path::new(&home)
281:                 .join(".cache/huggingface/hub/models--gpustack--bge-m3-GGUF/snapshots/2d48f1737679ad900d5c26c5aad5410e9c70fdca/bge-m3-Q8_0.gguf");
282:             if hf_candidate.exists() {
283:                 if let Ok(meta) = fs::metadata(&hf_candidate) {
284:                     if meta.len() == expected_size {
285:                         #[cfg(unix)]
286:                         let _ = std::os::unix::fs::symlink(&hf_candidate, target);
287:                         #[cfg(windows)]
288:                         let _ = std::os::windows::fs::symlink_file(&hf_candidate, target);
289:                     }
290:                 }
291:             }
292:         }
293:     }
294: 
295:     if !target.exists() {
296:         return LocalModelReport {
297:             installed: false,
298:             path: target.to_string_lossy().to_string(),
299:             bytes: 0,
300:             sha256_ok: false,
301:         };
302:     }
303: 
304:     let meta = match fs::metadata(target) {
305:         Ok(m) => m,
306:         Err(_) => {
307:             return LocalModelReport {
308:                 installed: false,
309:                 path: target.to_string_lossy().to_string(),
310:                 bytes: 0,
311:                 sha256_ok: false,
312:             };
313:         }
314:     };
315:     let bytes = meta.len();
316:     let mtime_ms = meta
317:         .modified()
318:         .ok()
319:         .and_then(|t| t.duration_since(std::time::UNIX_EPOCH).ok())
320:         .map(|d| d.as_millis() as u64)
321:         .unwrap_or(0);
322:     let target_str = target.to_string_lossy().to_string();
323: 
324:     // Se la dimensione su disco non coincide con quella attesa per il modello, non è il modello atteso.
325:     if bytes != expected_size {
326:         eprintln!("[llama] File size mismatch on {}: {} vs expected {}", target.display(), bytes, expected_size);
327:         let _ = fs::remove_file(cache_file);
328:         return LocalModelReport {
329:             installed: false,
330:             path: target_str,
331:             bytes,
332:             sha256_ok: false,
333:         };
334:     }
335: 
336:     // 1. Fast path: verifica tramite cache dei metadati (se non è richiesto force_recheck)
337:     if !force_recheck {
338:         if let Some(cache) = read_model_cache(cache_file) {
339:             let is_ntfs = crate::ai::is_high_precision_fs(target);
340:             let mtime_matches = if is_ntfs {
341:                 cache.mtime_ms == mtime_ms
342:             } else {
343:                 cache.mtime_ms.abs_diff(mtime_ms) <= 2000
344:             };
345: 
346:             let path_matches = Path::new(&cache.path) == target || cache.path == target_str;
347:             let sha256_matches = cache.sha256.eq_ignore_ascii_case(expected_sha256);
348:             let size_matches = cache.bytes == expected_size && bytes == expected_size;
349: 
350:             if size_matches && path_matches && sha256_matches && mtime_matches && cache.sha256_ok {
351:                 return LocalModelReport {
352:                     installed: true,
353:                     path: target_str,
354:                     bytes,
355:                     sha256_ok: true,
356:                 };
357:             }
358:         }
359:     }
360: 
361:     // 2. Slow path: verifica completa SHA-256 su disco
362:     let sha256_ok = match compute_file_sha256(target) {
363:         Ok(hash) => {
364:             let ok = hash.eq_ignore_ascii_case(expected_sha256);
365:             if ok {
366:                 let _ = save_model_cache(cache_file, &ModelVerificationCache {
367:                     path: target_str.clone(),
368:                     bytes,
369:                     mtime_ms,
370:                     sha256: hash,
371:                     sha256_ok: true,
372:                     verified_at: chrono::Utc::now().to_rfc3339(),
373:                 });
374:             } else {
375:                 eprintln!("[llama] Hash mismatch on {}: {} vs expected {}", target.display(), hash, expected_sha256);
376:                 let _ = fs::remove_file(cache_file);
377:             }
378:             ok
379:         }
380:         Err(_) => {
381:             let _ = fs::remove_file(cache_file);
382:             false
383:         }
384:     };
385: 
386:     LocalModelReport {
387:         installed: sha256_ok,
388:         path: target_str,
389:         bytes,
390:         sha256_ok,
391:     }
392: }

394: /// Install model from source file with SINGLE I/O pass (streaming copy + simultaneous hash computation).
395: pub fn install_model_for_paths(
396:     source_path: &Path,
397:     target: &Path,
398:     cache_file: &Path,
399:     expected_size: u64,
400:     expected_sha256: &str,
401: ) -> Result<LocalModelReport, String> {
402:     let t0 = Instant::now();
403:     if !source_path.is_file() {
404:         return Err(format!("File non trovato: {:?}", source_path));
405:     }
406:     let meta = fs::metadata(source_path).map_err(|e| e.to_string())?;
407:     let bytes = meta.len();
408:     if bytes != expected_size {
409:         return Err(format!(
410:             "Dimensione file non corretta: {} byte (richiesti esattamente {} byte)",
411:             bytes,
412:             expected_size
413:         ));
414:     }
415: 
416:     // Copia e calcolo hash in UN'UNICA passata di I/O (singola lettura del file sorgente)
417:     let temp_target = target.with_extension("installing");
418:     let _ = fs::remove_file(&temp_target);
419: 
420:     let mut source_file = fs::File::open(source_path).map_err(|e| e.to_string())?;
421:     let mut dest_file = fs::File::create(&temp_target).map_err(|e| e.to_string())?;
422:     let mut hasher = Sha256::new();
423:     let mut buffer = vec![0u8; 1024 * 1024];
424: 
425:     loop {
426:         let count = source_file.read(&mut buffer).map_err(|e| e.to_string())?;
427:         if count == 0 {
428:             break;
429:         }
430:         dest_file.write_all(&buffer[..count]).map_err(|e| e.to_string())?;
431:         hasher.update(&buffer[..count]);
432:     }
433:     dest_file.flush().map_err(|e| e.to_string())?;
434:     drop(dest_file);
435:     drop(source_file);
436: 
437:     MODEL_SHA256_COMPUTE_COUNT.fetch_add(1, Ordering::Relaxed);
438: 
439:     let hash = format!("{:x}", hasher.finalize());
440:     if hash.to_lowercase() != expected_sha256.to_lowercase() {
441:         let _ = fs::remove_file(&temp_target);
442:         return Err(format!(
443:             "Verifica SHA-256 fallita: calcolato {} vs atteso {}",
444:             hash, expected_sha256
445:         ));
446:     }
447: 
448:     if target.exists() {
449:         let _ = fs::remove_file(target);
450:     }
451:     fs::rename(&temp_target, target).or_else(|_| {
452:         fs::copy(&temp_target, target).map(|_| ())?;
453:         let _ = fs::remove_file(&temp_target);
454:         Ok::<(), std::io::Error>(())
455:     }).map_err(|e| e.to_string())?;
456: 
457:     let dest_meta = fs::metadata(target).map_err(|e| e.to_string())?;
458:     let dest_mtime_ms = dest_meta
459:         .modified()
460:         .ok()
461:         .and_then(|t| t.duration_since(std::time::UNIX_EPOCH).ok())
462:         .map(|d| d.as_millis() as u64)
463:         .unwrap_or(0);
464:     let target_str = target.to_string_lossy().to_string();
465: 
466:     save_model_cache(cache_file, &ModelVerificationCache {
467:         path: target_str.clone(),
468:         bytes,
469:         mtime_ms: dest_mtime_ms,
470:         sha256: hash,
471:         sha256_ok: true,
472:         verified_at: chrono::Utc::now().to_rfc3339(),
473:     })?;
474: 
475:     let elapsed = t0.elapsed().as_millis() as u64;
476:     log_local_model_timing("FILE_SELECTION", elapsed, &format!("path={}, bytes={}", target_str, bytes));
477: 
478:     Ok(LocalModelReport {
479:         installed: true,
480:         path: target_str,
481:         bytes,
482:         sha256_ok: true,
483:     })
484: }

515: pub async fn download_model_with_progress<F>(on_progress: F) -> Result<LocalModelReport, String>
516: where
517:     F: Fn(u64, u64, f64) + Send + 'static,
518: {
519:     let target = get_target_model_path()?;
520:     let temp_target = target.with_extension("downloading");
521:     let _ = fs::remove_file(&temp_target);
522: 
523:     let client = reqwest::Client::builder()
524:         .connect_timeout(Duration::from_secs(15))
525:         .timeout(Duration::from_secs(1800))
526:         .build()
527:         .map_err(|e| e.to_string())?;
528: 
529:     let mut resp = client
530:         .get(MODEL_DOWNLOAD_URL)
531:         .send()
532:         .await
533:         .map_err(|e| format!("Richiesta download fallita: {}", e))?;
534: 
535:     if !resp.status().is_success() {
536:         return Err(format!("Download fallito con stato HTTP: {}", resp.status()));
537:     }
538: 
539:     let total_bytes = resp
540:         .content_length()
541:         .unwrap_or(EXPECTED_MODEL_SIZE_BYTES);
542: 
543:     let mut out = fs::File::create(&temp_target).map_err(|e| e.to_string())?;
544: 
545:     let mut downloaded: u64 = 0;
546:     let mut hasher = Sha256::new();
547:     let mut last_emit = Instant::now();
548: 
549:     while let Some(chunk) = resp.chunk().await.map_err(|e| format!("Errore lettura chunk: {}", e))? {
550:         out.write_all(&chunk).map_err(|e| e.to_string())?;
551:         hasher.update(&chunk);
552:         downloaded += chunk.len() as u64;
553: 
554:         if last_emit.elapsed() >= Duration::from_millis(200) || (total_bytes > 0 && downloaded == total_bytes) {
555:             let pct = if total_bytes > 0 {
556:                 (downloaded as f64 / total_bytes as f64) * 100.0
557:             } else {
558:                 0.0
559:             };
560:             on_progress(downloaded, total_bytes, pct);
561:             last_emit = Instant::now();
562:         }
563:     }
564: 
565:     out.flush().map_err(|e| e.to_string())?;
566:     drop(out);

568:     let final_hash = format!("{:x}", hasher.finalize());

575:     }

578: 
579:     let dest_meta = fs::metadata(&target).map_err(|e| e.to_string())?;
580:     let dest_mtime_ms = dest_meta
581:         .modified()
582:         .ok()
583:         .and_then(|t| t.duration_since(std::time::UNIX_EPOCH).ok())
584:         .map(|d| d.as_millis() as u64)
585:         .unwrap_or(0);
586:     let target_str = target.to_string_lossy().to_string();

588:     if let Ok(cache_file) = get_model_cache_meta_path() {

591:             bytes: dest_meta.len(),
592:             mtime_ms: dest_mtime_ms,
593:             sha256: final_hash,
594:             sha256_ok: true,
595:             verified_at: chrono::Utc::now().to_rfc3339(),
596:         });
597:     }
598:

601:         path: target_str,
602:         bytes: dest_meta.len(),
603:         sha256_ok: true,

769: /// Esegue un comando esterno con limite di tempo garantito, terminando il processo in caso di scadenza.
770: pub fn run_command_with_timeout(
771:     mut cmd: std::process::Command,
772:     timeout: Duration,
773: ) -> Option<std::process::Output> {
774:     let mut child = cmd
775:         .stdout(std::process::Stdio::piped())
776:         .stderr(std::process::Stdio::piped())
777:         .spawn()
778:         .ok()?;
779: 
780:     let t_start = Instant::now();
781:     let poll_interval = Duration::from_millis(30);
782:     while t_start.elapsed() < timeout {
783:         match child.try_wait() {
784:             Ok(Some(_status)) => {
785:                 return child.wait_with_output().ok();
786:             }
787:             Ok(None) => {
788:                 std::thread::sleep(poll_interval);
789:             }
790:             Err(_) => {
791:                 let _ = child.kill();
792:                 return None;
793:             }
794:         }
795:     }
796: 
797:     // Timeout scaduto: termina forzatamente il processo
798:     let _ = child.kill();
799:     #[cfg(windows)]
800:     {
801:         let pid = child.id();
802:         unsafe {
803:             let handle = windows_sys::Win32::System::Threading::OpenProcess(
804:                 windows_sys::Win32::System::Threading::PROCESS_TERMINATE,
805:                 0,
806:                 pid,
807:             );
808:             if !handle.is_null() {
809:                 windows_sys::Win32::System::Threading::TerminateProcess(handle, 1);
810:                 windows_sys::Win32::Foundation::CloseHandle(handle);
811:             }
812:         }
813:     }
814:     let _ = child.try_wait();
815:     None
816: }

2057:         let (child, owned_pid) = {
