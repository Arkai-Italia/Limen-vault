47: fn increment_catalog_disk_read_count(vault_path: &Path) {
48:     CATALOG_DISK_READ_COUNT.fetch_add(1, Ordering::SeqCst);
49:     let mut guard = CATALOG_DISK_READ_COUNTS.lock().unwrap();
50:     let entry = guard.entry(vault_path.to_path_buf()).or_insert(0);
51:     *entry += 1;
52: }

225: /// Load the catalog as an Arc<CatalogState>, using thread-safe in-memory caching.
226: pub fn load_catalog_arc(vault_path: &Path) -> Result<Arc<CatalogState>, String> {
227:     let r = root(vault_path)?;
228:     let sys = match child(&r, "00_SYSTEM") {
229:         Ok(d) => d,
230:         Err(_) => return Ok(Arc::new(CatalogState::default())),
231:     };
232:     if !names(&sys)?.iter().any(|s| s == CATALOG_FILE) {
233:         return Ok(Arc::new(CatalogState::default()));
234:     }
235: 
236:     let cat_file = vault_path.join("00_SYSTEM").join(CATALOG_FILE);
237:     let current_rev = get_catalog_revision(vault_path);
238: 
239:     if let Ok(meta) = std::fs::metadata(&cat_file) {
240:         let size = meta.len();
241:         let mtime = meta.modified().unwrap_or(SystemTime::UNIX_EPOCH);
242:         if let Ok(guard) = CATALOG_CACHE.lock() {
243:             if let Some(entry) = guard.get(vault_path) {
244:                 let is_high_prec = crate::ai::is_high_precision_fs(vault_path);
245:                 let is_mtime_match = if is_high_prec {
246:                     entry.mtime == mtime
247:                 } else {
248:                     entry.mtime == mtime
249:                         || entry.mtime.duration_since(mtime).map(|d| d.as_millis() < 2000).unwrap_or(false)
250:                         || mtime.duration_since(entry.mtime).map(|d| d.as_millis() < 2000).unwrap_or(false)
251:                 };
252:                 if entry.size == size && is_mtime_match && entry.revision == current_rev {
253:                     return Ok(entry.data.clone());
254:                 }
255:             }
256:         }
257:     }
258: 
259:     increment_catalog_disk_read_count(vault_path);
260:     let bytes = read(&sys, CATALOG_FILE)?;
261:     if bytes.len() > 32 * 1024 * 1024 {
262:         return Err("Catalog file exceeds 32 MB".into());
263:     }
264:     let catalog: CatalogState = serde_json::from_slice(&bytes).map_err(err)?;
265:     let data_arc = Arc::new(catalog);
266: 
267:     if let Ok(meta) = std::fs::metadata(&cat_file) {
268:         let size = meta.len();
269:         let mtime = meta.modified().unwrap_or(SystemTime::UNIX_EPOCH);
270:         let current_rev = get_catalog_revision(vault_path);
271:         if let Ok(mut guard) = CATALOG_CACHE.lock() {
272:             guard.insert(
273:                 vault_path.to_path_buf(),
274:                 CatalogCacheEntry {
275:                     vault_path: vault_path.to_path_buf(),
276:                     size,
277:                     mtime,
278:                     revision: current_rev,
279:                     data: data_arc.clone(),
280:                 },
281:             );
282:         }
283:     }
284: 
285:     Ok(data_arc)
286: }

288: /// Load the catalog from 00_SYSTEM/VAULT_CATALOG.json or return Default if missing (cloned from cache).
289: pub fn load_catalog(vault_path: &Path) -> Result<CatalogState, String> {

293: /// Atomically save the catalog to 00_SYSTEM/VAULT_CATALOG.json
294: pub fn save_catalog(vault_path: &Path, catalog: &mut CatalogState) -> Result<(), String> {
295:     catalog.catalog_revision += 1;
296:     catalog.last_updated_at = now_iso();
297:     let r = root(vault_path)?;
298:     let sys = child(&r, "00_SYSTEM")?;
299:     let tmp = format!(".catalog-{}.tmp", crate::ai::random_token()?);
300:     let bytes = serde_json::to_vec_pretty(&catalog).map_err(err)?;
301:     write_new(&sys, &tmp, &bytes)?;
302:     let res = sys.rename(&tmp, &sys, CATALOG_FILE).map_err(err);
303:     if res.is_err() {
304:         let _ = sys.remove_file(&tmp);
305:         return res;
306:     }
307:     bump_catalog_revision(vault_path);
308:     let rev = get_catalog_revision(vault_path);
309:     let arc_cat = Arc::new(catalog.clone());
310:     let cat_file = vault_path.join("00_SYSTEM").join(CATALOG_FILE);
311:     if let Ok(meta) = std::fs::metadata(&cat_file) {
312:         let size = meta.len();
313:         let mtime = meta.modified().unwrap_or(SystemTime::UNIX_EPOCH);
314:         if let Ok(mut guard) = CATALOG_CACHE.lock() {
315:             guard.insert(
316:                 vault_path.to_path_buf(),
317:                 CatalogCacheEntry {
318:                     vault_path: vault_path.to_path_buf(),
319:                     size,
320:                     mtime,
321:                     revision: rev,
322:                     data: arc_cat,
323:                 },
324:             );
325:         }
326:     }
327:     Ok(())
328: }

335: /// Chunk text into overlapping passages with locators
336: pub fn chunk_text_to_passages(document_id: &str, text: &str) -> Vec<DocumentPassage> {
337:     let mut passages = Vec::new();
338:     if text.trim().is_empty() {
339:         return passages;
340:     }
341: 
342:     let target_chunk: usize = 1200;
343:     let overlap_chars_count: usize = 150;
344: 
345:     let normalized_text = text.replace("\r\n", "\n");
346:     let raw_paragraphs: Vec<&str> = normalized_text.split("\n\n").filter(|p| !p.trim().is_empty()).collect();
347: 
348:     struct Segment {
349:         text: String,
350:         locator: Option<String>,
351:         paragraph_num: usize,
352:     }
353: 
354:     let mut segments: Vec<Segment> = Vec::new();
355:     let mut current_page_or_slide: Option<String> = None;
356: 
357:     for (p_idx, p) in raw_paragraphs.iter().enumerate() {
358:         let trimmed = p.trim();
359:         if trimmed.starts_with("## Pagina ") {
360:             let page_num = trimmed.trim_start_matches("## Pagina ").trim();
361:             current_page_or_slide = Some(format!("Pagina {}", page_num));
362:         } else if trimmed.starts_with("## Slide ") {
363:             let slide_num = trimmed.trim_start_matches("## Slide ").trim();
364:             current_page_or_slide = Some(format!("Slide {}", slide_num));
365:         }
366: 
367:         // If a single paragraph is longer than target_chunk - overlap, split into smaller non-overlapping segments
368:         let max_seg_len = target_chunk.saturating_sub(overlap_chars_count);
369:         if p.chars().count() > max_seg_len {
370:             let chars: Vec<char> = p.chars().collect();
371:             let mut start = 0;
372:             while start < chars.len() {
373:                 let mut end = if start + max_seg_len < chars.len() {
374:                     start + max_seg_len
375:                 } else {
376:                     chars.len()
377:                 };
378: 
379:                 // Try to find a sentence boundary (. or \n or ! or ?)
380:                 if end < chars.len() {
381:                     let min_bound = start + max_seg_len / 2;
382:                     let mut found = false;
383:                     for i in (min_bound..=end).rev() {
384:                         if chars[i] == '.' || chars[i] == '\n' || chars[i] == '!' || chars[i] == '?' {
385:                             end = i + 1;
386:                             found = true;
387:                             break;
388:                         }
389:                     }
390:                     if !found {
391:                         for i in (min_bound..=end).rev() {
392:                             if chars[i] == ' ' {
393:                                 end = i + 1;
394:                                 break;
395:                             }
396:                         }
397:                     }
398:                 }
399: 
400:                 let slice: String = chars[start..end].iter().collect();
401:                 if !slice.trim().is_empty() {
402:                     segments.push(Segment {
403:                         text: slice,
404:                         locator: current_page_or_slide.clone(),
405:                         paragraph_num: p_idx + 1,
406:                     });
407:                 }
408: 
409:                 if end >= chars.len() {
410:                     break;
411:                 }
412:                 start = end;
413:             }
414:         } else {
415:             segments.push(Segment {
416:                 text: p.to_string(),
417:                 locator: current_page_or_slide.clone(),
418:                 paragraph_num: p_idx + 1,
419:             });
420:         }
421:     }
422: 
423:     if segments.is_empty() {
424:         return passages;
425:     }
426: 
427:     let mut current_chunk = String::new();
428:     let mut cur_locator: Option<String> = None;
429:     let mut start_p_num = 1;
430:     let mut end_p_num = 1;
431:     let mut p_idx = 0;
432: 
433:     for seg in &segments {
434:         if current_chunk.is_empty() {
435:             start_p_num = seg.paragraph_num;
436:             end_p_num = seg.paragraph_num;
437:             cur_locator = seg.locator.clone();
438:         }
439: 
440:         let page_or_slide_changed = seg.locator.is_some() && cur_locator.is_some() && seg.locator != cur_locator;
441:         let chunk_oversized = current_chunk.chars().count() + seg.text.chars().count() > target_chunk;
442: 
443:         if (page_or_slide_changed || chunk_oversized) && !current_chunk.is_empty() {
444:             let chunk_str = current_chunk.trim().to_string();
445:             let hash = compute_sha256(chunk_str.as_bytes());
446:             let locator = if let Some(ref loc) = cur_locator {
447:                 loc.clone()
448:             } else if start_p_num == end_p_num {
449:                 format!("Paragrafo {}", start_p_num)
450:             } else {
451:                 format!("Paragrafi {}-{}", start_p_num, end_p_num)
452:             };
453: 
454:             passages.push(DocumentPassage {
455:                 passage_id: format!("{}_p{}", document_id, p_idx),
456:                 locator,
457:                 char_count: chunk_str.chars().count(),
458:                 sha256: hash,
459:                 text: chunk_str.clone(),
460:             });
461:             p_idx += 1;
462: 
463:             // Carry over overlap only if within the same page/slide section
464:             let chunk_chars: Vec<char> = chunk_str.chars().collect();
465:             let overlap_chars = if !page_or_slide_changed && chunk_chars.len() > overlap_chars_count {
466:                 chunk_chars[chunk_chars.len() - overlap_chars_count..].iter().collect::<String>()
467:             } else {
468:                 String::new()
469:             };
470:             current_chunk = overlap_chars;
471:             start_p_num = seg.paragraph_num;
472:             cur_locator = seg.locator.clone();
473:         }
474: 
475:         if !current_chunk.is_empty() {
476:             current_chunk.push_str("\n\n");
477:         }
478:         current_chunk.push_str(&seg.text);
479:         end_p_num = seg.paragraph_num;
480:         if cur_locator.is_none() && seg.locator.is_some() {
481:             cur_locator = seg.locator.clone();
482:         }
483:     }
484: 
485:     if !current_chunk.trim().is_empty() {
486:         let chunk_str = current_chunk.trim().to_string();
487:         let hash = compute_sha256(chunk_str.as_bytes());
488:         let locator = if let Some(ref loc) = cur_locator {
489:             loc.clone()
490:         } else if start_p_num == end_p_num {
491:             format!("Paragrafo {}", start_p_num)
492:         } else {
493:             format!("Paragrafi {}-{}", start_p_num, end_p_num)
494:         };
495: 
496:         passages.push(DocumentPassage {
497:             passage_id: format!("{}_p{}", document_id, p_idx),
498:             locator,
499:             char_count: chunk_str.chars().count(),
500:             sha256: hash,
501:             text: chunk_str,
502:         });
503:     }
504: 
505:     passages
506: }

508: fn collect_entries_recursive(
509:     d: &cap_std::fs::Dir,
510:     prefix: &str,
511:     depth: usize,
512:     out: &mut Vec<(String, String, Vec<u8>)>,
513: ) -> Result<(), String> {
514:     if depth > 32 {
515:         return Ok(());
516:     }
517:     for name in names(d)? {
518:         if name.starts_with('.') {
519:             continue;
520:         }
521:         let rel = if prefix.is_empty() {
522:             name.clone()
523:         } else {
524:             format!("{}/{}", prefix, name)
525:         };
526:         let m = d.symlink_metadata(&name).map_err(err)?;
527:         if m.is_symlink() {
528:             continue;
529:         }
530:         if m.is_dir() {
531:             if let Ok(sub) = child(d, &name) {
532:                 let _ = collect_entries_recursive(&sub, &rel, depth + 1, out);
533:             }
534:         } else if m.is_file() {
535:             if let Ok(bytes) = read(d, &name) {
536:                 out.push((rel, name, bytes));
537:             }
538:         }
539:     }
540:     Ok(())
541: }

543: /// Scan RAW sources, migrate existing AUTO_KNOWLEDGE jobs if any, and reconcile the catalog.
544: /// Does not delete original files or alter legacy proposal status.
545: pub fn sync_catalog_from_vault(vault_path: &Path) -> Result<CatalogState, String> {
546:     let mut catalog = load_catalog(vault_path)?;
547:     let r = root(vault_path)?;
548:     let mut discovered_paths = std::collections::BTreeSet::new();
549: 
550:     // 1. Scan 20_RAW_SOURCES with content-hash deduplication and alias preservation
551:     if let Ok(raw_dir) = child(&r, "20_RAW_SOURCES") {
552:         let mut raw_entries = Vec::new();
553:         let _ = collect_entries_recursive(&raw_dir, "", 0, &mut raw_entries);
554:         for (sub_rel, name, bytes) in raw_entries {
555:             let rel_path = format!("20_RAW_SOURCES/{}", sub_rel);
556:             let hash = compute_sha256(&bytes);
557:             discovered_paths.insert(rel_path.clone());
558: 
559:             let doc_id = make_document_id(&rel_path);
560:             let existing_match = catalog
561:                 .documents
562:                 .values_mut()
563:                 .find(|d| d.original_path == rel_path || d.document_id == doc_id);
564: 
565:             if let Some(existing) = existing_match {
566:                 if existing.content_hash != hash {
567:                     // Content changed: keep stable identity, increment revision, reset extraction
568:                     existing.content_hash = hash;
569:                     existing.file_size = bytes.len() as u64;
570:                     existing.revision += 1;
571:                     existing.updated_at = now_iso();
572:                     existing.extraction_status = ExtractionStatus::Pending;
573:                     existing.extraction_error = None;
574:                     existing.extracted_text_path = None;
575:                     existing.extracted_text_hash = None;
576:                     existing.passages.clear();
577:                     existing.lexical_status = PhaseInfo::default();
578:                     existing.semantic_status = PhaseInfo::default();
579:                     existing.classification_status = PhaseInfo::default();
580:                     existing.wiki_status = PhaseInfo::default();
581:                 }
582:                 continue;
583:             }
584: 
585:             // Deduplication: if another document has identical content hash, record alias
586:             let alias_match = catalog
587:                 .documents
588:                 .values_mut()
589:                 .find(|d| d.content_hash == hash);
590: 
591:             if let Some(existing_alias) = alias_match {
592:                 if existing_alias.original_path != rel_path && !existing_alias.aliases.contains(&rel_path) {
593:                     existing_alias.aliases.push(rel_path.clone());
594:                     existing_alias.updated_at = now_iso();
595:                 }
596:                 continue;
597:             }
598: 
599:             let ext = Path::new(&name)
600:                 .extension()
601:                 .and_then(|e| e.to_str())
602:                 .unwrap_or("")
603:                 .to_lowercase();
604:             let mime = match ext.as_str() {
605:                 "pdf" => "application/pdf",

907: fn extract_plain_text(bytes: &[u8], ext: &str) -> Result<String, String> {
908:     let text = if bytes.starts_with(&[0xff, 0xfe]) || bytes.starts_with(&[0xfe, 0xff]) {
909:         if bytes.len() % 2 != 0 {
910:             return Err("Testo UTF-16 incompleto".into());
911:         }
912:         let le = bytes[0] == 0xff;
913:         let units: Vec<_> = bytes[2..]
914:             .chunks_exact(2)
915:             .map(|b| {
916:                 if le {
917:                     u16::from_le_bytes([b[0], b[1]])
918:                 } else {
919:                     u16::from_be_bytes([b[0], b[1]])
920:                 }
921:             })
922:             .collect();
923:         String::from_utf16(&units).map_err(|_| "Testo UTF-16 non valido")?
924:     } else {
925:         String::from_utf8(bytes.to_vec()).map_err(|_| "Codifica testo non UTF-8")?
926:     };
927:     let text = text.trim_start_matches('\u{feff}').replace("\r\n", "\n");
928:     let text = if ext == "html" || ext == "htm" {
929:         crate::compiler::sanitize_html(&text)
930:     } else {
931:         text
932:     };
933:     if text.trim().is_empty() {
934:         return Err("Documento privo di testo leggibile".into());
935:     }
936:     Ok(text)
937: }

939: fn extract_spreadsheet_content(bytes: &[u8]) -> Result<String, String> {
940:     use calamine::Reader;
941:     let mut workbook = calamine::open_workbook_auto_from_rs(std::io::Cursor::new(bytes.to_vec()))
942:         .map_err(|_| "Foglio di calcolo non leggibile o protetto")?;
943:     let mut output = String::new();
944:     for sheet in workbook.sheet_names() {
945:         let range = workbook
946:             .worksheet_range(&sheet)
947:             .map_err(|_| "Impossibile leggere tutte le celle del foglio")?;
948:         output.push_str(&format!("## Foglio: {sheet}\n\n"));
949:         for (row, col, value) in range.cells() {
950:             if value.to_string().is_empty() {
951:                 continue;
952:             }
953:             output.push_str(&format!(
954:                 "- Riga {}, colonna {}: {}\n",
955:                 row + range.start().map(|p| p.0 as usize).unwrap_or(0) + 1,
956:                 col + range.start().map(|p| p.1 as usize).unwrap_or(0) + 1,
957:                 value
958:             ));
959:             if output.len() > 16 * 1024 * 1024 {
960:                 return Err("Contenuto estratto oltre il limite di 16 MB".into());
961:             }
962:         }
963:     }
964:     if output.trim().is_empty() {
965:         return Err("Foglio privo di contenuto".into());
966:     }
967:     Ok(output)
968: }

985: /// Extract and generate passages for any document in Pending status.
986: /// Operates completely OFFLINE without any API key or network connection!
987: pub fn process_pending_extractions(vault_path: &Path) -> Result<usize, String> {
988:     let mut catalog = load_catalog(vault_path)?;
989:     let mut processed = 0;
990:     let r = root(vault_path)?;
991: 
992:     for doc in catalog.documents.values_mut() {
993:         if doc.extraction_status != ExtractionStatus::Pending {
994:             continue;
995:         }
996: 
997:         doc.extraction_status = ExtractionStatus::Processing;
998:         let original_bytes = match read_path(&r, &doc.original_path) {
999:             Ok(b) => b,
1000:             Err(e) => {
1001:                 doc.extraction_status = ExtractionStatus::Failed;
1002:                 doc.extraction_error = Some(format!("Impossibile leggere l'originale: {}", e));
1003:                 continue;
1004:             }
1005:         };
1006: 
1007:         match extract_document_content(&original_bytes, &doc.file_name) {
1008:             Ok(text) => {
1009:                 let trimmed = text.trim().to_string();
1010:                 if trimmed.is_empty() {
1011:                     doc.extraction_status = ExtractionStatus::Ready;
1012:                     doc.passages.clear();
1013:                 } else {
1014:                     let hash = compute_sha256(trimmed.as_bytes());
1015:                     doc.extracted_text_hash = Some(hash);
1016:                     doc.passages = chunk_text_to_passages(&doc.document_id, &trimmed);
1017:                     doc.extraction_status = ExtractionStatus::Ready;
1018:                     doc.extraction_error = None;
1019:                 }
1020:                 doc.updated_at = now_iso();
1021:                 processed += 1;
1022:             }
1023:             Err(e) => {
1024:                 if e.contains("corrotto") || e.contains("protetto") {
1025:                     doc.extraction_status = ExtractionStatus::Protected;
1026:                 } else if e.contains("Formato") || e.contains("non consentito") {
1027:                     doc.extraction_status = ExtractionStatus::Unsupported;
1028:                 } else {
1029:                     doc.extraction_status = ExtractionStatus::Failed;
1030:                 }
1031:                 doc.extraction_error = Some(e);
1032:                 doc.updated_at = now_iso();
1033:             }
1034:         }
1035:     }
1036: 
1037:     if processed > 0 || catalog.documents.values().any(|d| d.extraction_status == ExtractionStatus::Processing) {
1038:         save_catalog(vault_path, &mut catalog)?;
1039:     }
1040: 
1041:     Ok(processed)
1042: }

1044: /// List documents matching options with pagination
1045: pub fn list_documents(vault_path: &Path, options: CatalogListOptions) -> Result<CatalogListResponse, String> {
1046:     let catalog = load_catalog_arc(vault_path)?;
1047:     let total = catalog.documents.len();
1048: 
1049:     let filter_term = options.filter.as_deref().map(|s| s.to_lowercase());
1050:     let cat_filter = options.category.as_deref();
1051:     let status_filter = options.status.as_deref();
1052:     let client_filter = options.client.as_deref().map(|s| s.to_lowercase());
1053:     let project_filter = options.project.as_deref().map(|s| s.to_lowercase());
1054:     let format_filter = options.format.as_deref().map(|s| s.to_lowercase());
1055: 
1056:     let mut filtered: Vec<DocumentRecord> = catalog
1057:         .documents
1058:         .values()
1059:         .cloned()
1060:         .filter(|doc| {
1061:             if let Some(ref term) = filter_term {
1062:                 let match_name = doc.file_name.to_lowercase().contains(term);
1063:                 let match_path = doc.original_path.to_lowercase().contains(term);
1064:                 let match_passages = doc.passages.iter().any(|p| p.text.to_lowercase().contains(term));
1065:                 if !match_name && !match_path && !match_passages {
1066:                     return false;
1067:                 }
1068:             }
1069:             if let Some(cat) = cat_filter {
1070:                 if cat != "all" && doc.category.as_deref() != Some(cat) {
1071:                     return false;
1072:                 }
1073:             }
1074:             if let Some(st) = status_filter {
1075:                 let st_str = match doc.extraction_status {
1076:                     ExtractionStatus::Ready => "ready",
1077:                     ExtractionStatus::Processing => "processing",
1078:                     ExtractionStatus::Pending => "pending",
1079:                     ExtractionStatus::Unsupported => "unsupported",
1080:                     ExtractionStatus::Protected => "protected",
1081:                     ExtractionStatus::Failed => "failed",
1082:                 };
1083:                 if st != "all" && st != st_str {
1084:                     return false;
1085:                 }
1086:             }
1087:             if let Some(ref fmt) = format_filter {
1088:                 if fmt != "all" && doc.extension.to_lowercase() != *fmt {
1089:                     return false;
1090:                 }
1091:             }
1092:             if let Some(ref cl) = client_filter {
1093:                 if doc.client.as_ref().map(|s| s.to_lowercase()) != Some(cl.clone()) {
1094:                     return false;
1095:                 }
1096:             }
1097:             if let Some(ref pr) = project_filter {
1098:                 if doc.project.as_ref().map(|s| s.to_lowercase()) != Some(pr.clone()) {
1099:                     return false;
1100:                 }
1101:             }
1102:             true
1103:         })
1104:         .collect();
1105: 
1106:     // Sort: newest first
1107:     filtered.sort_by(|a, b| b.imported_at.cmp(&a.imported_at));
1108: 
1109:     let filtered_total = filtered.len();
1110:     let offset = options.offset.unwrap_or(0);
1111:     let limit = options.limit.unwrap_or(50);
1112: 
1113:     let paged = filtered.into_iter().skip(offset).take(limit).collect();
1114: 
1115:     Ok(CatalogListResponse {
1116:         total,
1117:         filtered_total,
1118:         catalog_revision: catalog.catalog_revision,
1119:         documents: paged,
1120:     })
1121: }
